## 📖 约束概述

### 定义

**优先级约束**（Priority Constraint）规定任务、资源、请求的处理顺序，确保高优先级项目优先获得处理。

**数学定义**：

```
对于任务集合 T，定义优先级函数 p: T → ℝ

排序规则:
  对于任务 i, j，如果 p(i) > p(j)
  则 i 在调度顺序中排在 j 之前

抢占规则:
  高优先级任务可以中断低优先级任务的执行
```

### 优先级类型

#### 1. **静态优先级** (Static Priority)

```
优先级在任务创建时确定，执行过程中不变

例如:
  VIP客户订单: priority = 10
  普通客户订单: priority = 5
  批量处理任务: priority = 1
```

#### 2. **动态优先级** (Dynamic Priority)

```
优先级随时间、状态变化

计算公式:
  p(t) = base_priority + urgency_factor(t) + aging_factor(t)

例如:
  urgency_factor(t) = (deadline - t) / processing_time
  aging_factor(t) = waiting_time / 10
```

#### 3. **抢占式优先级** (Preemptive Priority)

```
高优先级任务可以中断正在执行的低优先级任务

规则:
  if p(new_task) > p(current_task):
      suspend(current_task)
      execute(new_task)
```

#### 4. **非抢占式优先级** (Non-Preemptive Priority)

```
高优先级任务必须等待当前任务完成

规则:
  等待当前任务完成
  然后按优先级顺序调度下一个任务
```

## 💡 核心理论

### 优先级调度算法

#### 1. 最高优先级优先 (Highest Priority First, HPF)

```python
def hpf_schedule(tasks):
    """
    最高优先级优先调度

    步骤:
      1. 按优先级排序任务
      2. 依次调度最高优先级任务
    """
    # 按优先级降序排序
    sorted_tasks = sorted(tasks, key=lambda t: t.priority, reverse=True)

    schedule = []
    current_time = 0

    for task in sorted_tasks:
        schedule.append({
            'task_id': task.id,
            'start_time': current_time,
            'end_time': current_time + task.duration
        })
        current_time += task.duration

    return schedule
```

#### 2. 加权最短处理时间 (Weighted Shortest Processing Time, WSPT)

```python
def wspt_schedule(tasks):
    """
    加权最短处理时间调度

    优先级 = weight / processing_time
    最大化加权完工时间性能

    理论保证: 最小化总加权完工时间
    """
    # 按 weight/duration 降序排序
    sorted_tasks = sorted(
        tasks,
        key=lambda t: t.weight / t.duration if t.duration > 0 else 0,
        reverse=True
    )

    schedule = []
    current_time = 0

    for task in sorted_tasks:
        schedule.append({
            'task_id': task.id,
            'start_time': current_time,
            'end_time': current_time + task.duration,
            'completion_time': current_time + task.duration,
            'weighted_completion': (current_time + task.duration) * task.weight
        })
        current_time += task.duration

    return schedule
```

### 多准则优先级 - AHP层次分析法

```python
import numpy as np

def ahp_priority(criteria_matrix):
    """
    层次分析法计算权重

    参数:
      criteria_matrix: n×n 成对比较矩阵
                      element[i][j] = criteria_i相对于criteria_j的重要性
                      1: 同等重要
                      3: 稍微重要
                      5: 明显重要
                      7: 强烈重要
                      9: 极端重要

    返回:
      weights: 各准则权重向量
      consistency_ratio: 一致性比率 (< 0.1 为可接受)
    """
    n = len(criteria_matrix)
    A = np.array(criteria_matrix)

    # 计算特征值和特征向量
    eigenvalues, eigenvectors = np.linalg.eig(A)

    # 最大特征值对应的特征向量
    max_index = np.argmax(eigenvalues)
    max_eigenvalue = eigenvalues[max_index].real
    weights = eigenvectors[:, max_index].real

    # 归一化权重
    weights = weights / weights.sum()

    # 一致性检验
    CI = (max_eigenvalue - n) / (n - 1)  # 一致性指标

    # 随机一致性指标
    RI = {1: 0, 2: 0, 3: 0.58, 4: 0.90, 5: 1.12,
          6: 1.24, 7: 1.32, 8: 1.41, 9: 1.45}

    CR = CI / RI[n] if n in RI else 0  # 一致性比率

    return weights, CR

# 示例：客户订单优先级评估
criteria_matrix = [
    [1,   3,   5],  # 客户等级 vs 紧急程度 vs 订单金额
    [1/3, 1,   3],  # 紧急程度相对重要性
    [1/5, 1/3, 1]   # 订单金额相对重要性
]

weights, cr = ahp_priority(criteria_matrix)
# weights = [0.63, 0.26, 0.11] → 客户等级最重要
# cr < 0.1 → 判断一致性可接受
```

### 老化因子 (Aging Factor)

```python
def calculate_dynamic_priority(task, current_time):
    """
    动态优先级计算

    优先级随等待时间增加，防止饥饿
    """
    base_priority = task.priority

    # 等待时间
    waiting_time = current_time - task.arrival_time

    # 老化因子：等待越久，优先级越高
    aging_factor = waiting_time / 60  # 每小时增加1

    # 紧急度因子
    if task.deadline:
        time_to_deadline = task.deadline - current_time
        urgency_factor = max(0, 100 / time_to_deadline) if time_to_deadline > 0 else 1000
    else:
        urgency_factor = 0

    # 综合优先级
    dynamic_priority = base_priority + aging_factor + urgency_factor

    return dynamic_priority
```

## 🔧 代码实现

```python
from typing import Dict, List, Tuple, Optional
from dataclasses import dataclass, field
from datetime import datetime, timedelta
from enum import Enum
import heapq

class PriorityType(Enum):
    """优先级类型"""
    STATIC = "static"           # 静态优先级
    DYNAMIC = "dynamic"         # 动态优先级
    PREEMPTIVE = "preemptive"   # 抢占式
    NON_PREEMPTIVE = "non_preemptive"  # 非抢占式

@dataclass
class Task:
    """任务定义"""
    id: str
    priority: int                    # 基础优先级 (数值越大越优先)
    duration: timedelta
    arrival_time: datetime = field(default_factory=datetime.now)
    deadline: Optional[datetime] = None
    weight: float = 1.0              # 重要性权重
    is_preemptible: bool = True      # 是否可抢占

    def __lt__(self, other):
        """支持优先队列排序（数值大的优先）"""
        return self.priority > other.priority

@dataclass
class ScheduledTask:
    """已调度任务"""
    task: Task
    start_time: datetime
    end_time: datetime
    was_preempted: bool = False

class PriorityConstraintHandler:
    """优先级约束处理器"""

    def __init__(self, priority_type: PriorityType = PriorityType.STATIC):
        """
        参数:
          priority_type: 优先级类型
        """
        self.priority_type = priority_type

    def validate(self, schedule: List[ScheduledTask]) -> Tuple[bool, List[Dict]]:
        """
        验证调度是否满足优先级约束

        检查:
          1. 高优先级任务是否比低优先级任务早开始
          2. 是否存在优先级反转

        返回:
          (is_valid, violations)
        """
        violations = []

        for i, task1 in enumerate(schedule):
            for j, task2 in enumerate(schedule[i + 1:], start=i + 1):
                # 检查优先级顺序
                if task1.task.priority < task2.task.priority:
                    # 低优先级任务在高优先级任务之前开始
                    if task1.start_time < task2.start_time:
                        violations.append({
                            'type': 'priority_inversion',
                            'low_priority_task': task1.task.id,
                            'high_priority_task': task2.task.id,
                            'low_priority': task1.task.priority,
                            'high_priority': task2.task.priority,
                            'low_start': task1.start_time,
                            'high_start': task2.start_time,
                            'penalty': (task2.task.priority - task1.task.priority) * 10
                        })

        return len(violations) == 0, violations

    def hpf_schedule(self, tasks: List[Task],
                    start_time: datetime) -> List[ScheduledTask]:
        """
        最高优先级优先调度（非抢占式）

        参数:
          tasks: 任务列表
          start_time: 开始时间

        返回:
          schedule: 调度结果
        """
        # 按优先级降序排序
        sorted_tasks = sorted(tasks, key=lambda t: t.priority, reverse=True)

        schedule = []
        current_time = start_time

        for task in sorted_tasks:
            end_time = current_time + task.duration

            schedule.append(ScheduledTask(
                task=task,
                start_time=current_time,
                end_time=end_time
            ))

            current_time = end_time

        return schedule

    def preemptive_schedule(self, tasks: List[Task],
                           start_time: datetime) -> List[ScheduledTask]:
        """
        抢占式优先级调度

        参数:
          tasks: 任务列表
          start_time: 开始时间

        返回:
          schedule: 调度结果
        """
        schedule = []
        current_time = start_time

        # 按到达时间排序
        events = sorted(tasks, key=lambda t: t.arrival_time)

        ready_queue = []  # 优先队列
        completed = set()
        current_task = None
        current_task_remaining = timedelta(0)

        event_index = 0

        while event_index < len(events) or ready_queue or current_task:
            # 处理新到达的任务
            while event_index < len(events) and events[event_index].arrival_time <= current_time:
                new_task = events[event_index]
                heapq.heappush(ready_queue, (-new_task.priority, new_task.id, new_task))

                # 检查是否需要抢占
                if current_task and new_task.priority > current_task.priority:
                    if current_task.is_preemptible:
                        # 抢占当前任务
                        schedule.append(ScheduledTask(
                            task=current_task,
                            start_time=current_time - current_task_remaining,
                            end_time=current_time,
                            was_preempted=True
                        ))

                        # 将剩余部分放回队列
                        preempted_task = Task(
                            id=current_task.id,
                            priority=current_task.priority,
                            duration=current_task_remaining,
                            arrival_time=current_time,
                            deadline=current_task.deadline,
                            weight=current_task.weight,
                            is_preemptible=current_task.is_preemptible
                        )
                        heapq.heappush(ready_queue, (-preempted_task.priority, preempted_task.id, preempted_task))
                        current_task = None
                        current_task_remaining = timedelta(0)

                event_index += 1

            # 选择下一个任务
            if not current_task and ready_queue:
                _, _, current_task = heapq.heappop(ready_queue)
                current_task_remaining = current_task.duration

            # 执行当前任务
            if current_task:
                execution_time = min(current_task_remaining, timedelta(hours=1))  # 时间片

                current_time += execution_time
                current_task_remaining -= execution_time

                # 任务完成
                if current_task_remaining == timedelta(0):
                    schedule.append(ScheduledTask(
                        task=current_task,
                        start_time=current_time - current_task.duration,
                        end_time=current_time
                    ))
                    completed.add(current_task.id)
                    current_task = None

            elif ready_queue:
                continue
            else:
                # 推进时间到下一个事件
                if event_index < len(events):
                    current_time = events[event_index].arrival_time

        return schedule

    def wspt_schedule(self, tasks: List[Task],
                     start_time: datetime) -> List[ScheduledTask]:
        """
        加权最短处理时间调度

        最小化总加权完工时间

        返回:
          schedule: 调度结果
        """
        # 按 weight/duration 降序排序
        sorted_tasks = sorted(
            tasks,
            key=lambda t: t.weight / t.duration.total_seconds() if t.duration.total_seconds() > 0 else 0,
            reverse=True
        )

        schedule = []
        current_time = start_time

        for task in sorted_tasks:
            end_time = current_time + task.duration

            schedule.append(ScheduledTask(
                task=task,
                start_time=current_time,
                end_time=end_time
            ))

            current_time = end_time

        return schedule

    def calculate_dynamic_priority(self, task: Task,
                                   current_time: datetime) -> float:
        """
        计算动态优先级

        综合考虑:
          - 基础优先级
          - 老化因子（等待时间）
          - 紧急度因子（截止期）
        """
        base_priority = task.priority

        # 等待时间（小时）
        waiting_hours = (current_time - task.arrival_time).total_seconds() / 3600

        # 老化因子：每小时增加1
        aging_factor = waiting_hours

        # 紧急度因子
        if task.deadline:
            time_to_deadline_hours = (task.deadline - current_time).total_seconds() / 3600

            if time_to_deadline_hours > 0:
                urgency_factor = 100 / time_to_deadline_hours
            else:
                urgency_factor = 1000  # 已超期，极高优先级
        else:
            urgency_factor = 0

        # 综合优先级
        dynamic_priority = base_priority + aging_factor + urgency_factor

        return dynamic_priority

    def update_priorities(self, tasks: List[Task],
                         current_time: datetime) -> List[Task]:
        """
        更新所有任务的动态优先级

        返回:
          updated_tasks: 优先级已更新的任务列表
        """
        updated = []

        for task in tasks:
            if self.priority_type == PriorityType.DYNAMIC:
                task.priority = self.calculate_dynamic_priority(task, current_time)

            updated.append(task)

        return updated

```

## 📊 应用场景

### 场景1: 客户订单优先级处理

```python
# 定义订单任务
tasks = [
    Task(id='vip_order1', priority=10, duration=timedelta(hours=2), weight=1000),
    Task(id='normal_order1', priority=5, duration=timedelta(hours=1), weight=100),
    Task(id='bulk_order', priority=1, duration=timedelta(hours=4), weight=50),
    Task(id='vip_order2', priority=10, duration=timedelta(hours=1), weight=1000),
]

handler = PriorityConstraintHandler(priority_type=PriorityType.STATIC)

# HPF调度
start_time = datetime.now()
schedule = handler.hpf_schedule(tasks, start_time)

for item in schedule:
    print(f"{item.task.id}: {item.start_time.strftime('%H:%M')} - {item.end_time.strftime('%H:%M')}")

# 验证优先级约束
is_valid, violations = handler.validate(schedule)
```

### 场景2: 加权最短处理时间调度

```python
# WSPT调度（最小化总加权完工时间）
schedule_wspt = handler.wspt_schedule(tasks, start_time)

# 计算总加权完工时间
total_weighted_completion = sum(
    (item.end_time - start_time).total_seconds() / 3600 * item.task.weight
    for item in schedule_wspt
)

print(f"总加权完工时间: {total_weighted_completion:.2f}")
```

### 场景3: 动态优先级与老化

```python
# 动态优先级处理器
dynamic_handler = PriorityConstraintHandler(priority_type=PriorityType.DYNAMIC)

# 任务等待一段时间后更新优先级
current_time = datetime.now() + timedelta(hours=2)
updated_tasks = dynamic_handler.update_priorities(tasks, current_time)

for task in updated_tasks:
    old_priority = task.priority
    new_priority = dynamic_handler.calculate_dynamic_priority(task, current_time)
    print(f"{task.id}: {old_priority} → {new_priority:.2f}")
```

### 优先队列优化

```python
import heapq

class FastPriorityQueue:
    """使用堆实现的高效优先队列"""

    def __init__(self):
        self.heap = []
        self.counter = 0

    def push(self, task: Task):
        """插入任务 O(log n)"""
        heapq.heappush(self.heap, (-task.priority, self.counter, task))
        self.counter += 1

    def pop(self) -> Task:
        """弹出最高优先级任务 O(log n)"""
        if not self.heap:
            return None
        _, _, task = heapq.heappop(self.heap)
        return task

    def peek(self) -> Task:
        """查看最高优先级任务 O(1)"""
        if not self.heap:
            return None
        return self.heap[0][2]
```
