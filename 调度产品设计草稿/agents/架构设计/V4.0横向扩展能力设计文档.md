class CompatibilityAnalyzer:
    """兼容性自动分析器"""

    def analyze_compatibility(self, source_code: str, algorithm_info: Dict) -> Dict:
        """执行兼容性分析"""

        # 1. 代码结构分析
        code_analysis = self.code_analyzer.analyze(source_code)

        # 2. 算法模式识别
        algorithm_patterns = self.pattern_recognizer.identify_patterns(code_analysis)

        # 3. 模块兼容性评估
        compatibility_scores = {
            'algorithm_module': self._evaluate_algorithm_compatibility(algorithm_patterns),
            'constraint_module': self._evaluate_constraint_compatibility(code_analysis),
            'objective_module': self._evaluate_objective_compatibility(code_analysis),
            'domain_module': self._evaluate_domain_compatibility(algorithm_info)
        }

        # 4. 综合评分计算
        overall_score = self._calculate_overall_compatibility(compatibility_scores)

        # 5. 集成策略推荐
        integration_strategy = self._recommend_integration_strategy(
            overall_score,
            compatibility_scores
        )

        return {
            'overall_compatibility': overall_score,
            'module_compatibility': compatibility_scores,
            'integration_strategy': integration_strategy,
            'estimated_effort': self._estimate_integration_effort(overall_score),
            'key_modifications': self._identify_key_modifications(code_analysis)
        }
