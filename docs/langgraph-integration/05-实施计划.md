class APSLangGraphCompiler:
    def compile(self) -> Dict:
        """主编译流程"""
        self._parse_phase()       # 解析
        self._validate_phase()    # 验证
        self._generate_phase()    # 生成
        self._write_phase()       # 写入
        self._finalize_phase()    # 收尾
