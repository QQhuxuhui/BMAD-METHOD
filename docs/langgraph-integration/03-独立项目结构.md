## 项目概览

**aps-langgraph** 是一个完全独立的 Git 仓库，包含从 BMAD-METHOD 编译生成的可运行 LangGraph 应用。

### 设计原则

1. **自包含性**: 所有运行时依赖都在项目内
2. **代码分层**: 生成代码 vs 自定义代码严格分离
3. **可测试性**: 完整的测试框架
4. **生产就绪**: Docker、CI/CD、监控配置完整

### 仓库信息

```yaml
Repository: aps-langgraph
Type: Independent Git Repository
Language: Python 3.11+
Framework: LangGraph
Deployment: Docker + Kubernetes
CI/CD: GitHub Actions
```

### 1. 应用入口 (src/app.py)

```python
"""
APS LangGraph 应用入口
自动生成自: BMAD-METHOD/bmad/aps
编译时间: 2025-10-31 10:30:00

⚠️  注意:
- 此文件首次由编译器生成
- 后续不会被自动覆盖
- 可以安全地添加自定义逻辑
"""

import os
import sys
from typing import Optional, Dict, Any
from pathlib import Path
from dotenv import load_dotenv

# 添加项目根目录到Python路径
ROOT_DIR = Path(__file__).parent.parent
sys.path.insert(0, str(ROOT_DIR))

from langgraph.graph import StateGraph, START, END
from langgraph.checkpoint.postgres import PostgresSaver
from langchain_anthropic import ChatAnthropic

# 导入生成的组件
from src.generated.state import APSState
from src.generated.agents import (
    orchestrator_node,
    algorithm_expert_node,
    constraint_expert_node,
    objective_expert_node,
    domain_expert_node,
    extension_guide_node,
    quality_evaluator_node
)
from src.generated.workflows.scheduling_orchestration import (
    build_scheduling_orchestration_graph
)

# 导入自定义扩展（可选）
try:
    from src.custom.extensions import get_custom_nodes
    HAS_CUSTOM_EXTENSIONS = True
except ImportError:
    HAS_CUSTOM_EXTENSIONS = False

# 导入自定义中间件（可选）
try:
    from src.custom.middleware import get_middleware_stack
    HAS_CUSTOM_MIDDLEWARE = True
except ImportError:
    HAS_CUSTOM_MIDDLEWARE = False

# 加载环境变量
load_dotenv()


class APSLangGraphApp:
    """
    APS LangGraph 应用主类
    """

    def __init__(
        self,
        db_uri: Optional[str] = None,
        llm_model: str = "claude-3-5-sonnet-20241022"
    ):
        """
        初始化应用

        Args:
            db_uri: 数据库连接URI
            llm_model: LLM模型名称
        """
        self.db_uri = db_uri or os.getenv("DATABASE_URL")
        self.llm_model = llm_model

        # 验证配置
        if not self.db_uri:
            raise ValueError("DATABASE_URL 环境变量未设置")

        if not os.getenv("ANTHROPIC_API_KEY"):
            raise ValueError("ANTHROPIC_API_KEY 环境变量未设置")

        # 初始化checkpointer
        self.checkpointer = PostgresSaver.from_conn_string(self.db_uri)

        # 构建图
        self.graph = self._build_graph()

    def _build_graph(self) -> StateGraph:
        """
        构建完整的APS工作流图

        Returns:
            编译后的StateGraph
        """
        # 使用自动生成的工作流构建器
        builder = build_scheduling_orchestration_graph()

        # ========================================
        # 🎯 自定义扩展区域
        # 可以在这里添加自定义节点和边
        # ========================================

        if HAS_CUSTOM_EXTENSIONS:
            # 添加自定义节点
            custom_nodes = get_custom_nodes()
            for node_name, node_fn in custom_nodes.items():
                builder.add_node(node_name, node_fn)
                # 根据需要添加边...

        # 编译图
        compiled_graph = builder.compile(checkpointer=self.checkpointer)

        # 应用中间件（如果有）
        if HAS_CUSTOM_MIDDLEWARE:
            middleware_stack = get_middleware_stack()
            for middleware in middleware_stack:
                compiled_graph = middleware(compiled_graph)

        return compiled_graph

    def run(
        self,
        user_input: str,
        thread_id: str = "default",
        **kwargs
    ) -> Dict[str, Any]:
        """
        运行APS工作流

        Args:
            user_input: 用户输入的调度需求
            thread_id: 会话线程ID
            **kwargs: 其他参数

        Returns:
            执行结果字典
        """
        config = {"configurable": {"thread_id": thread_id}}

        initial_state = {
            "messages": [{"role": "user", "content": user_input}],
            "user_name": kwargs.get("user_name", "User"),
            "communication_language": kwargs.get("language", "Chinese")
        }

        result = self.graph.invoke(initial_state, config)
        return result

    def stream(
        self,
        user_input: str,
        thread_id: str = "default",
        **kwargs
    ):
        """
        流式运行APS工作流

        Args:
            user_input: 用户输入
            thread_id: 线程ID
            **kwargs: 其他参数

        Yields:
            状态更新
        """
        config = {"configurable": {"thread_id": thread_id}}

        initial_state = {
            "messages": [{"role": "user", "content": user_input}],
            "user_name": kwargs.get("user_name", "User"),
            "communication_language": kwargs.get("language", "Chinese")
        }

        for chunk in self.graph.stream(
            initial_state,
            config,
            stream_mode="values"
        ):
            yield chunk


# ============================================================================
# CLI 入口
# ============================================================================

def main():
    """命令行入口"""
    import argparse

    parser = argparse.ArgumentParser(
        description="APS LangGraph 应用"
    )
    parser.add_argument(
        "input",
        help="调度需求描述"
    )
    parser.add_argument(
        "--thread-id",
        default="cli-session",
        help="会话线程ID"
    )
    parser.add_argument(
        "--stream",
        action="store_true",
        help="流式输出"
    )

    args = parser.parse_args()

    print("🚀 启动APS LangGraph...")

    try:
        app = APSLangGraphApp()
    except Exception as e:
        print(f"❌ 初始化失败: {e}")
        sys.exit(1)

    print(f"📝 用户输入: {args.input}")
    print("⚙️  执行中...\n")

    try:
        if args.stream:
            # 流式输出
            for chunk in app.stream(args.input, args.thread_id):
                if "messages" in chunk:
                    last_msg = chunk["messages"][-1]
                    if hasattr(last_msg, "content"):
                        print(last_msg.content)
                        print("---")
        else:
            # 一次性输出
            result = app.run(args.input, args.thread_id)
            print("\n✅ 执行完成!")
            print(f"📊 结果: {result.get('final_report', '无')}")

    except KeyboardInterrupt:
        print("\n⚠️  用户中断")
        sys.exit(0)
    except Exception as e:
        print(f"\n❌ 执行失败: {e}")
        import traceback
        traceback.print_exc()
        sys.exit(1)


if __name__ == "__main__":
    main()
```

### 核心同步脚本 (scripts/sync_from_bmad.sh)

```bash
#!/bin/bash
# BMAD同步脚本
# 从BMAD-METHOD项目同步编译产物

set -e  # 遇到错误立即退出

# ============================================================================
# 配置
# ============================================================================

BMAD_PROJECT="${BMAD_PROJECT:-../BMAD-METHOD}"
COMPILE_OUTPUT="${COMPILE_OUTPUT:-../_langgraph-dist}"
CURRENT_DIR="$(pwd)"
SYNC_CONFIG=".bmad-sync-config.yml"

# 颜色定义
RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
BLUE='\033[0;34m'
NC='\033[0m' # No Color

# ============================================================================
# 函数定义
# ============================================================================

log_info() {
    echo -e "${BLUE}$1${NC}"
}

log_success() {
    echo -e "${GREEN}$1${NC}"
}

log_warning() {
    echo -e "${YELLOW}$1${NC}"
}

log_error() {
    echo -e "${RED}$1${NC}"
}

check_prerequisites() {
    # 检查必要工具
    for cmd in rsync jq python3; do
        if ! command -v $cmd &> /dev/null; then
            log_error "❌ 错误: 缺少必要工具 $cmd"
            exit 1
        fi
    done

    # 检查BMAD项目
    if [ ! -d "$BMAD_PROJECT" ]; then
        log_error "❌ 错误: 找不到BMAD-METHOD项目: $BMAD_PROJECT"
        exit 1
    fi

    # 检查同步配置
    if [ ! -f "$SYNC_CONFIG" ]; then
        log_error "❌ 错误: 找不到同步配置文件: $SYNC_CONFIG"
        exit 1
    fi
}

# ============================================================================
# 主流程
# ============================================================================

main() {
    log_info "🔄 开始智能同步..."
    echo ""

    # 步骤0: 检查前置条件
    check_prerequisites

    # 步骤1: 编译
    log_info "⚙️  步骤1: 编译BMAD到LangGraph..."
    cd "$BMAD_PROJECT"
    if npm run compile:aps:langgraph; then
        log_success "   ✓ 编译成功"
    else
        log_error "   ✗ 编译失败"
        exit 1
    fi
    cd "$CURRENT_DIR"
    echo ""

    # 检查编译输出
    if [ ! -d "$COMPILE_OUTPUT" ]; then
        log_error "❌ 错误: 编译输出目录不存在: $COMPILE_OUTPUT"
        exit 1
    fi

    # 步骤2: 版本检查
    log_info "🔍 步骤2: 检查版本兼容性..."
    if python3 scripts/check_version_compatibility.py "$COMPILE_OUTPUT/.bmad-manifest.json"; then
        log_success "   ✓ 版本兼容"
    else
        log_warning "   ⚠️  版本可能不兼容"
        read -p "   是否继续? (y/n) " -n 1 -r
        echo
        if [[ ! $REPLY =~ ^[Yy]$ ]]; then
            log_info "   同步已取消"
            exit 0
        fi
    fi
    echo ""

    # 步骤3: 备份
    log_info "💾 步骤3: 备份当前版本..."
    BACKUP_DIR="backups/$(date +%Y%m%d_%H%M%S)"
    mkdir -p "$BACKUP_DIR"
    if [ -d "src/generated" ]; then
        cp -r src/generated "$BACKUP_DIR/" 2>/dev/null || true
    fi
    if [ -d "knowledge" ]; then
        cp -r knowledge "$BACKUP_DIR/" 2>/dev/null || true
    fi
    log_success "   ✓ 备份到: $BACKUP_DIR"
    echo ""

    # 步骤4: 同步自动管理的目录
    log_info "📁 步骤4: 同步自动管理的目录..."

    # 同步src/generated/
    log_info "   同步 src/generated/..."
    mkdir -p src/generated
    rsync -av --delete "$COMPILE_OUTPUT/src/" "$CURRENT_DIR/src/generated/"
    log_success "   ✓ src/generated/ 同步完成"

    # 同步knowledge/
    log_info "   同步 knowledge/..."
    rsync -av --delete "$COMPILE_OUTPUT/knowledge/" "$CURRENT_DIR/knowledge/"
    log_success "   ✓ knowledge/ 同步完成"
    echo ""

    # 步骤5: 处理generate_once文件
    log_info "📝 步骤5: 处理首次生成文件..."

    # app.py
    if [ ! -f "src/app.py" ]; then
        cp "$COMPILE_OUTPUT/src/app.py" "src/app.py"
        log_success "   ✓ 创建 src/app.py"
    else
        log_info "   ⊙ src/app.py 已存在，跳过"
    fi

    # config.py
    if [ ! -f "src/config.py" ]; then
        cp "$COMPILE_OUTPUT/src/config.py" "src/config.py"
        log_success "   ✓ 创建 src/config.py"
    else
        log_info "   ⊙ src/config.py 已存在，跳过"
    fi

    # .env.example
    if [ ! -f ".env.example" ]; then
        cp "$COMPILE_OUTPUT/.env.example" ".env.example"
        log_success "   ✓ 创建 .env.example"
    else
        # 对比并提示
        if ! diff -q "$COMPILE_OUTPUT/.env.example" ".env.example" > /dev/null 2>&1; then
            log_warning "   ⚠️  .env.example 有变更，请手动检查"
            log_info "      对比: diff $COMPILE_OUTPUT/.env.example .env.example"
        fi
    fi
    echo ""

    # 步骤6: 智能合并依赖
    log_info "🔗 步骤6: 合并依赖文件..."
    python3 scripts/merge_requirements.py \
        "$COMPILE_OUTPUT/requirements.txt" \
        "$CURRENT_DIR/requirements.txt"
    log_success "   ✓ requirements.txt 合并完成"
    echo ""

    # 步骤7: 更新manifest
    log_info "📋 步骤7: 更新编译清单..."
    cp "$COMPILE_OUTPUT/.bmad-manifest.json" ".bmad-manifest.json"
    log_success "   ✓ 清单已更新"
    echo ""

    # 步骤8: 生成变更报告
    log_info "📊 步骤8: 生成变更报告..."
    python3 scripts/generate_sync_report.py
    log_success "   ✓ 报告已生成: sync-report.md"
    echo ""

    # 显示统计
    log_success "✅ 同步完成!"
    echo ""
    log_info "📊 变更统计:"
    MANIFEST=".bmad-manifest.json"
    if [ -f "$MANIFEST" ]; then
        echo "   编译时间: $(jq -r '.compiled_at' $MANIFEST)"
        echo "   智能体数: $(jq -r '.stats.agents_count' $MANIFEST)"
        echo "   总文件数: $(jq -r '.stats.total_files' $MANIFEST)"
    fi
    echo ""
    log_info "🔍 下一步:"
    echo "   1. 查看变更: cat sync-report.md"
    echo "   2. 运行测试: pytest tests/"
    echo "   3. 检查差异: git diff"
    echo "   4. 提交变更: git add . && git commit -m 'sync: 从BMAD同步'"
}

# 执行主流程
main "$@"
```
