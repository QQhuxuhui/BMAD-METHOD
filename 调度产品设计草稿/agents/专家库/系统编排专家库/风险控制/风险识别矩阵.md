## 📖 矩阵概述

**核心理念**: 系统化识别技术、业务、项目三大类风险，量化评估风险等级，指导风险控制决策

**风险分类**:

- 技术风险：算法性能、集成复杂性、技术成熟度、扩展性
- 业务风险：需求理解、业务流程、用户接受度、投资回报
- 项目风险：时间、资源、团队、变更

## 🔧 代码实现

```python
from typing import Dict, List
from enum import Enum
from dataclasses import dataclass

class RiskCategory(Enum):
    """风险类别"""
    TECHNICAL = "technical"
    BUSINESS = "business"
    PROJECT = "project"

class RiskLevel(Enum):
    """风险等级"""
    CRITICAL = 5   # 严重
    HIGH = 4       # 高
    MEDIUM = 3     # 中
    LOW = 2        # 低
    MINIMAL = 1    # 极小

@dataclass
class Risk:
    """风险项"""
    id: str
    category: RiskCategory
    name: str
    description: str
    probability: float  # 发生概率 (0-1)
    impact: float       # 影响程度 (0-1)
    level: RiskLevel
    mitigation: List[str]

class RiskIdentificationMatrix:
    """风险识别矩阵"""

    def __init__(self):
        self.risk_templates = self._initialize_risk_templates()

    def _initialize_risk_templates(self) -> Dict:
        """初始化风险模板"""
        return {
            'technical': {

                    ]
                },
                'scalability': {

                    ]

                    'indicators': [
                        ('user_involvement', 'low'),

                    ]

                    ]
                },
                'scope_change': {

                    'description': '需求变更导致的风险',
                    'indicators': [
                        ('requirement_stability', 'low'),

                        ('scope_definition', 'unclear')
                    ]
                }

    def identify_risks(self, project_context: Dict) -> List[Risk]:
        """
        识别项目风险

        参数:
        - project_context: 项目上下文信息

        返回:
        - 识别到的风险列表
        """
        identified_risks = []
        risk_id = 1

        # 识别各类风险
        for category_name, risk_templates in self.risk_templates.items():
            category = RiskCategory(category_name)

            for risk_key, template in risk_templates.items():
                # 检查风险指标
                triggered, probability, impact = self._check_risk_indicators(
                    template['indicators'],
                    project_context
                )

                if triggered:
                    # 计算风险等级
                    level = self._calculate_risk_level(probability, impact)

                    # 生成风险缓解措施
                    mitigation = self._generate_mitigation(category, risk_key)

                    risk = Risk(
                        id=f"RISK-{risk_id:03d}",
                        category=category,
                        name=template['name'],
                        description=template['description'],
                        probability=probability,
                        impact=impact,
                        level=level,
                        mitigation=mitigation
                    )

                    identified_risks.append(risk)
                    risk_id += 1

        # 按风险等级排序
        identified_risks.sort(key=lambda r: r.level.value, reverse=True)

        return identified_risks

    def _check_risk_indicators(self, indicators: List, context: Dict) -> tuple:
        """检查风险指标"""
        triggered_count = 0
        total_indicators = len(indicators)

        for indicator_key, indicator_value in indicators:
            context_value = context.get(indicator_key)

            if self._matches_indicator(context_value, indicator_value):
                triggered_count += 1

        # 计算触发比例
        trigger_ratio = triggered_count / total_indicators if total_indicators > 0 else 0

        # 风险触发阈值：50%指标触发即认为风险存在
        triggered = trigger_ratio >= 0.5

        # 概率 = 触发比例
        probability = trigger_ratio

        # 影响程度基于触发的指标数量
        impact = min(1.0, triggered_count * 0.3)

        return triggered, probability, impact

    def _matches_indicator(self, context_value, indicator_value) -> bool:
        """匹配指标值"""
        if context_value is None:
            return False

        # 布尔匹配
        if isinstance(indicator_value, bool):
            return context_value == indicator_value

        # 数值比较
        if isinstance(indicator_value, str) and indicator_value.startswith('>'):
            threshold = int(indicator_value[1:])
            return int(context_value) > threshold

        if isinstance(indicator_value, str) and indicator_value.startswith('<'):
            threshold = int(indicator_value[1:])
            return int(context_value) < threshold

        # 字符串匹配
        return str(context_value).lower() == str(indicator_value).lower()

    def _calculate_risk_level(self, probability: float, impact: float) -> RiskLevel:
        """计算风险等级"""
        risk_score = probability * impact

        if risk_score >= 0.7:
            return RiskLevel.CRITICAL
        elif risk_score >= 0.5:
            return RiskLevel.HIGH
        elif risk_score >= 0.3:
            return RiskLevel.MEDIUM
        elif risk_score >= 0.1:
            return RiskLevel.LOW
        else:
            return RiskLevel.MINIMAL

    def _generate_mitigation(self, category: RiskCategory, risk_key: str) -> List[str]:
        """生成风险缓解措施"""
        mitigation_strategies = {
            ('technical', 'algorithm_performance'): [
                '进行性能基准测试验证',
                '准备算法降级方案',
                '考虑分布式计算提升性能'
            ],
            ('technical', 'integration_complexity'): [
                '制定详细集成计划',
                '建立集成测试环境',
                '分阶段集成降低风险'
            ],
            ('technical', 'technology_maturity'): [
                '进行技术POC验证',
                '寻找成功案例参考',
                '准备技术备选方案'
            ],
            ('technical', 'scalability'): [
                '设计可扩展架构',
                '进行压力测试',
                '预留扩展接口'
            ],
            ('business', 'requirement_understanding'): [
                '增加需求澄清会议',
                '建立需求验证机制',
                '引入领域专家'
            ],
            ('business', 'business_process_impact'): [
                '制定变更管理计划',
                '分阶段推进变更',
                '提供充分培训'
            ],
            ('business', 'user_acceptance'): [
                '提前进行用户调研',
                '建立用户反馈机制',
                '展示明确的价值'
            ],
            ('business', 'roi'): [
                '细化成本估算',
                '量化收益指标',
                '建立ROI监控机制'
            ],
            ('project', 'timeline'): [
                '增加缓冲时间',
                '识别关键路径',
                '准备加速方案'
            ],
            ('project', 'resource'): [
                '寻求额外预算',
                '优化资源配置',
                '考虑外部资源'
            ],
            ('project', 'team'): [
                '提供技能培训',
                '引入外部专家',
                '加强知识传递'
            ],
            ('project', 'scope_change'): [
                '建立变更控制流程',
                '冻结关键需求',
                '预留变更预算'
            ]
        }

        key = (category.value, risk_key)
        return mitigation_strategies.get(key, ['制定专项应对计划'])

    def generate_risk_matrix_report(self, risks: List[Risk]) -> Dict:
        """生成风险矩阵报告"""
        # 统计各类风险
        category_stats = {}
        level_stats = {}

        for risk in risks:
            # 按类别统计
            cat = risk.category.value
            category_stats[cat] = category_stats.get(cat, 0) + 1

            # 按等级统计
            level = risk.level.name
            level_stats[level] = level_stats.get(level, 0) + 1

        # 计算整体风险评分
        total_risk_score = sum(
            risk.probability * risk.impact for risk in risks
        ) / len(risks) if risks else 0

        # 确定整体风险等级
        if total_risk_score >= 0.7:
            overall_risk = "HIGH"
        elif total_risk_score >= 0.4:
            overall_risk = "MEDIUM"
        else:
            overall_risk = "LOW"

        return {
            'total_risks': len(risks),
            'category_distribution': category_stats,
            'level_distribution': level_stats,
            'overall_risk_score': total_risk_score,
            'overall_risk_level': overall_risk,
            'critical_risks': [r for r in risks if r.level == RiskLevel.CRITICAL],
            'high_risks': [r for r in risks if r.level == RiskLevel.HIGH]
        }
```

## 📋 使用示例

### 场景1: 识别VRP项目风险

```python
matrix = RiskIdentificationMatrix()

# 项目上下文
project_context = {
    'complexity_class': 'NP-hard',
    'scale': 'large',
    'time_constraint': 'strict',
    'legacy_system': True,
    'interface_count': 8,
    'new_technology': False,
    'team_experience': 'medium',
    'requirement_clarity': 'high',
    'timeline': 'aggressive',
    'budget': 'adequate'
}

# 识别风险
risks = matrix.identify_risks(project_context)

print(f"识别到 {len(risks)} 个风险:\n")

for risk in risks:
    print(f"[{risk.level.name}] {risk.name}")
    print(f"  ID: {risk.id}")
    print(f"  类别: {risk.category.value}")
    print(f"  概率: {risk.probability:.2%}, 影响: {risk.impact:.2%}")
    print(f"  缓解措施:")
    for mitigation in risk.mitigation:
        print(f"    - {mitigation}")
    print()
```

## 📊 风险矩阵可视化

| 风险等级 | 概率 × 影响 | 应对策略           |
| -------- | ----------- | ------------------ |
| **严重** | ≥0.70       | 立即处理，高优先级 |
| **高**   | 0.50-0.69   | 制定缓解计划       |
| **中**   | 0.30-0.49   | 监控和评估         |
| **低**   | 0.10-0.29   | 接受或最小化       |
| **极小** | <0.10       | 接受               |
