```python
# backend/app/schemas/workflow.py
from pydantic import BaseModel, Field
from typing import Optional, List, Dict, Any
from datetime import datetime
import uuid

class CreateWorkflowRequest(BaseModel):
    problem_description: str = Field(..., description="问题描述")
    domain: Optional[str] = Field(None, description="应用领域")
    constraints: List[str] = Field(default_factory=list, description="约束条件列表")

    class Config:
        json_schema_extra = {
            "example": {
                "problem_description": "优化100辆车的配送路线",
                "domain": "logistics",
                "constraints": ["实时交通", "电动车电池"]
            }
        }

class WorkflowResponse(BaseModel):
    id: uuid.UUID
    thread_id: str
    status: str
    current_phase: str
    started_at: datetime
    completed_at: Optional[datetime]

class ResumeWorkflowRequest(BaseModel):
    decision: str = Field(..., description="approved/rejected/modified")
    feedback: str = Field("", description="用户反馈")
    modified_data: Optional[Dict[str, Any]] = Field(None)
```

### 权限控制

```python
from fastapi import Depends, HTTPException
from backend.app.core.auth import get_current_user

async def get_workflow_or_403(
    workflow_id: UUID,
    current_user: User = Depends(get_current_user)
) -> WorkflowExecution:
    workflow = await db.get(WorkflowExecution, workflow_id)
    if not workflow:
        raise HTTPException(404, "Workflow not found")
    if workflow.user_id != current_user.id:
        raise HTTPException(403, "Access denied")
    return workflow
```
