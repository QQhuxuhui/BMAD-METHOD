## 📖 策略概述

**核心理念**: 确保方案在逻辑、技术、业务三个层面的一致性，建立系统级一致性保证机制

**一致性维度**:

- 逻辑一致性：方案在逻辑上自洽
- 技术一致性：技术方案的一致性
- 业务一致性：与业务目标的一致性
- 架构一致性：与系统架构的一致性

## 🔧 代码实现

```python
from typing import Dict, List, Tuple
from enum import Enum

class ConsistencyLevel(Enum):
    """一致性等级"""
    PERFECT = 1.0      # 完全一致
    HIGH = 0.9         # 高度一致
    MODERATE = 0.75    # 中等一致
    LOW = 0.6          # 低度一致
    INCONSISTENT = 0.5 # 不一致

class ConsistencyAssurance:
    """一致性保证策略"""

    def __init__(self):
        self.consistency_rules = self._initialize_rules()

    def _initialize_rules(self) -> Dict:
        """初始化一致性规则"""
        return {
            'logic': [
                'algorithm_constraint_compatibility',
                'objective_algorithm_alignment',
                'domain_constraint_match'
            ],
            'technical': [
                'algorithm_implementation_feasibility',
                'constraint_validation_capability',
                'performance_requirement_match'
            ],
            'business': [
                'objective_business_goal_alignment',
                'domain_business_scenario_match',
                'cost_benefit_consistency'
            ],
            'architecture': [
                'component_interface_compatibility',
                'data_flow_consistency',
                'module_dependency_correctness'
            ]
        }

    def check_consistency(self, solution: Dict) -> Dict:
        """
        检查方案一致性

        返回:
        {
            'overall_consistency': float,
            'dimensions': {...},
            'issues': [...],
            'passed': bool
        }
        """
        # 各维度一致性检查
        logic_score = self._check_logic_consistency(solution)
        technical_score = self._check_technical_consistency(solution)
        business_score = self._check_business_consistency(solution)
        architecture_score = self._check_architecture_consistency(solution)

        # 计算总体一致性
        overall = (
            logic_score * 0.30 +
            technical_score * 0.30 +
            business_score * 0.25 +
            architecture_score * 0.15
        )

        # 识别问题
        issues = self._identify_inconsistencies({
            'logic': logic_score,
            'technical': technical_score,
            'business': business_score,
            'architecture': architecture_score
        }, solution)

        return {
            'overall_consistency': overall,
            'dimensions': {
                'logic': logic_score,
                'technical': technical_score,
                'business': business_score,
                'architecture': architecture_score
            },
            'issues': issues,
            'passed': overall >= 0.85,
            'level': self._get_consistency_level(overall)
        }

    def _check_logic_consistency(self, solution: Dict) -> float:
        """检查逻辑一致性"""
        score = 1.0
        penalties = []

        # 规则1: 算法与约束的兼容性
        algorithm = solution.get('algorithm_design', {}).get('algorithm')
        constraints = solution.get('constraint_model', {}).get('constraints', [])

        if algorithm and constraints:
            # 检查算法是否支持所有约束类型
            if algorithm == 'greedy' and 'hard_constraint' in str(constraints):
                penalties.append(0.15)  # 贪心算法不适合硬约束

            if algorithm == 'exact' and len(constraints) > 10:
                penalties.append(0.10)  # 精确算法约束过多会很慢

        # 规则2: 目标与算法的对齐
        objectives = solution.get('objective_function', {}).get('objectives', [])
        if len(objectives) > 1 and algorithm not in ['pareto', 'weighted_sum', 'evolutionary']:
            penalties.append(0.20)  # 多目标需要专用算法

        # 规则3: 领域与约束的匹配
        domain = solution.get('domain_analysis', {}).get('domain')
        if domain == 'VRP':
            expected_constraints = ['vehicle_capacity', 'distance', 'time']
            has_expected = any(c in str(constraints) for c in expected_constraints)
            if not has_expected:
                penalties.append(0.15)

        # 计算最终得分
        final_score = max(0.5, score - sum(penalties))
        return final_score

    def _check_technical_consistency(self, solution: Dict) -> float:
        """检查技术一致性"""
        score = 1.0
        penalties = []

        # 规则1: 算法实现可行性
        algorithm = solution.get('algorithm_design', {})
        complexity_class = algorithm.get('complexity_class')

        if complexity_class in ['NP-hard', 'NP-complete']:
            implementation = algorithm.get('implementation_method')
            if implementation == 'exact':
                scale = solution.get('domain_analysis', {}).get('scale')
                if scale == 'large':
                    penalties.append(0.20)  # 大规模NP问题不应用精确算法

        # 规则2: 约束验证能力
        constraints = solution.get('constraint_model', {}).get('constraints', [])
        if len(constraints) > 15:
            validation_method = solution.get('constraint_model', {}).get('validation')
            if not validation_method:
                penalties.append(0.15)  # 复杂约束需要明确验证方法

        # 规则3: 性能需求匹配
        performance_req = solution.get('performance_requirement', {})
        algorithm_perf = algorithm.get('expected_performance', {})

        if performance_req.get('time_limit'):
            expected_time = algorithm_perf.get('time_estimate')
            if expected_time and expected_time > performance_req['time_limit']:
                penalties.append(0.25)  # 算法预期时间超限

        final_score = max(0.5, score - sum(penalties))
        return final_score

    def _check_business_consistency(self, solution: Dict) -> float:
        """检查业务一致性"""
        score = 1.0
        penalties = []

        # 规则1: 目标与业务目标的对齐
        business_goals = solution.get('business_goals', [])
        objectives = solution.get('objective_function', {}).get('objectives', [])

        if business_goals:
            # 映射关系
            goal_objective_map = {
                'cost_reduction': 'minimize_cost',
                'efficiency_improvement': 'maximize_efficiency',
                'quality_enhancement': 'maximize_quality'
            }

            aligned = 0
            for goal in business_goals:
                expected_obj = goal_objective_map.get(goal)
                if expected_obj and expected_obj in objectives:
                    aligned += 1

            if business_goals and aligned / len(business_goals) < 0.7:
                penalties.append(0.20)  # 目标对齐度不足70%

        # 规则2: 领域与业务场景匹配
        domain = solution.get('domain_analysis', {}).get('domain')
        business_scenario = solution.get('business_scenario')

        scenario_domain_map = {
            'logistics': 'VRP',
            'manufacturing': 'production_scheduling',
            'service': 'staff_scheduling'
        }

        if business_scenario:
            expected_domain = scenario_domain_map.get(business_scenario)
            if expected_domain and domain != expected_domain:
                penalties.append(0.15)

        # 规则3: 成本效益一致性
        estimated_cost = solution.get('estimated_cost', 0)
        expected_benefit = solution.get('expected_benefit', 0)

        if estimated_cost > 0 and expected_benefit > 0:
            roi = (expected_benefit - estimated_cost) / estimated_cost
            if roi < 0.2:  # ROI < 20%
                penalties.append(0.10)

        final_score = max(0.5, score - sum(penalties))
        return final_score

    def _check_architecture_consistency(self, solution: Dict) -> float:
        """检查架构一致性"""
        score = 1.0
        penalties = []

        # 规则1: 组件接口兼容性
        components = solution.get('components', [])
        if len(components) > 1:
            # 检查接口定义
            interfaces_defined = all(
                'interface' in c for c in components
            )
            if not interfaces_defined:
                penalties.append(0.15)

        # 规则2: 数据流一致性
        data_flow = solution.get('data_flow', {})
        if data_flow:
            # 检查输入输出匹配
            for component in components:
                input_type = component.get('input_type')
                output_type = component.get('output_type')

                # 简化检查：类型匹配
                if input_type and output_type:
                    if input_type != output_type and component.get('type') != 'transformer':
                        penalties.append(0.10)

        # 规则3: 模块依赖正确性
        dependencies = solution.get('dependencies', {})
        if dependencies:
            # 检查循环依赖
            has_circular = self._check_circular_dependency(dependencies)
            if has_circular:
                penalties.append(0.25)  # 循环依赖是严重问题

        final_score = max(0.5, score - sum(penalties))
        return final_score

    def _check_circular_dependency(self, dependencies: Dict) -> bool:
        """检查循环依赖"""
        # 简化实现：使用DFS检测环
        visited = set()
        rec_stack = set()

        def has_cycle(node):
            visited.add(node)
            rec_stack.add(node)

            for neighbor in dependencies.get(node, []):
                if neighbor not in visited:
                    if has_cycle(neighbor):
                        return True
                elif neighbor in rec_stack:
                    return True

            rec_stack.remove(node)
            return False

        for node in dependencies:
            if node not in visited:
                if has_cycle(node):
                    return True

        return False

    def _identify_inconsistencies(self, scores: Dict, solution: Dict) -> List[Dict]:
        """识别不一致问题"""
        issues = []

        for dimension, score in scores.items():
            if score < 0.85:
                issues.append({
                    'dimension': dimension,
                    'score': score,
                    'severity': 'high' if score < 0.7 else 'medium',
                    'description': self._get_issue_description(dimension, score, solution)
                })

        return issues

    def _get_issue_description(self, dimension: str, score: float,
                               solution: Dict) -> str:
        """获取问题描述"""
        descriptions = {
            'logic': '逻辑一致性不足：算法、约束、目标间存在不匹配',
            'technical': '技术一致性不足：实现方案与需求存在差距',
            'business': '业务一致性不足：方案与业务目标对齐度低',
            'architecture': '架构一致性不足：组件接口或依赖存在问题'
        }

        return descriptions.get(dimension, '一致性检查发现问题')

    def _get_consistency_level(self, score: float) -> ConsistencyLevel:
        """获取一致性等级"""
        if score >= 0.95:
            return ConsistencyLevel.PERFECT
        elif score >= 0.85:
            return ConsistencyLevel.HIGH
        elif score >= 0.75:
            return ConsistencyLevel.MODERATE
        elif score >= 0.60:
            return ConsistencyLevel.LOW
        else:
            return ConsistencyLevel.INCONSISTENT
```

## 📋 使用示例

### 场景1: 完整方案一致性检查

```python
checker = ConsistencyAssurance()

# 待检查方案
solution = {
    'algorithm_design': {
        'algorithm': 'hybrid_genetic',
        'complexity_class': 'NP-hard',
        'expected_performance': {
            'time_estimate': 7200  # 2小时
        }
    },
    'constraint_model': {
        'constraints': [
            'vehicle_capacity',
            'time_window',
            'driver_hours',
            'customer_priority'
        ],
        'validation': 'incremental_check'
    },
    'objective_function': {
        'objectives': ['minimize_cost', 'minimize_time']
    },
    'domain_analysis': {
        'domain': 'VRP',
        'scale': 'medium'
    },
    'business_goals': ['cost_reduction', 'efficiency_improvement'],
    'performance_requirement': {
        'time_limit': 10800  # 3小时
    }
}

# 检查一致性
result = checker.check_consistency(solution)

print(f"一致性检查结果:")
print(f"  总体一致性: {result['overall_consistency']:.2%}")
print(f"  一致性等级: {result['level'].name}")
print(f"  是否通过: {result['passed']}")

print(f"\n各维度得分:")
for dim, score in result['dimensions'].items():
    print(f"  {dim}: {score:.2%}")

if result['issues']:
    print(f"\n发现问题:")
    for issue in result['issues']:
        print(f"  [{issue['severity']}] {issue['dimension']}: {issue['description']}")
```

输出示例：

```

  一致性等级: HIGH
  是否通过: True

各维度得分:
  logic: 95.00%
  technical: 85.00%
  business: 90.00%
  architecture: 85.00%

发现问题: 无
```

### 场景2: 不一致方案检测

```python
inconsistent_solution = {
    'algorithm_design': {
        'algorithm': 'exact',  # 精确算法
        'complexity_class': 'NP-complete'
    },
    'constraint_model': {
        'constraints': [f'constraint_{i}' for i in range(20)]  # 20个约束
    },
    'objective_function': {
        'objectives': ['obj1', 'obj2', 'obj3']  # 3个目标
    },
    'domain_analysis': {
        'domain': 'VRP',
        'scale': 'large'  # 大规模
    }
}

result = checker.check_consistency(inconsistent_solution)
print(f"一致性: {result['overall_consistency']:.2%}")
print(f"通过: {result['passed']}")
# 输出: 一致性: 65.50%, 通过: False
```
