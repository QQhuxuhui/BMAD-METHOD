## 📖 定义

**帕累托最优**: 不存在其他解在所有目标上都不比它差，且至少在一个目标上更好

**帕累托前沿**: 所有帕累托最优解的集合

## 💡 核心算法

### 非支配排序

```python
def dominates(obj1: Dict[str, float], obj2: Dict[str, float],
              objectives: List) -> bool:
    """
    判断obj1是否支配obj2

    支配条件:
      - obj1在所有目标上都不比obj2差
      - obj1至少在一个目标上比obj2好
    """
    at_least_one_better = False

    for obj in objectives:
        obj_type = obj.optimization_type
        val1 = obj1.get(obj.name, 0)
        val2 = obj2.get(obj.name, 0)

        if obj_type == "minimize":
            if val1 > val2:
                return False  # obj1在这个目标上更差
            elif val1 < val2:
                at_least_one_better = True
        else:  # maximize
            if val1 < val2:
                return False
            elif val1 > val2:
                at_least_one_better = True

    return at_least_one_better
```

## 🔧 代码实现

```python
from typing import Dict, List, Tuple

class ParetoFrontCalculator:
    """帕累托前沿计算器"""

    def __init__(self, objectives: List):
        """
        参数:
          objectives: 目标函数列表
        """
        self.objectives = objectives

    def calculate_pareto_front(self, solutions: List[Dict]) -> List[Dict]:
        """
        计算帕累托前沿

        参数:
          solutions: 解集合，每个解包含所有目标的评估值

        返回:
          帕累托前沿解集
        """
        pareto_front = []

        for candidate in solutions:
            is_dominated = False

            # 检查是否被其他解支配
            for other in solutions:
                if other == candidate:
                    continue

                if self._dominates(other, candidate):
                    is_dominated = True
                    break

            if not is_dominated:
                pareto_front.append(candidate)

        return pareto_front

    def _dominates(self, sol1: Dict, sol2: Dict) -> bool:
        """判断sol1是否支配sol2"""
        at_least_one_better = False

        for obj in self.objectives:
            val1 = sol1.get(obj.name, 0)
            val2 = sol2.get(obj.name, 0)

            if obj.optimization_type == "minimize":
                if val1 > val2:
                    return False
                elif val1 < val2:
                    at_least_one_better = True
            else:  # maximize
                if val1 < val2:
                    return False
                elif val1 > val2:
                    at_least_one_better = True

        return at_least_one_better
```

## 📊 示例

```python
# 定义目标
from dataclasses import dataclass

@dataclass
class Objective:
    name: str
    optimization_type: str

objectives = [
    Objective("cost", "minimize"),
    Objective("time", "minimize"),
    Objective("quality", "maximize")
]

# 解集合
solutions = [
    {"id": "S1", "cost": 100, "time": 10, "quality": 80},
    {"id": "S2", "cost": 120, "time": 8,  "quality": 85},
    {"id": "S3", "cost": 90,  "time": 12, "quality": 75},
    {"id": "S4", "cost": 110, "time": 9,  "quality": 90}
]

# 计算帕累托前沿
calculator = ParetoFrontCalculator(objectives)
pareto_front = calculator.calculate_pareto_front(solutions)

print(f"帕累托前沿包含 {len(pareto_front)} 个解:")
for sol in pareto_front:
    print(f"  {sol['id']}: cost={sol['cost']}, time={sol['time']}, quality={sol['quality']}")
```
