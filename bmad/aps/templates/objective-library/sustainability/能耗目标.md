## 📖 定义

**能耗** = ∑(设备功率 × 运行时间)

## 🔧 代码实现

```python
class EnergyConsumptionObjective:
    """能耗目标函数"""

    def __init__(self, power_ratings: Dict[str, float], weight: float = 1.0):
        """
        参数:
          power_ratings: {resource_id: power_kw} 设备功率(kW)
        """
        self.name = "energy_consumption"
        self.optimization_type = "minimize"
        self.weight = weight
        self.power_ratings = power_ratings

    def evaluate(self, solution: Dict[str, List[Dict]]) -> float:
        """评估总能耗 (kWh)"""
        total_energy = 0.0

        for resource_id, tasks in solution.items():
            power_kw = self.power_ratings.get(resource_id, 0.0)
            operating_hours = sum(task.get('duration', 0) for task in tasks)

            energy_kwh = power_kw * operating_hours
            total_energy += energy_kwh

        return total_energy
```

## 📊 示例

```python
# 设备功率配置
power_ratings = {
    'M1': 50,   # 50kW
    'M2': 75,   # 75kW
    'M3': 60    # 60kW
}

# 调度方案
solution = {
    'M1': [{'duration': 2}, {'duration': 3}],  # 运行5小时
    'M2': [{'duration': 4}]                     # 运行4小时
}

objective = EnergyConsumptionObjective(power_ratings)
energy = objective.evaluate(solution)
print(f"总能耗: {energy} kWh")
# 输出: 总能耗: 550 kWh (M1: 50*5=250, M2: 75*4=300)
```
