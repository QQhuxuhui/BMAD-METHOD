## 📖 策略概述

**核心理念**: 自动检测专家建议间的冲突，基于系统性知识进行仲裁，寻找最优妥协方案

**冲突类型**:

- 观点冲突：专家建议相互矛盾
- 目标冲突：优化目标相互排斥
- 资源冲突：资源需求超出限制
- 技术冲突：技术方案不兼容

## 🔧 代码实现

```python
from enum import Enum
from typing import List, Dict, Tuple

class ConflictType(Enum):
    """冲突类型"""
    OPINION = "opinion_conflict"      # 观点冲突
    OBJECTIVE = "objective_conflict"  # 目标冲突
    RESOURCE = "resource_conflict"    # 资源冲突
    TECHNICAL = "technical_conflict"  # 技术冲突

class ConflictSeverity(Enum):
    """冲突严重程度"""
    LOW = 1       # 低：可忽略
    MEDIUM = 2    # 中：需要协调
    HIGH = 3      # 高：必须解决
    CRITICAL = 4  # 严重：阻断性

class ConflictResolution:
    """冲突解决策略"""

    def __init__(self):
        self.resolution_history = []

    def detect_conflicts(self, expert_suggestions: Dict) -> List[Dict]:
        """
        检测专家建议间的冲突

        返回:
        [
            {
                'type': ConflictType,
                'severity': ConflictSeverity,
                'parties': [...],
                'description': str,
                'details': {...}
            }
        ]
        """
        conflicts = []

        # 检测观点冲突
        opinion_conflicts = self._detect_opinion_conflicts(expert_suggestions)
        conflicts.extend(opinion_conflicts)

        # 检测目标冲突
        objective_conflicts = self._detect_objective_conflicts(expert_suggestions)
        conflicts.extend(objective_conflicts)

        # 检测资源冲突
        resource_conflicts = self._detect_resource_conflicts(expert_suggestions)
        conflicts.extend(resource_conflicts)

        # 检测技术冲突
        technical_conflicts = self._detect_technical_conflicts(expert_suggestions)
        conflicts.extend(technical_conflicts)

        return conflicts

    def _detect_opinion_conflicts(self, suggestions: Dict) -> List[Dict]:
        """检测观点冲突"""
        conflicts = []

        # 示例：算法专家推荐精确算法，但领域专家认为规模过大需要启发式
        algo_expert = suggestions.get('algorithm', {})
        domain_expert = suggestions.get('domain', {})

        if algo_expert and domain_expert:
            algo_recommendation = algo_expert.get('recommended_algorithm')
            scale = domain_expert.get('estimated_scale')

            if algo_recommendation == 'exact_algorithm' and scale == 'large':
                conflicts.append({
                    'type': ConflictType.OPINION,
                    'severity': ConflictSeverity.HIGH,
                    'parties': ['algorithm', 'domain'],
                    'description': '算法选择与问题规模不匹配',
                    'details': {
                        'algorithm': algo_recommendation,
                        'scale': scale,
                        'issue': '精确算法不适用于大规模问题'
                    }
                })

        return conflicts

    def _detect_objective_conflicts(self, suggestions: Dict) -> List[Dict]:
        """检测目标冲突"""
        conflicts = []

        objective_expert = suggestions.get('objective', {})
        objectives = objective_expert.get('objectives', [])

        # 检测相互排斥的目标
        conflicting_pairs = [
            ('minimize_cost', 'maximize_quality'),
            ('minimize_time', 'minimize_cost'),
            ('maximize_throughput', 'maximize_quality')
        ]

        for obj1, obj2 in conflicting_pairs:
            if obj1 in objectives and obj2 in objectives:
                conflicts.append({
                    'type': ConflictType.OBJECTIVE,
                    'severity': ConflictSeverity.MEDIUM,
                    'parties': ['objective'],
                    'description': f'目标 {obj1} 与 {obj2} 相互冲突',
                    'details': {
                        'objective1': obj1,
                        'objective2': obj2,
                        'tradeoff_required': True
                    }
                })

        return conflicts

    def _detect_resource_conflicts(self, suggestions: Dict) -> List[Dict]:
        """检测资源冲突"""
        conflicts = []

        # 检测时间资源冲突
        estimated_times = []
        for expert, suggestion in suggestions.items():
            if 'estimated_time' in suggestion:
                estimated_times.append({
                    'expert': expert,
                    'time': suggestion['estimated_time']
                })

        if estimated_times:
            total_time = sum(t['time'] for t in estimated_times)
            available_time = suggestions.get('_context', {}).get('time_limit', float('inf'))

            if total_time > available_time:
                conflicts.append({
                    'type': ConflictType.RESOURCE,
                    'severity': ConflictSeverity.HIGH,
                    'parties': [t['expert'] for t in estimated_times],
                    'description': '时间资源不足',
                    'details': {
                        'required_time': total_time,
                        'available_time': available_time,
                        'deficit': total_time - available_time
                    }
                })

        return conflicts

    def _detect_technical_conflicts(self, suggestions: Dict) -> List[Dict]:
        """检测技术冲突"""
        conflicts = []

        # 示例：约束专家要求硬约束，但算法专家的算法不支持
        constraint_expert = suggestions.get('constraint', {})
        algo_expert = suggestions.get('algorithm', {})

        if constraint_expert and algo_expert:
            hard_constraints = constraint_expert.get('hard_constraints', [])
            algo_capabilities = algo_expert.get('algorithm_capabilities', [])

            if hard_constraints and 'handle_hard_constraints' not in algo_capabilities:
                conflicts.append({
                    'type': ConflictType.TECHNICAL,
                    'severity': ConflictSeverity.CRITICAL,
                    'parties': ['constraint', 'algorithm'],
                    'description': '算法不支持硬约束处理',
                    'details': {
                        'hard_constraints': hard_constraints,
                        'algorithm_capabilities': algo_capabilities
                    }
                })

        return conflicts

    def resolve(self, conflict: Dict, suggestions: Dict) -> Dict:
        """
        解决单个冲突

        返回:
        {
            'resolution': str,  # 解决方案描述
            'actions': [...],   # 具体行动
            'compromises': {...}  # 妥协点
        }
        """
        if conflict['type'] == ConflictType.OPINION:
            return self._resolve_opinion_conflict(conflict, suggestions)
        elif conflict['type'] == ConflictType.OBJECTIVE:
            return self._resolve_objective_conflict(conflict, suggestions)
        elif conflict['type'] == ConflictType.RESOURCE:
            return self._resolve_resource_conflict(conflict, suggestions)
        elif conflict['type'] == ConflictType.TECHNICAL:
            return self._resolve_technical_conflict(conflict, suggestions)

    def _resolve_opinion_conflict(self, conflict: Dict, suggestions: Dict) -> Dict:
        """解决观点冲突"""
        # 策略：基于问题特征仲裁

        if '算法选择与问题规模不匹配' in conflict['description']:
            return {
                'resolution': '采用领域专家的规模判断，修改算法选择',
                'actions': [
                    '将精确算法改为启发式算法',
                    '使用两阶段方法：启发式快速解+局部精确优化'
                ],
                'compromises': {
                    'algorithm': '放弃精确性保证',
                    'domain': '接受近似解',
                    'quality_tradeoff': '99% -> 95% 最优性'
                }
            }

        return {
            'resolution': '需要人工仲裁',
            'actions': ['提交专家会议讨论'],
            'compromises': {}
        }

    def _resolve_objective_conflict(self, conflict: Dict, suggestions: Dict) -> Dict:
        """解决目标冲突"""
        # 策略：多目标优化

        obj1 = conflict['details']['objective1']
        obj2 = conflict['details']['objective2']

        return {
            'resolution': '使用多目标优化方法',
            'actions': [
                f'采用Pareto最优方法同时考虑 {obj1} 和 {obj2}',
                '设置权重进行加权优化',
                '生成多个Pareto解供用户选择'
            ],
            'compromises': {
                obj1: '不追求单一目标最优',
                obj2: '不追求单一目标最优',
                'method': 'weighted_sum',
                'weights': {obj1: 0.6, obj2: 0.4}  # 示例权重
            }
        }

    def _resolve_resource_conflict(self, conflict: Dict, suggestions: Dict) -> Dict:
        """解决资源冲突"""
        # 策略：分阶段实施或范围调整

        deficit = conflict['details']['deficit']

        if deficit < conflict['details']['available_time'] * 0.2:
            # 轻微超时：优化流程
            return {
                'resolution': '优化工作流程减少时间',
                'actions': [
                    '并行化部分任务',
                    '减少验证轮次',
                    '使用快速近似算法'
                ],
                'compromises': {
                    'time_saved': deficit,
                    'quality_impact': 'minimal'
                }
            }
        else:
            # 严重超时：分阶段或调整范围
            return {
                'resolution': '分阶段实施',
                'actions': [
                    '第一阶段：核心功能实现',
                    '第二阶段：优化和扩展',
                    '调整项目范围'
                ],
                'compromises': {
                    'phased_delivery': True,
                    'initial_scope': '70% 功能',
                    'timeline_extension': f'+{deficit} 时间单位'
                }
            }

    def _resolve_technical_conflict(self, conflict: Dict, suggestions: Dict) -> Dict:
        """解决技术冲突"""
        # 策略：技术方案调整

        if '算法不支持硬约束处理' in conflict['description']:
            return {
                'resolution': '修改技术方案以支持硬约束',
                'actions': [
                    '选择支持硬约束的算法',
                    '或将硬约束转换为惩罚函数',
                    '或使用约束满足前处理'
                ],
                'compromises': {
                    'algorithm': '使用约束编程混合算法',
                    'constraint': '部分硬约束软化为高惩罚',
                    'performance': '可能牺牲10-15%性能'
                }
            }

        return {
            'resolution': '技术架构需要重新设计',
            'actions': ['召集技术评审会议'],
            'compromises': {}
        }

    def resolve_all(self, conflicts: List[Dict], suggestions: Dict) -> Dict:
        """解决所有冲突"""
        resolutions = []

        for conflict in conflicts:
            resolution = self.resolve(conflict, suggestions)
            resolutions.append({
                'conflict': conflict,
                'resolution': resolution
            })

            # 记录解决历史
            self.resolution_history.append({
                'conflict_type': conflict['type'].value,
                'severity': conflict['severity'].value,
                'resolution': resolution['resolution']
            })

        return {
            'total_conflicts': len(conflicts),
            'resolutions': resolutions,
            'overall_status': self._assess_overall_status(resolutions)
        }

    def _assess_overall_status(self, resolutions: List[Dict]) -> str:
        """评估整体解决状态"""
        unresolved = sum(
            1 for r in resolutions
            if '需要人工仲裁' in r['resolution']['resolution']
        )

        if unresolved == 0:
            return 'all_resolved'
        elif unresolved < len(resolutions) * 0.3:
            return 'mostly_resolved'
        else:
            return 'requires_human_intervention'
```

## 📋 使用示例

### 场景1: 算法与规模冲突

```python
resolver = ConflictResolution()

# 专家建议
suggestions = {
    'algorithm': {
        'recommended_algorithm': 'exact_algorithm',
        'reason': '保证最优性'
    },
    'domain': {
        'estimated_scale': 'large',
        'customer_count': 500
    }
}

# 检测冲突
conflicts = resolver.detect_conflicts(suggestions)
print(f"检测到 {len(conflicts)} 个冲突")

for conflict in conflicts:
    print(f"\n冲突类型: {conflict['type'].value}")
    print(f"严重程度: {conflict['severity'].name}")
    print(f"描述: {conflict['description']}")

    # 解决冲突
    resolution = resolver.resolve(conflict, suggestions)
    print(f"解决方案: {resolution['resolution']}")
    print(f"行动: {resolution['actions']}")
```

### 场景2: 多目标冲突

```python
suggestions = {
    'objective': {
        'objectives': [
            'minimize_cost',
            'minimize_time',
            'maximize_quality'
        ]
    }
}

conflicts = resolver.detect_conflicts(suggestions)
result = resolver.resolve_all(conflicts, suggestions)

print(f"总冲突数: {result['total_conflicts']}")
print(f"整体状态: {result['overall_status']}")

for item in result['resolutions']:
    print(f"\n冲突: {item['conflict']['description']}")
    print(f"妥协: {item['resolution']['compromises']}")
```

## 📊 冲突解决矩阵

| 冲突类型     | 严重程度 | 解决策略     | 妥协方式          |
| ------------ | -------- | ------------ | ----------------- |
| **观点冲突** | 中-高    | 专家权威仲裁 | 采纳主导专家意见  |
| **目标冲突** | 中       | 多目标优化   | 权重平衡          |
| **资源冲突** | 高       | 分阶段实施   | 范围调整/时间延长 |
| **技术冲突** | 严重     | 架构调整     | 技术方案变更      |
