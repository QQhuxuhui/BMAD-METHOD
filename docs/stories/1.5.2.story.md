## Dev Notes

### StateGraph构建示例

```python
from langgraph.graph import StateGraph, END
from backend.app.core.langgraph.state import WorkflowState
from backend.app.core.langgraph.agents import *
from backend.app.core.langgraph.routers import *

async def create_bmad_workflow():
    """创建BMAD工作流"""
    workflow = StateGraph(WorkflowState)

    # 添加agent节点
    workflow.add_node("orchestrator", orchestrator_node)
    workflow.add_node("algorithm", algorithm_expert_node)
    workflow.add_node("constraint", constraint_expert_node)
    workflow.add_node("objective", objective_expert_node)
    workflow.add_node("domain", domain_expert_node)
    workflow.add_node("code_impl", code_impl_expert_node)
    workflow.add_node("extension", extension_expert_node)
    workflow.add_node("quality", quality_expert_node)

    # 添加approval节点
    workflow.add_node("p1_approval", p1_approval_node)
    workflow.add_node("p25_approval", p25_approval_node)

    # 设置入口点
    workflow.set_entry_point("orchestrator")

    # Phase 0 → Phase 1
    workflow.add_edge("orchestrator", "algorithm")
    workflow.add_edge("algorithm", "constraint")
    workflow.add_edge("constraint", "objective")

    # Phase 1 → P1 Approval
    workflow.add_edge("objective", "p1_approval")
    workflow.add_conditional_edges(
        "p1_approval",
        route_after_p1_approval,
        {
            "approved": "domain",
            "rejected": "algorithm",
            "__end__": END
        }
    )

    # Phase 2 → Phase 3
    workflow.add_edge("domain", "code_impl")
    workflow.add_edge("code_impl", "extension")

    # Phase 3 → P2.5 Approval
    workflow.add_edge("extension", "p25_approval")
    workflow.add_conditional_edges(
        "p25_approval",
        route_after_p25_approval,
        {
            "approved": "quality",
            "rejected": "code_impl",
            "__end__": END
        }
    )

    # Phase 4 → End
    workflow.add_edge("quality", END)

    # 配置checkpointer
    checkpointer = await create_checkpointer()

    return workflow.compile(checkpointer=checkpointer)
```

### Checkpoint使用模式

```python
# 执行工作流（自动保存checkpoint）
config = {"configurable": {"thread_id": "unique-id"}}
async for event in workflow.astream(initial_state, config):
    print(event)  # 每个节点输出后自动保存checkpoint

# 从checkpoint恢复
state = await workflow.aget_state(config)
print(state.values)  # 最新状态
print(state.next)    # 下一个要执行的节点

# 继续执行
async for event in workflow.astream(None, config):  # None = 使用当前状态
    print(event)
```

**依赖版本**:

**数据库表结构** (自动创建):

```sql
CREATE TABLE checkpoints (
    thread_id TEXT NOT NULL,
    checkpoint_ns TEXT NOT NULL,
    checkpoint_id TEXT NOT NULL,
    parent_checkpoint_id TEXT,
    type TEXT,
    checkpoint JSONB NOT NULL,
    metadata JSONB NOT NULL,
    PRIMARY KEY (thread_id, checkpoint_ns, checkpoint_id)
);

CREATE TABLE checkpoint_blobs (
    thread_id TEXT NOT NULL,
    checkpoint_ns TEXT NOT NULL,
    channel TEXT NOT NULL,
    version TEXT NOT NULL,
    type TEXT NOT NULL,
    blob BYTEA,
    PRIMARY KEY (thread_id, checkpoint_ns, channel, version)
);

CREATE TABLE checkpoint_writes (
    thread_id TEXT NOT NULL,
    checkpoint_ns TEXT NOT NULL,
    checkpoint_id TEXT NOT NULL,
    task_id TEXT NOT NULL,
    idx INTEGER NOT NULL,
    channel TEXT NOT NULL,
    type TEXT,
    blob BYTEA NOT NULL,
    PRIMARY KEY (thread_id, checkpoint_ns, checkpoint_id, task_id, idx)
);
```

**Thread ID生成**:

```python
import uuid

thread_id = str(uuid.uuid4())  # 防止枚举攻击
```

**错误处理**:

- Checkpoint保存失败：记录错误但继续执行
- 节点执行失败：保存checkpoint后抛出异常
- 超时控制：单个节点30秒，整个工作流60分钟
