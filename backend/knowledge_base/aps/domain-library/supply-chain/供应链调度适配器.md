## 🔧 代码实现

```python
from dataclasses import dataclass
import math

@dataclass
class Product:
    """产品定义"""
    id: str
    demand_rate: float  # 需求速率
    unit_cost: float
    holding_cost_rate: float
    ordering_cost: float

class SupplyChainAdapter:
    """供应链调度适配器"""

    def calculate_eoq(self, product: Product) -> float:
        """
        计算经济订货批量

        EOQ = √(2 × D × S / H)
        """
        eoq = math.sqrt(
            (2 * product.demand_rate * product.ordering_cost) /
            (product.unit_cost * product.holding_cost_rate)
        )
        return eoq

    def calculate_reorder_point(self, product: Product, lead_time: float) -> float:
        """计算再订货点"""
        reorder_point = product.demand_rate * lead_time
        return reorder_point
```
