## 📖 目标概述

### 定义

**加权延迟**（Weighted Tardiness）考虑任务优先级，对重要任务的延迟施加更大惩罚。

**数学定义**：

```
Weighted Tardiness = ∑ w_j · T_j

其中:
  w_j = 任务 j 的权重
  T_j = max{0, C_j - d_j}
  C_j = 完成时间
  d_j = 截止期

目标: minimize ∑ w_j · T_j
```

### ATC规则 (Apparent Tardiness Cost)

```python
import math

def atc_schedule(jobs, k_value=2.0):
    """
    ATC规则: 表现延迟成本

    ATC_j = (w_j / p_j) · exp(-max{0, d_j - t - p_j} / (k · p_avg))

    参数:
      k_value: 松弛因子 (通常 1-4)
    """
    avg_duration = sum(j.get('duration', 0) for j in jobs) / len(jobs)

    def atc_priority(job, current_time):
        w = job.get('weight', 1.0)
        p = job.get('duration', 1)
        d = job.get('deadline', float('inf'))

        slack = max(0, d - current_time - p)
        priority = (w / p) * math.exp(-slack / (k_value * avg_duration))

        return priority

    schedule = []
    remaining_jobs = jobs.copy()
    current_time = 0

    while remaining_jobs:
        # 计算每个任务的ATC优先级
        priorities = [(atc_priority(j, current_time), j) for j in remaining_jobs]

        # 选择最高优先级任务
        _, next_job = max(priorities, key=lambda x: x[0])

        # 调度
        start = current_time
        finish = start + next_job.get('duration', 0)
        deadline = next_job.get('deadline', float('inf'))
        weight = next_job.get('weight', 1.0)
        tardiness = max(0, finish - deadline)
        weighted_tardiness = weight * tardiness

        schedule.append({
            'job_id': next_job.get('id'),
            'start_time': start,
            'finish_time': finish,
            'deadline': deadline,
            'weight': weight,
            'tardiness': tardiness,
            'weighted_tardiness': weighted_tardiness
        })

        current_time = finish
        remaining_jobs.remove(next_job)

    return schedule
```

## 🔧 代码实现

```python
from typing import Dict, List
from dataclasses import dataclass
import math

@dataclass
class Job:
    """任务定义"""
    id: str
    duration: float
    deadline: float
    weight: float  # 重要性权重

class WeightedTardinessObjective:
    """加权延迟目标函数"""

    def __init__(self, weight: float = 1.0):
        self.name = "weighted_tardiness"
        self.optimization_type = "minimize"
        self.weight = weight

    def evaluate(self, solution: List[Dict]) -> float:
        """评估加权延迟"""
        weighted_tardiness = 0.0

        for task in solution:
            finish = task.get('finish_time', 0)
            deadline = task.get('deadline', float('inf'))
            weight = task.get('weight', 1.0)

            tardiness = max(0, finish - deadline)
            weighted_tardiness += weight * tardiness

        return weighted_tardiness

    def atc_schedule(self, jobs: List[Job], k_value: float = 2.0) -> List[Dict]:
        """ATC规则调度"""
        if not jobs:
            return []

        avg_duration = sum(j.duration for j in jobs) / len(jobs)

        def atc_priority(job, current_time):
            slack = max(0, job.deadline - current_time - job.duration)
            return (job.weight / job.duration) * math.exp(-slack / (k_value * avg_duration))

        schedule = []
        remaining = jobs.copy()
        current_time = 0

        while remaining:
            priorities = [(atc_priority(j, current_time), j) for j in remaining]
            _, next_job = max(priorities, key=lambda x: x[0])

            start = current_time
            finish = start + next_job.duration
            tardiness = max(0, finish - next_job.deadline)

            schedule.append({
                'job_id': next_job.id,
                'start_time': start,
                'finish_time': finish,
                'deadline': next_job.deadline,
                'weight': next_job.weight,
                'tardiness': tardiness,
                'weighted_tardiness': next_job.weight * tardiness
            })

            current_time = finish
            remaining.remove(next_job)

        return schedule
```

## 📊 应用场景

### 场景: VIP客户订单优先

```python
# VIP订单权重更高
orders = [
    Job(id='VIP1', duration=5, deadline=12, weight=10.0),  # VIP客户
    Job(id='O1', duration=3, deadline=10, weight=1.0),     # 普通客户
    Job(id='VIP2', duration=4, deadline=15, weight=8.0),   # VIP客户
    Job(id='O2', duration=2, deadline=8, weight=1.0),      # 普通客户
]

objective = WeightedTardinessObjective()

# ATC调度
schedule = objective.atc_schedule(orders, k_value=2.0)

# 评估
total_wt = objective.evaluate(schedule)
print(f"加权延迟总计: {total_wt}")

for task in schedule:
    print(f"{task['job_id']}: 权重={task['weight']}, "
          f"延迟={task['tardiness']}, "
          f"加权延迟={task['weighted_tardiness']}")
```
