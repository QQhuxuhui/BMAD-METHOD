## 📖 约束概述

### 定义

**车辆容量约束**（Vehicle Capacity Constraint）规定每辆车所能装载的货物总量不能超过其容量限制。这是车辆路径问题（VRP）中最基本的约束之一。

**数学定义**：

```
对于车辆 k，容量为 Q_k
对于任务集合 S_k 分配给车辆 k

约束条件：
  Σ(i∈S_k) d_i ≤ Q_k

其中：
  d_i = 任务 i 的需求量
  Q_k = 车辆 k 的容量
  S_k = 分配给车辆 k 的任务集合
```

### 约束类型

1. **单维容量约束**
   - 仅考虑重量或体积
   - 最常见的容量约束
   - 示例: 车辆最大载重100kg

2. **多维容量约束**
   - 同时考虑重量、体积、数量等
   - 需要满足所有维度
   - 示例: 重量≤100kg 且 体积≤5m³

3. **同质/异质车队**
   - 同质: 所有车辆容量相同
   - 异质: 不同车辆容量不同
   - 影响算法复杂度

## 💡 核心理论

### 容量可行性

**基本可行性**：

```python
def is_feasible(demands: List[float], capacity: float) -> bool:
    """
    检查任务集合是否可以被单车辆服务
    """
    return sum(demands) <= capacity
```

**路线容量检查**：

```python
def check_route_capacity(route: List[int], demands: Dict[int, float],
                          capacity: float) -> Tuple[bool, float]:
    """
    检查路线的容量约束
    返回: (is_valid, total_load)
    """
    total_load = sum(demands[node] for node in route)
    is_valid = total_load <= capacity
    return is_valid, total_load
```

### 容量利用率

**装载率计算**：

```
装载率 = (实际载重 / 车辆容量) × 100%

目标：
  - 最大化装载率（减少车辆数）
  - 平衡车辆负载
```

## 🔧 代码实现

### 约束处理器类

```python
from typing import Dict, List, Tuple, Optional
from dataclasses import dataclass

@dataclass
class Vehicle:
    """车辆数据结构"""
    id: str
    capacity: float  # 容量
    current_load: float = 0.0  # 当前载重
    available: bool = True

    def remaining_capacity(self) -> float:
        """剩余容量"""
        return self.capacity - self.current_load

    def utilization(self) -> float:
        """利用率 (0-1)"""
        return self.current_load / self.capacity if self.capacity > 0 else 0

class VehicleCapacityConstraintHandler:
    """
    车辆容量约束处理器

    功能:
      - 验证车辆分配是否满足容量约束
      - 检测容量超载情况
      - 计算超载惩罚
      - 修复容量违反
      - 优化车辆装载率
    """

    def __init__(self, vehicles: List[Vehicle], demands: Dict[str, float]):
        """
        参数:
          vehicles: 车辆列表
          demands: 任务需求 {task_id: demand}
        """
        self.vehicles = {v.id: v for v in vehicles}
        self.demands = demands

    def validate(self, solution: Dict[str, List[str]]) -> Tuple[bool, List[Dict]]:
        """
        验证解决方案是否满足容量约束

        参数:
          solution: {vehicle_id: [task_ids]}

        返回:
          (is_valid, violations)
        """
        violations = []

        for vehicle_id, tasks in solution.items():
            vehicle = self.vehicles.get(vehicle_id)
            if not vehicle:
                continue

            # 计算总需求
            total_demand = sum(self.demands.get(task_id, 0) for task_id in tasks)

            # 检查是否超载
            if total_demand > vehicle.capacity:
                overload = total_demand - vehicle.capacity
                overload_percent = (overload / vehicle.capacity) * 100

                violations.append({
                    'type': 'capacity_overload',
                    'vehicle': vehicle_id,
                    'capacity': vehicle.capacity,
                    'total_demand': total_demand,
                    'overload': overload,
                    'overload_percent': overload_percent,
                    'num_tasks': len(tasks),
                    'penalty': overload_percent  # 惩罚与超载比例成正比
                })

        is_valid = len(violations) == 0
        return is_valid, violations

    def get_penalty(self, violations: List[Dict]) -> float:
        """
        计算容量违反的总惩罚

        惩罚策略:
          penalty = Σ (overload_percent * severity_factor)

          severity_factor:
            - overload < 10%: 1x
            - overload 10-20%: 2x
            - overload > 20%: 5x
        """
        total_penalty = 0.0

        for v in violations:
            overload_percent = v['overload_percent']

            # 根据超载严重程度设置惩罚系数
            if overload_percent < 10:
                severity = 1.0
            elif overload_percent < 20:
                severity = 2.0
            else:
                severity = 5.0

            total_penalty += overload_percent * severity

        return total_penalty

    def repair(self, solution: Dict[str, List[str]],
               violations: List[Dict]) -> Dict[str, List[str]]:
        """
        修复容量违反

        策略:
          1. 移除超载车辆的任务
          2. 尝试重新分配到其他车辆
          3. 如果无法分配，创建新路线

        返回:
          修复后的解决方案
        """
        repaired_solution = {k: list(v) for k, v in solution.items()}

        for violation in violations:
            vehicle_id = violation['vehicle']
            overload = violation['overload']

            # 获取该车辆的任务列表
            tasks = repaired_solution[vehicle_id]

            # 按需求量降序排序，优先移除大需求任务
            tasks_with_demand = [
                (task_id, self.demands.get(task_id, 0))
                for task_id in tasks
            ]
            tasks_with_demand.sort(key=lambda x: x[1], reverse=True)

            # 移除任务直到满足容量约束
            removed_tasks = []
            removed_demand = 0.0

            for task_id, demand in tasks_with_demand:
                if removed_demand >= overload:
                    break

                repaired_solution[vehicle_id].remove(task_id)
                removed_tasks.append((task_id, demand))
                removed_demand += demand

            # 尝试将移除的任务分配到其他车辆
            for task_id, demand in removed_tasks:
                assigned = self._assign_to_vehicle(
                    task_id, demand, repaired_solution
                )

                if not assigned:
                    # 无法分配到现有车辆，创建新路线
                    available_vehicle = self._find_available_vehicle()
                    if available_vehicle:
                        repaired_solution[available_vehicle.id] = [task_id]

        return repaired_solution

    def _assign_to_vehicle(self, task_id: str, demand: float,
                           solution: Dict[str, List[str]]) -> bool:
        """
        尝试将任务分配到有剩余容量的车辆
        """
        for vehicle_id, tasks in solution.items():
            vehicle = self.vehicles[vehicle_id]
            current_load = sum(self.demands.get(t, 0) for t in tasks)
            remaining = vehicle.capacity - current_load

            if remaining >= demand:
                solution[vehicle_id].append(task_id)
                return True

        return False

    def _find_available_vehicle(self) -> Optional[Vehicle]:
        """
        找到可用的空闲车辆
        """
        for vehicle in self.vehicles.values():
            if vehicle.available and vehicle.current_load == 0:
                return vehicle
        return None

    def optimize_loading(self, solution: Dict[str, List[str]]) -> Dict[str, List[str]]:
        """
        优化装载方案，提高车辆利用率

        策略:
          1. 识别低利用率车辆
          2. 尝试合并任务到其他车辆
          3. 减少使用的车辆数量
        """
        optimized = {k: list(v) for k, v in solution.items()}

        # 计算每个车辆的利用率
        utilizations = []
        for vehicle_id, tasks in optimized.items():
            vehicle = self.vehicles[vehicle_id]
            total_demand = sum(self.demands.get(t, 0) for t in tasks)
            util = total_demand / vehicle.capacity if vehicle.capacity > 0 else 0

            utilizations.append({
                'vehicle_id': vehicle_id,
                'utilization': util,
                'tasks': tasks,
                'total_demand': total_demand
            })

        # 按利用率升序排序
        utilizations.sort(key=lambda x: x['utilization'])

        # 尝试合并低利用率车辆的任务
        for low_util in utilizations:
            if low_util['utilization'] > 0.5:  # 利用率>50%不合并
                break

            tasks_to_reassign = low_util['tasks']

            # 尝试分配到其他车辆
            all_assigned = True
            for task_id in tasks_to_reassign:
                demand = self.demands.get(task_id, 0)
                assigned = self._assign_to_vehicle(task_id, demand, optimized)

                if not assigned:
                    all_assigned = False
                    break

            if all_assigned:
                # 成功合并，清空该车辆
                optimized[low_util['vehicle_id']] = []

        # 移除空车辆
        optimized = {k: v for k, v in optimized.items() if len(v) > 0}

        return optimized

    def get_statistics(self, solution: Dict[str, List[str]]) -> Dict:
        """
        获取容量使用统计信息
        """
        stats = {
            'num_vehicles_used': len(solution),
            'total_capacity': sum(v.capacity for v in self.vehicles.values()),
            'total_demand': sum(self.demands.values()),
            'vehicles': []
        }

        for vehicle_id, tasks in solution.items():
            vehicle = self.vehicles[vehicle_id]
            total_demand = sum(self.demands.get(t, 0) for t in tasks)
            utilization = (total_demand / vehicle.capacity * 100) if vehicle.capacity > 0 else 0

            stats['vehicles'].append({
                'vehicle_id': vehicle_id,
                'capacity': vehicle.capacity,
                'load': total_demand,
                'utilization': round(utilization, 2),
                'num_tasks': len(tasks)
            })

        # 平均利用率
        if stats['vehicles']:
            avg_util = sum(v['utilization'] for v in stats['vehicles']) / len(stats['vehicles'])
            stats['avg_utilization'] = round(avg_util, 2)
        else:
            stats['avg_utilization'] = 0.0

        return stats
```

## 📊 应用场景

### 场景1: 快递配送

**问题描述**：

```yaml
车辆: 5辆小货车，容量均为 100kg
包裹:
  - 客户A: 15kg
  - 客户B: 25kg
  - 客户C: 35kg
  - 客户D: 20kg
  - 客户E: 30kg
  - 客户F: 40kg

约束: 每辆车的总载重 ≤ 100kg
目标: 最少使用车辆数
```

**解决方案**：

```python
# 定义车辆和需求
vehicles = [
    Vehicle(id='V1', capacity=100),
    Vehicle(id='V2', capacity=100),
    Vehicle(id='V3', capacity=100)
]

demands = {
    'A': 15, 'B': 25, 'C': 35,
    'D': 20, 'E': 30, 'F': 40
}

# 创建约束处理器
handler = VehicleCapacityConstraintHandler(vehicles, demands)

# 初始方案（可能不优）
solution = {
    'V1': ['A', 'B', 'C'],  # 75kg
    'V2': ['D', 'E'],       # 50kg
    'V3': ['F']             # 40kg
}

# 验证
is_valid, violations = handler.validate(solution)
print(f"Valid: {is_valid}")

# 优化装载率
optimized = handler.optimize_loading(solution)
# 结果可能: {'V1': ['A', 'B', 'C'], 'V2': ['D', 'E', 'F']}
# 从3辆车优化到2辆车

# 统计
stats = handler.get_statistics(optimized)
print(f"Vehicles used: {stats['num_vehicles_used']}")
print(f"Avg utilization: {stats['avg_utilization']}%")
```

```yaml
车辆: 容量 100kg 且 8m³
货物:

约束: 重量 ≤ 100kg 且 体积 ≤ 8m³
```

**扩展实现**：

```python
@dataclass
class MultiDimVehicle:
    id: str
    weight_capacity: float
    volume_capacity: float

class MultiDimCapacityHandler:
    def __init__(self, vehicles, demands):
        self.vehicles = vehicles
        self.weight_demands = demands['weight']
        self.volume_demands = demands['volume']

    def validate(self, solution):
        violations = []

        for vehicle_id, tasks in solution.items():
            vehicle = self.vehicles[vehicle_id]

            total_weight = sum(self.weight_demands[t] for t in tasks)
            total_volume = sum(self.volume_demands[t] for t in tasks)

            if total_weight > vehicle.weight_capacity:
                violations.append({'type': 'weight_overload', ...})

            if total_volume > vehicle.volume_capacity:
                violations.append({'type': 'volume_overload', ...})

        return len(violations) == 0, violations
```

## 🎯 性能优化

### 1. 增量验证

```python
def incremental_validate(self, vehicle_id: str, new_task: str) -> bool:
    """
    增量验证：检查添加新任务是否超载
    性能：O(1) vs O(n)
    """
    vehicle = self.vehicles[vehicle_id]
    current_load = vehicle.current_load
    new_demand = self.demands.get(new_task, 0)

    return (current_load + new_demand) <= vehicle.capacity
```

### 2. 预计算剩余容量

```python
def precompute_remaining_capacities(self, solution):
    """
    预计算每辆车的剩余容量
    加速任务分配决策
    """
    self.remaining_capacities = {}

    for vehicle_id, tasks in solution.items():
        vehicle = self.vehicles[vehicle_id]
        total_demand = sum(self.demands.get(t, 0) for t in tasks)
        self.remaining_capacities[vehicle_id] = vehicle.capacity - total_demand
```

### 3. Bin Packing优化

```python
def first_fit_decreasing(self, tasks: List[str]) -> Dict[str, List[str]]:
    """
    First Fit Decreasing (FFD) 装箱算法
    保证近似比 11/9 OPT

    步骤:
      1. 按需求量降序排序任务
      2. 依次放入第一个能容纳的车辆
    """
    sorted_tasks = sorted(
        tasks,
        key=lambda t: self.demands.get(t, 0),
        reverse=True
    )

    solution = {}

    for task in sorted_tasks:
        demand = self.demands.get(task, 0)

        # 尝试放入现有车辆
        placed = False
        for vehicle_id in solution:
            if self.incremental_validate(vehicle_id, task):
                solution[vehicle_id].append(task)
                placed = True
                break

        # 需要新车辆
        if not placed:
            new_vehicle = self._find_available_vehicle()
            if new_vehicle:
                solution[new_vehicle.id] = [task]

    return solution
```

## 🧪 测试用例

```python
import unittest

class TestVehicleCapacityConstraint(unittest.TestCase):
    def setUp(self):
        self.vehicles = [
            Vehicle(id='V1', capacity=100),
            Vehicle(id='V2', capacity=100)
        ]

        self.demands = {
            'A': 30, 'B': 40, 'C': 50,
            'D': 20, 'E': 35
        }

        self.handler = VehicleCapacityConstraintHandler(
            self.vehicles, self.demands
        )

    def test_valid_solution(self):
        """测试满足容量约束的方案"""
        solution = {
            'V1': ['A', 'B'],  # 70kg
            'V2': ['C', 'D']   # 70kg
        }

        is_valid, violations = self.handler.validate(solution)

        self.assertTrue(is_valid)
        self.assertEqual(len(violations), 0)

    def test_capacity_violation(self):
        """测试容量超载"""
        solution = {
            'V1': ['A', 'B', 'C']  # 120kg > 100kg
        }

        is_valid, violations = self.handler.validate(solution)

        self.assertFalse(is_valid)
        self.assertEqual(len(violations), 1)
        self.assertEqual(violations[0]['type'], 'capacity_overload')
        self.assertEqual(violations[0]['overload'], 20)

    def test_repair(self):
        """测试修复功能"""
        solution = {
            'V1': ['A', 'B', 'C'],  # 120kg超载
            'V2': []
        }

        _, violations = self.handler.validate(solution)
        repaired = self.handler.repair(solution, violations)

        # 验证修复后满足约束
        is_valid, _ = self.handler.validate(repaired)
        self.assertTrue(is_valid)

    def test_optimize_loading(self):
        """测试装载优化"""
        solution = {
            'V1': ['A'],      # 30kg (30%利用率)
            'V2': ['B', 'C']  # 90kg (90%利用率)
        }

        optimized = self.handler.optimize_loading(solution)

        # 应该合并到一辆车
        self.assertEqual(len(optimized), 1)

        # 验证优化后满足约束
        is_valid, _ = self.handler.validate(optimized)
        self.assertTrue(is_valid)

    def test_statistics(self):
        """测试统计功能"""
        solution = {
            'V1': ['A', 'B'],
            'V2': ['C', 'D']
        }

        stats = self.handler.get_statistics(solution)

        self.assertEqual(stats['num_vehicles_used'], 2)
        self.assertTrue(0 <= stats['avg_utilization'] <= 100)
```
