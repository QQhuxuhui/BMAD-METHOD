## 📖 约束概述

### 定义

**工作时间约束**（Working Hours Constraint）规定资源（人员、设备）的可工作时间范围，包括工作班次、休息时间、假期等时间限制。

**数学定义**：

```
对于资源 r 和时间 t
定义工作时间集合 W_r = {[s_1, e_1], [s_2, e_2], ..., [s_n, e_n]}

约束条件:
  任务分配给资源 r 时，其执行时间必须完全包含在 W_r 中
  ∀ task_i assigned to r: [start_i, end_i] ⊆ W_r
```

### 约束类型

#### 1. **固定班次** (Fixed Shift)

```
工作时段固定不变
例如: 8:00-17:00 (包含午休 12:00-13:00)

W = {[8:00, 12:00], [13:00, 17:00]}
```

#### 2. **轮班制** (Shift Rotation)

```
多个班次轮换
例如:
  早班: 8:00-16:00
  中班: 16:00-24:00
  夜班: 0:00-8:00

轮班周期: 3天一轮换
```

#### 3. **弹性工作时间** (Flexible Hours)

```
核心工作时间 + 弹性时间
例如:
  核心时间: 10:00-15:00 (必须在岗)
  弹性时间: 8:00-10:00, 15:00-19:00 (灵活安排)
  每天工作满8小时
```

#### 4. **假期约束** (Holiday Constraint)

```
法定假期、周末、带薪休假
不可工作日: D_holiday = {2025-01-01, 2025-02-10, ...}
周末: Saturday, Sunday
```

## 💡 核心理论

### 时间段交集判断

```python
def time_intervals_overlap(interval1, interval2):
    """
    判断两个时间段是否有交集

    interval = (start, end)

    重叠条件:
      start1 < end2 AND start2 < end1
    """
    start1, end1 = interval1
    start2, end2 = interval2

    return start1 < end2 and start2 < end1

def time_interval_intersection(interval1, interval2):
    """
    计算两个时间段的交集

    返回: (start, end) 或 None
    """
    start1, end1 = interval1
    start2, end2 = interval2

    if not time_intervals_overlap(interval1, interval2):
        return None

    intersection_start = max(start1, start2)
    intersection_end = min(end1, end2)

    return (intersection_start, intersection_end)
```

### 可工作时间计算

```python
def calculate_available_duration(work_intervals, task_start, task_end):
    """
    计算任务时间段内的实际可工作时长

    参数:
      work_intervals: [(start, end), ...] 工作时段列表
      task_start, task_end: 任务开始和结束时间

    返回:
      available_duration: 可工作时长
    """
    total_duration = timedelta(0)

    for work_start, work_end in work_intervals:
        # 计算交集
        intersection = time_interval_intersection(
            (task_start, task_end),
            (work_start, work_end)
        )

        if intersection:
            int_start, int_end = intersection
            total_duration += (int_end - int_start)

    return total_duration
```

## 🔧 代码实现

```python
from typing import Dict, List, Tuple, Set
from dataclasses import dataclass
from datetime import datetime, timedelta, time
from enum import Enum

class ShiftType(Enum):
    """班次类型"""
    FIXED = "fixed"           # 固定班次
    ROTATION = "rotation"     # 轮班制
    FLEXIBLE = "flexible"     # 弹性工作

@dataclass
class TimeSlot:
    """时间段"""
    start: time
    end: time

    def to_datetime_interval(self, date: datetime.date) -> Tuple[datetime, datetime]:
        """转换为绝对时间区间"""
        start_dt = datetime.combine(date, self.start)
        end_dt = datetime.combine(date, self.end)

        # 处理跨天情况
        if self.end < self.start:
            end_dt += timedelta(days=1)

        return start_dt, end_dt

@dataclass
class Shift:
    """班次定义"""
    name: str
    work_slots: List[TimeSlot]  # 工作时段（可能有多个，中间有休息）
    type: ShiftType = ShiftType.FIXED

@dataclass
class Resource:
    """资源定义"""
    id: str
    shifts: Dict[str, Shift]  # {date_str: Shift}
    holidays: Set[str] = None  # 假期日期集合 {"2025-01-01", ...}
    max_overtime_hours: float = 0  # 最大加班时长（小时/天）

    def __post_init__(self):
        if self.holidays is None:
            self.holidays = set()

class WorkingHoursConstraintHandler:
    """工作时间约束处理器"""

    def __init__(self, resources: Dict[str, Resource],
                 weekend_days: Set[int] = {5, 6}):  # 5=Saturday, 6=Sunday
        """
        参数:
          resources: {resource_id: Resource}
          weekend_days: 周末日期 (0=Monday, 6=Sunday)
        """
        self.resources = resources
        self.weekend_days = weekend_days

    def get_working_intervals(self, resource_id: str,
                              start_date: datetime,
                              end_date: datetime) -> List[Tuple[datetime, datetime]]:
        """
        获取资源在指定时间范围内的所有工作时段

        返回:
          [(start_dt, end_dt), ...]
        """
        resource = self.resources.get(resource_id)
        if not resource:
            return []

        working_intervals = []
        current_date = start_date.date()
        end_dt = end_date.date()

        while current_date <= end_dt:
            # 检查是否是假期或周末
            date_str = current_date.strftime('%Y-%m-%d')
            if date_str in resource.holidays:
                current_date += timedelta(days=1)
                continue

            if current_date.weekday() in self.weekend_days:
                current_date += timedelta(days=1)
                continue

            # 获取当天的班次
            shift = resource.shifts.get(date_str)
            if not shift:
                current_date += timedelta(days=1)
                continue

            # 添加工作时段
            for slot in shift.work_slots:
                start_dt, end_dt = slot.to_datetime_interval(current_date)
                working_intervals.append((start_dt, end_dt))

            current_date += timedelta(days=1)

        return working_intervals

    def validate(self, assignments: Dict[str, List[Tuple[datetime, datetime]]]) -> Tuple[bool, List[Dict]]:
        """
        验证任务分配是否满足工作时间约束

        参数:
          assignments: {resource_id: [(task_start, task_end), ...]}

        返回:
          (is_valid, violations)
        """
        violations = []

        for resource_id, tasks in assignments.items():
            resource = self.resources.get(resource_id)
            if not resource:
                continue

            for task_start, task_end in tasks:
                # 获取资源在任务时间段的工作时段
                working_intervals = self.get_working_intervals(
                    resource_id, task_start, task_end
                )

                # 计算可工作时长
                available_duration = self._calculate_available_duration(
                    working_intervals, task_start, task_end
                )

                # 计算任务所需时长
                required_duration = task_end - task_start

                # 检查是否有足够的工作时间
                if available_duration < required_duration:
                    shortage = required_duration - available_duration
                    shortage_hours = shortage.total_seconds() / 3600

                    # 检查是否违反加班限制
                    max_overtime = timedelta(hours=resource.max_overtime_hours)
                    if shortage > max_overtime:
                        violations.append({
                            'type': 'insufficient_working_hours',
                            'resource': resource_id,
                            'task_start': task_start,
                            'task_end': task_end,
                            'required_hours': required_duration.total_seconds() / 3600,
                            'available_hours': available_duration.total_seconds() / 3600,
                            'shortage_hours': shortage_hours,
                            'max_overtime_hours': resource.max_overtime_hours,
                            'penalty': shortage_hours * 20
                        })

                # 检查任务是否在非工作时间执行
                non_working_duration = self._calculate_non_working_duration(
                    working_intervals, task_start, task_end
                )

                if non_working_duration > timedelta(0):
                    non_working_hours = non_working_duration.total_seconds() / 3600

                    violations.append({
                        'type': 'non_working_hours_violation',
                        'resource': resource_id,
                        'task_start': task_start,
                        'task_end': task_end,
                        'non_working_hours': non_working_hours,
                        'penalty': non_working_hours * 30
                    })

        return len(violations) == 0, violations

    def _calculate_available_duration(self,
                                     work_intervals: List[Tuple[datetime, datetime]],
                                     task_start: datetime,
                                     task_end: datetime) -> timedelta:
        """计算可工作时长"""
        total_duration = timedelta(0)

        for work_start, work_end in work_intervals:
            # 计算交集
            intersection_start = max(task_start, work_start)
            intersection_end = min(task_end, work_end)

            if intersection_start < intersection_end:
                total_duration += (intersection_end - intersection_start)

        return total_duration

    def _calculate_non_working_duration(self,
                                       work_intervals: List[Tuple[datetime, datetime]],
                                       task_start: datetime,
                                       task_end: datetime) -> timedelta:
        """计算非工作时间执行时长"""
        task_duration = task_end - task_start
        working_duration = self._calculate_available_duration(
            work_intervals, task_start, task_end
        )

        return task_duration - working_duration

    def repair(self, assignments: Dict[str, List[Tuple[datetime, datetime]]],
               violations: List[Dict]) -> Dict[str, List[Tuple[datetime, datetime]]]:
        """
        修复工作时间违反

        策略:
          1. 将任务移动到最近的工作时段
          2. 如果任务跨多个工作时段，拆分任务
          3. 考虑加班限制
        """
        repaired = {}

        for resource_id, tasks in assignments.items():
            resource = self.resources.get(resource_id)
            if not resource:
                repaired[resource_id] = tasks
                continue

            repaired_tasks = []

            for task_start, task_end in tasks:
                task_duration = task_end - task_start

                # 获取工作时段
                working_intervals = self.get_working_intervals(
                    resource_id,
                    task_start,
                    task_start + timedelta(days=7)  # 向后搜索7天
                )

                # 找到第一个可以容纳任务的工作时段
                current_time = task_start
                remaining_duration = task_duration

                for work_start, work_end in working_intervals:
                    if work_start < current_time:
                        work_start = current_time

                    available = work_end - work_start

                    if available <= timedelta(0):
                        continue

                    # 分配任务片段
                    if remaining_duration <= available:
                        # 任务完全适配
                        repaired_tasks.append((work_start, work_start + remaining_duration))
                        break
                    else:
                        # 任务需要拆分
                        repaired_tasks.append((work_start, work_end))
                        remaining_duration -= available
                        current_time = work_end

            repaired[resource_id] = repaired_tasks

        return repaired

    def calculate_shift_coverage(self, resource_id: str,
                                 start_date: datetime,
                                 end_date: datetime) -> float:
        """
        计算班次覆盖率

        返回:
          coverage: 工作时间占总时间的百分比
        """
        working_intervals = self.get_working_intervals(resource_id, start_date, end_date)

        total_working_hours = sum(
            (end - start).total_seconds() / 3600
            for start, end in working_intervals
        )

        total_hours = (end_date - start_date).total_seconds() / 3600

        coverage = (total_working_hours / total_hours * 100) if total_hours > 0 else 0

        return round(coverage, 2)

```

## 📊 应用场景

### 场景1: 服务人员排班

```python
# 定义班次
morning_shift = Shift(
    name="早班",
    work_slots=[
        TimeSlot(time(8, 0), time(12, 0)),
        TimeSlot(time(13, 0), time(17, 0))
    ],
    type=ShiftType.FIXED
)

evening_shift = Shift(
    name="晚班",
    work_slots=[
        TimeSlot(time(17, 0), time(22, 0))
    ],
    type=ShiftType.FIXED
)

# 定义资源
resources = {
    'employee1': Resource(
        id='employee1',
        shifts={
            '2025-01-06': morning_shift,  # 周一早班
            '2025-01-07': morning_shift,  # 周二早班
            '2025-01-08': evening_shift,  # 周三晚班
        },
        holidays={'2025-01-01', '2025-02-10'},
        max_overtime_hours=2
    )
}

handler = WorkingHoursConstraintHandler(resources)

# 验证任务分配
assignments = {
    'employee1': [
        (datetime(2025, 1, 6, 9, 0), datetime(2025, 1, 6, 16, 0)),  # 合法
        (datetime(2025, 1, 7, 18, 0), datetime(2025, 1, 7, 20, 0)),  # 违反：非工作时间
    ]
}

is_valid, violations = handler.validate(assignments)

if not is_valid:
    print(f"发现 {len(violations)} 个违反")
    for v in violations:
        print(f"  {v['type']}: {v['resource']} - {v.get('non_working_hours', 0)} 小时")

    repaired = handler.repair(assignments, violations)
```

## 🎯 性能优化

### 工作时段预计算

```python
class OptimizedWorkingHoursHandler(WorkingHoursConstraintHandler):
    """优化版本 - 预计算工作时段"""

    def __init__(self, *args, **kwargs):
        super().__init__(*args, **kwargs)
        self._interval_cache = {}

    def get_working_intervals(self, resource_id, start_date, end_date):
        cache_key = (resource_id, start_date, end_date)

        if cache_key not in self._interval_cache:
            self._interval_cache[cache_key] = super().get_working_intervals(
                resource_id, start_date, end_date
            )

        return self._interval_cache[cache_key]
```

### 区间树优化查询

```python
from intervaltree import IntervalTree

class IntervalTreeWorkingHoursHandler(WorkingHoursConstraintHandler):
    """使用区间树优化时间段查询"""

    def __init__(self, *args, **kwargs):
        super().__init__(*args, **kwargs)
        self._build_interval_trees()

    def _build_interval_trees(self):
        """为每个资源构建区间树"""
        self.interval_trees = {}

        for resource_id, resource in self.resources.items():
            tree = IntervalTree()

            # 预计算一个月的工作时段
            start_date = datetime.now()
            end_date = start_date + timedelta(days=30)

            intervals = self.get_working_intervals(resource_id, start_date, end_date)

            for start, end in intervals:
                tree.addi(start.timestamp(), end.timestamp(), (start, end))

            self.interval_trees[resource_id] = tree

    def fast_validate(self, assignments):
        """使用区间树快速验证"""
        violations = []

        for resource_id, tasks in assignments.items():
            tree = self.interval_trees.get(resource_id)
            if not tree:
                continue

            for task_start, task_end in tasks:
                # 查询重叠的工作时段
                overlapping = tree.overlap(task_start.timestamp(), task_end.timestamp())

                # 快速计算可工作时长
                available = sum(
                    min(task_end, end).timestamp() - max(task_start, start).timestamp()
                    for _, _, (start, end) in overlapping
                )

                required = (task_end - task_start).total_seconds()

                if available < required:
                    violations.append({
                        'type': 'insufficient_working_hours',
                        'resource': resource_id,
                        'shortage_hours': (required - available) / 3600
                    })

        return len(violations) == 0, violations
```
