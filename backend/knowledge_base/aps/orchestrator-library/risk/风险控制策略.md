## 📖 策略概述

**核心理念**: 建立三层风险控制机制（预防-监控-应对），确保项目风险可控，保障交付质量

**控制层次**:

- 预防性控制：事前预防，降低风险发生概率
- 监控性控制：过程监控，及时发现风险
- 应对性控制：事后应对，快速控制损失

## 🔧 代码实现

```python
from typing import Dict, List
from enum import Enum
from dataclasses import dataclass
from datetime import datetime

class ControlType(Enum):
    """控制类型"""
    PREVENTIVE = "preventive"    # 预防性
    MONITORING = "monitoring"    # 监控性
    RESPONSIVE = "responsive"    # 应对性

class ControlStatus(Enum):
    """控制状态"""
    PLANNED = "planned"
    ACTIVE = "active"
    COMPLETED = "completed"
    FAILED = "failed"

@dataclass
class RiskControl:
    """风险控制措施"""
    id: str
    risk_id: str
    type: ControlType
    description: str
    actions: List[str]
    responsible: str
    deadline: str
    status: ControlStatus
    effectiveness: float = 0.0  # 有效性评分 (0-1)

class RiskControlStrategy:
    """风险控制策略"""

    def __init__(self):
        self.controls = []
        self.monitoring_data = []

    def create_control_plan(self, risks: List) -> Dict:
        """
        创建风险控制计划

        返回:
        {
            'preventive_controls': [...],
            'monitoring_controls': [...],
            'responsive_controls': [...],
            'control_priorities': [...]
        }
        """
        plan = {
            'preventive_controls': [],
            'monitoring_controls': [],
            'responsive_controls': [],
            'control_priorities': []
        }

        for risk in risks:
            # 为每个风险创建三层控制措施
            preventive = self._create_preventive_control(risk)
            monitoring = self._create_monitoring_control(risk)
            responsive = self._create_responsive_control(risk)

            plan['preventive_controls'].append(preventive)
            plan['monitoring_controls'].append(monitoring)
            plan['responsive_controls'].append(responsive)

            self.controls.extend([preventive, monitoring, responsive])

        # 确定控制优先级
        plan['control_priorities'] = self._prioritize_controls(self.controls)

        return plan

    def _create_preventive_control(self, risk) -> RiskControl:
        """创建预防性控制"""
        control_id = f"CTRL-PREV-{risk.id}"

        # 根据风险类别生成预防措施
        preventive_actions = {
            'technical': {
                'algorithm_performance': [
                    '进行算法性能基准测试',
                    '建立性能监控基线',
                    '准备算法降级方案'
                ],
                'integration_complexity': [
                    '制定详细集成计划和时间表',
                    '建立独立集成测试环境',
                    '分阶段渐进式集成'
                ]
            },
            'business': {
                'requirement_understanding': [
                    '组织需求澄清工作坊',
                    '建立需求追溯矩阵',
                    '邀请领域专家参与评审'
                ],
                'user_acceptance': [
                    '早期进行用户调研',
                    '建立用户参与机制',
                    '演示原型获取反馈'
                ]
            },
            'project': {
                'timeline': [
                    '在关键路径上增加缓冲时间',
                    '识别可并行任务',
                    '准备资源加速预案'
                ],
                'resource': [
                    '提前锁定关键资源',
                    '建立资源储备池',
                    '准备外包备选方案'
                ]
            }
        }

        # 获取对应的预防行动
        category_actions = preventive_actions.get(risk.category.value, {})
        risk_key = self._extract_risk_key(risk.name)
        actions = category_actions.get(risk_key, ['制定专项预防计划'])

        return RiskControl(
            id=control_id,
            risk_id=risk.id,
            type=ControlType.PREVENTIVE,
            description=f"{risk.name}的预防性控制",
            actions=actions,
            responsible='项目经理',
            deadline='项目启动后1周内',
            status=ControlStatus.PLANNED
        )

    def _create_monitoring_control(self, risk) -> RiskControl:
        """创建监控性控制"""
        control_id = f"CTRL-MON-{risk.id}"

        monitoring_actions = {
            'technical': [
                '每周进行性能测试',
                '监控系统集成进度',
                '跟踪技术问题清单'
            ],
            'business': [
                '定期收集用户反馈',
                '监控需求变更频率',
                '跟踪业务指标'
            ],
            'project': [
                '每日站会跟踪进度',
                '每周审查资源使用情况',
                '监控关键里程碑'
            ]
        }

        actions = monitoring_actions.get(risk.category.value, ['定期风险评估'])

        return RiskControl(
            id=control_id,
            risk_id=risk.id,
            type=ControlType.MONITORING,
            description=f"{risk.name}的监控性控制",
            actions=actions,
            responsible='质量经理',
            deadline='项目全周期',
            status=ControlStatus.PLANNED
        )

    def _create_responsive_control(self, risk) -> RiskControl:
        """创建应对性控制"""
        control_id = f"CTRL-RESP-{risk.id}"

        responsive_actions = risk.mitigation  # 使用风险的缓解措施作为应对行动

        return RiskControl(
            id=control_id,
            risk_id=risk.id,
            type=ControlType.RESPONSIVE,
            description=f"{risk.name}的应对性控制",
            actions=responsive_actions,
            responsible='技术负责人',
            deadline='风险触发后24小时内',
            status=ControlStatus.PLANNED
        )

    def _extract_risk_key(self, risk_name: str) -> str:
        """从风险名称提取关键字"""
        # 简化映射
        mapping = {
            '算法性能': 'algorithm_performance',
            '集成复杂性': 'integration_complexity',
            '需求理解': 'requirement_understanding',
            '用户接受度': 'user_acceptance',
            '时间': 'timeline',
            '资源': 'resource'
        }

        for key, value in mapping.items():
            if key in risk_name:
                return value

        return 'unknown'

    def _prioritize_controls(self, controls: List[RiskControl]) -> List[str]:
        """确定控制措施优先级"""
        # 优先级规则：
        # 1. 严重风险的预防控制
        # 2. 高风险的预防控制
        # 3. 所有监控控制
        # 4. 应对控制（待命）

        priorities = []

        # 先处理预防控制
        preventive = [c for c in controls if c.type == ControlType.PREVENTIVE]
        priorities.extend([c.id for c in preventive])

        # 然后是监控控制
        monitoring = [c for c in controls if c.type == ControlType.MONITORING]
        priorities.extend([c.id for c in monitoring])

        # 最后是应对控制
        responsive = [c for c in controls if c.type == ControlType.RESPONSIVE]
        priorities.extend([c.id for c in responsive])

        return priorities

    def activate_control(self, control_id: str):
        """激活控制措施"""
        for control in self.controls:
            if control.id == control_id:
                control.status = ControlStatus.ACTIVE
                print(f"已激活控制措施: {control.description}")
                return True
        return False

    def record_monitoring_data(self, control_id: str, data: Dict):
        """记录监控数据"""
        self.monitoring_data.append({
            'control_id': control_id,
            'timestamp': datetime.now().isoformat(),
            'data': data
        })

    def evaluate_control_effectiveness(self, control_id: str) -> float:
        """评估控制措施有效性"""
        # 获取该控制的监控数据
        control_data = [
            d for d in self.monitoring_data
            if d['control_id'] == control_id
        ]

        if not control_data:
            return 0.0

        # 简化评估：基于监控数据的趋势
        # 实际应用中应根据具体指标计算

        # 示例：如果最近的数据显示风险在降低
        latest_data = control_data[-1]['data']
        risk_trend = latest_data.get('risk_trend', 'stable')

        effectiveness_map = {
            'decreasing': 0.9,
            'stable': 0.7,
            'increasing': 0.3
        }

        effectiveness = effectiveness_map.get(risk_trend, 0.5)

        # 更新控制有效性
        for control in self.controls:
            if control.id == control_id:
                control.effectiveness = effectiveness
                break

        return effectiveness

    def generate_control_report(self) -> Dict:
        """生成控制报告"""
        total_controls = len(self.controls)

        status_stats = {}
        type_stats = {}

        for control in self.controls:
            # 按状态统计
            status = control.status.value
            status_stats[status] = status_stats.get(status, 0) + 1

            # 按类型统计
            ctype = control.type.value
            type_stats[ctype] = type_stats.get(ctype, 0) + 1

        # 统计有效性
        active_controls = [c for c in self.controls if c.status == ControlStatus.ACTIVE]
        avg_effectiveness = (
            sum(c.effectiveness for c in active_controls) / len(active_controls)
            if active_controls else 0.0
        )

        return {
            'total_controls': total_controls,
            'status_distribution': status_stats,
            'type_distribution': type_stats,
            'active_controls': len(active_controls),
            'average_effectiveness': avg_effectiveness,
            'monitoring_records': len(self.monitoring_data)
        }
```

## 📋 使用示例

### 场景1: 创建风险控制计划

```python
from dataclasses import dataclass

# 简化的风险对象
@dataclass
class SimpleRisk:
    id: str
    name: str
    category: object
    level: object
    mitigation: list

# 创建风险对象
class Category:
    value = 'technical'

class Level:
    value = 5

risk = SimpleRisk(
    id='RISK-001',
    name='算法性能风险',
    category=Category(),
    level=Level(),
    mitigation=[
        '进行性能基准测试验证',
        '准备算法降级方案'
    ]
)

# 创建控制计划
strategy = RiskControlStrategy()
plan = strategy.create_control_plan([risk])

print(f"风险控制计划:")
print(f"  预防控制: {len(plan['preventive_controls'])}项")
print(f"  监控控制: {len(plan['monitoring_controls'])}项")
print(f"  应对控制: {len(plan['responsive_controls'])}项")

print(f"\n预防控制措施:")
for control in plan['preventive_controls']:
    print(f"  {control.id}: {control.description}")
    for action in control.actions:
        print(f"    - {action}")
```

### 场景2: 监控和评估控制有效性

```python
# 激活监控控制
strategy.activate_control('CTRL-MON-RISK-001')

# 记录监控数据
strategy.record_monitoring_data('CTRL-MON-RISK-001', {
    'risk_trend': 'decreasing',
    'performance_score': 0.92,
    'test_coverage': 0.88
})

# 评估有效性
effectiveness = strategy.evaluate_control_effectiveness('CTRL-MON-RISK-001')
print(f"控制有效性: {effectiveness:.2%}")

# 生成报告
report = strategy.generate_control_report()
print(f"\n控制报告:")
print(f"  总控制数: {report['total_controls']}")
print(f"  活跃控制: {report['active_controls']}")
print(f"  平均有效性: {report['average_effectiveness']:.2%}")
```

## 📊 风险控制框架

| 控制类型   | 时机   | 目标     | 关键行动         |
| ---------- | ------ | -------- | ---------------- |
| **预防性** | 事前   | 降低概率 | 计划、测试、培训 |
| **监控性** | 过程中 | 及时发现 | 跟踪、评估、预警 |
| **应对性** | 事后   | 快速控制 | 响应、恢复、改进 |

## 💡 最佳实践

### 1. 预防优于应对

```
投入比例建议：
- 预防性控制：50%
- 监控性控制：30%
- 应对性控制：20%
```

### 2. 持续监控

```
监控频率：
- 严重风险：每日
- 高风险：每周
- 中低风险：每月
```

### 3. 快速响应

```
响应时间：
- 严重风险触发：4小时内
- 高风险触发：24小时内
- 中风险触发：3天内
```
