## 📖 定义

**资源成本** = ∑(资源单价 × 资源使用量)

## 🔧 代码实现

```python
from typing import Dict, List

class ResourceCostObjective:
    """资源成本目标函数"""

    def __init__(self, resource_prices: Dict[str, float], weight: float = 1.0):
        self.name = "resource_cost"
        self.optimization_type = "minimize"
        self.weight = weight
        self.resource_prices = resource_prices

    def evaluate(self, solution: Dict[str, List[Dict]]) -> float:
        """评估资源成本"""
        total_cost = 0.0

        for resource_id, tasks in solution.items():
            price = self.resource_prices.get(resource_id, 0.0)
            usage = sum(task.get('duration', 0) for task in tasks)
            total_cost += price * usage

        return total_cost
```
