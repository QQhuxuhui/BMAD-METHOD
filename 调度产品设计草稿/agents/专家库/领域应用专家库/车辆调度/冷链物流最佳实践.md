## 📖 最佳实践

### 冷链核心要求

**温度控制**:

```
冷冻: -18°C 至 -25°C
冷藏: 0°C 至 4°C
恒温: 15°C 至 25°C
```

**时效要求**:

- 生鲜蔬菜: 2小时内送达
- 冷冻肉类: 4小时内送达
- 医药疫苗: 全程监控，严格时间窗

## 🔧 代码实现

```python
from dataclasses import dataclass
from enum import Enum

class TemperatureZone(Enum):
    """温区类型"""
    FROZEN = "frozen"      # 冷冻
    CHILLED = "chilled"    # 冷藏
    AMBIENT = "ambient"    # 常温

@dataclass
class ColdChainProduct:
    """冷链产品"""
    id: str
    name: str
    temperature_zone: TemperatureZone
    shelf_life: float  # 小时
    temperature_range: tuple  # (min, max) 温度范围

class ColdChainDomainAdapter:
    """冷链物流领域适配器"""

    def __init__(self):
        self.temperature_monitors = {}

    def validate_temperature_compatibility(self, products: List[ColdChainProduct]) -> bool:
        """验证产品温区兼容性"""
        zones = set(p.temperature_zone for p in products)

        # 同一车次只能装载相同温区的产品
        if len(zones) > 1:
            return False

        return True

    def calculate_max_delivery_time(self, product: ColdChainProduct) -> float:
        """计算最大配送时间"""
        # 冷链产品配送时间不得超过货架期的1/3
        max_time = product.shelf_life / 3
        return max_time

    def generate_cold_chain_constraints(self, products: List[ColdChainProduct]) -> List[Dict]:
        """生成冷链约束"""
        constraints = []

        # 温区兼容性约束
        constraints.append({
            'type': 'temperature_compatibility',
            'validator': lambda route: self.validate_temperature_compatibility(route.products)
        })

        # 时效约束
        for product in products:
            max_time = self.calculate_max_delivery_time(product)
            constraints.append({
                'type': 'shelf_life',
                'product_id': product.id,
                'max_time': max_time
            })

        return constraints
```
