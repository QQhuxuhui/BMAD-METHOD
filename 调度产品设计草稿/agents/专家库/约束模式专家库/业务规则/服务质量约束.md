## 📖 约束概述

### 定义

**服务质量约束**（Service Quality Constraint）规定服务必须达到的质量标准，包括响应时间、可用性、准确性、吞吐量等关键指标。

**数学定义**：

```
对于服务 s 和质量指标集 Q = {q_1, q_2, ..., q_n}

质量约束:
  ∀q_i ∈ Q: actual(q_i) ≥ threshold(q_i)

服务等级协议(SLA):
  SLA_score = ∑ w_i × satisfaction(q_i)
  其中 w_i 为权重，∑w_i = 1
```

### 约束类型

#### 1. **响应时间约束** (Response Time)

```
服务响应时间限制

指标:
  - 平均响应时间 ≤ T_avg
  - 95分位响应时间 ≤ T_p95
  - 最大响应时间 ≤ T_max

例如:
  API响应时间 < 200ms (95%)
  客服响应时间 < 30s
```

#### 2. **可用性约束** (Availability)

```
系统可用时间比例

计算:
  可用性 = (总时间 - 故障时间) / 总时间 × 100%

等级:
  - 99.9% (3个9): 年停机 8.76 小时
  - 99.99% (4个9): 年停机 52.6 分钟
  - 99.999% (5个9): 年停机 5.26 分钟
```

#### 3. **吞吐量约束** (Throughput)

```
单位时间处理能力

指标:
  - TPS (Transactions Per Second)
  - QPS (Queries Per Second)
  - 订单处理量 (Orders/hour)

例如:
  系统支持 10,000 TPS
  配送中心处理 500 订单/小时
```

#### 4. **准确性约束** (Accuracy)

```
服务结果正确性

指标:
  - 错误率 ≤ ε
  - 准确率 ≥ α
  - 缺陷率 ≤ δ

例如:
  预测准确率 ≥ 95%
  产品合格率 ≥ 99%
  配送错误率 ≤ 0.1%
```

## 💡 核心理论

### SLA计算模型

```python
def calculate_sla_score(metrics, thresholds, weights):
    """
    计算SLA得分

    参数:
      metrics: {metric_name: actual_value}
      thresholds: {metric_name: threshold_value}
      weights: {metric_name: weight}

    返回:
      sla_score: 0-100分
    """
    total_score = 0

    for metric_name in metrics:
        actual = metrics[metric_name]
        threshold = thresholds[metric_name]
        weight = weights.get(metric_name, 0)

        # 计算单项满足度
        if actual >= threshold:
            satisfaction = 1.0
        else:
            # 线性衰减
            satisfaction = actual / threshold

        total_score += weight * satisfaction * 100

    return total_score
```

```python
import numpy as np

def calculate_percentile_response_time(response_times, percentile=95):
    """
    计算分位数响应时间

    参数:
      response_times: 响应时间列表（毫秒）
      percentile: 分位数 (如95表示P95)

    返回:
      percentile_value: 分位数值
    """
    return np.percentile(response_times, percentile)

# 示例
response_times = [100, 150, 200, 180, 220, 500, 300]
p95 = calculate_percentile_response_time(response_times, 95)
# p95 = 470ms (95%的请求在470ms内完成)
```

### 可用性计算

```python
def calculate_availability(total_time, downtime):
    """
    计算可用性

    参数:
      total_time: 总时间（小时）
      downtime: 故障时间（小时）

    返回:
      availability: 可用性百分比
    """
    if total_time <= 0:
        return 0

    availability = (total_time - downtime) / total_time * 100
    return availability

# 计算允许的停机时间
def allowed_downtime(availability_target, time_period_hours):
    """
    根据可用性目标计算允许的停机时间

    参数:
      availability_target: 可用性目标 (如99.9)
      time_period_hours: 时间周期（小时）

    返回:
      allowed_downtime_hours: 允许停机时间（小时）
    """
    unavailability = (100 - availability_target) / 100
    return time_period_hours * unavailability

# 年度停机时间
one_year_hours = 365 * 24
print(f"99.9%可用性允许停机: {allowed_downtime(99.9, one_year_hours):.2f}小时")
# 输出: 8.76小时
```

## 🔧 代码实现

```python
from typing import Dict, List, Tuple, Optional
from dataclasses import dataclass, field
from datetime import datetime, timedelta
from enum import Enum
import statistics

class QualityMetric(Enum):
    """质量指标类型"""
    RESPONSE_TIME = "response_time"
    AVAILABILITY = "availability"
    THROUGHPUT = "throughput"
    ACCURACY = "accuracy"
    ERROR_RATE = "error_rate"

@dataclass
class SLATarget:
    """SLA目标"""
    metric: QualityMetric
    threshold: float
    weight: float = 0.25
    unit: str = ""

@dataclass
class QualityMeasurement:
    """质量测量"""
    metric: QualityMetric
    value: float
    timestamp: datetime = field(default_factory=datetime.now)
    unit: str = ""

class ServiceQualityConstraintHandler:
    """服务质量约束处理器"""

    def __init__(self, sla_targets: List[SLATarget]):
        """
        参数:
          sla_targets: SLA目标列表
        """
        self.sla_targets = {target.metric: target for target in sla_targets}
        self.measurements: Dict[QualityMetric, List[float]] = {
            metric: [] for metric in QualityMetric
        }

    def record_measurement(self, measurement: QualityMeasurement):
        """记录质量测量值"""
        if measurement.metric not in self.measurements:
            self.measurements[measurement.metric] = []

        self.measurements[measurement.metric].append(measurement.value)

    def validate(self, current_metrics: Dict[QualityMetric, float]) -> Tuple[bool, List[Dict]]:
        """
        验证当前质量指标是否满足SLA

        参数:
          current_metrics: {metric: value}

        返回:
          (meets_sla, violations)
        """
        violations = []

        for metric, value in current_metrics.items():
            target = self.sla_targets.get(metric)

            if not target:
                continue

            # 检查是否满足阈值
            meets_threshold = value >= target.threshold

            if not meets_threshold:
                gap = target.threshold - value
                gap_percent = (gap / target.threshold) * 100

                violations.append({
                    'type': 'sla_violation',
                    'metric': metric.value,
                    'actual': value,
                    'threshold': target.threshold,
                    'gap': gap,
                    'gap_percent': round(gap_percent, 2),
                    'unit': target.unit,
                    'penalty': gap_percent * target.weight * 10
                })

        return len(violations) == 0, violations

    def calculate_sla_score(self, current_metrics: Dict[QualityMetric, float]) -> float:
        """
        计算SLA得分 (0-100)

        参数:
          current_metrics: {metric: value}

        返回:
          sla_score: SLA得分
        """
        total_score = 0

        for metric, value in current_metrics.items():
            target = self.sla_targets.get(metric)

            if not target:
                continue

            # 计算满足度
            if value >= target.threshold:
                satisfaction = 1.0
            else:
                # 线性衰减
                satisfaction = max(0, value / target.threshold)

            # 加权得分
            total_score += target.weight * satisfaction * 100

        return round(total_score, 2)

    def calculate_response_time_percentile(self, percentile: int = 95) -> Optional[float]:
        """
        计算响应时间分位数

        参数:
          percentile: 分位数 (50, 95, 99)

        返回:
          percentile_value: 分位数值
        """
        response_times = self.measurements.get(QualityMetric.RESPONSE_TIME, [])

        if not response_times:
            return None

        sorted_times = sorted(response_times)
        index = int(len(sorted_times) * percentile / 100)

        return sorted_times[min(index, len(sorted_times) - 1)]

    def calculate_availability(self, total_time_hours: float,
                               downtime_hours: float) -> float:
        """
        计算可用性

        参数:
          total_time_hours: 总时间（小时）
          downtime_hours: 停机时间（小时）

        返回:
          availability: 可用性百分比
        """
        if total_time_hours <= 0:
            return 0

        availability = (total_time_hours - downtime_hours) / total_time_hours * 100
        return round(availability, 4)

    def calculate_error_rate(self, total_requests: int,
                            failed_requests: int) -> float:
        """
        计算错误率

        参数:
          total_requests: 总请求数
          failed_requests: 失败请求数

        返回:
          error_rate: 错误率百分比
        """
        if total_requests <= 0:
            return 0

        error_rate = (failed_requests / total_requests) * 100
        return round(error_rate, 4)

    def generate_quality_report(self, current_metrics: Dict[QualityMetric, float]) -> str:
        """
        生成质量报告

        参数:
          current_metrics: 当前指标值

        返回:
          report: 质量报告文本
        """
        sla_score = self.calculate_sla_score(current_metrics)
        meets_sla, violations = self.validate(current_metrics)

        report = f"📊 服务质量报告\n\n"
        report += f"SLA总分: {sla_score}/100\n"
        report += f"状态: {'✅ 达标' if meets_sla else '❌ 未达标'}\n\n"

        report += "质量指标:\n"

        for metric, value in current_metrics.items():
            target = self.sla_targets.get(metric)

            if target:
                status = "✅" if value >= target.threshold else "❌"
                report += f"  {status} {metric.value}: {value}{target.unit} "
                report += f"(目标: ≥{target.threshold}{target.unit})\n"

        if violations:
            report += f"\n⚠️ 违反项 ({len(violations)}):\n"

            for v in violations:
                report += f"  - {v['metric']}: 实际{v['actual']}{v['unit']}, "
                report += f"目标{v['threshold']}{v['unit']}, "
                report += f"差距{v['gap_percent']}%\n"

        return report

    def get_quality_statistics(self, metric: QualityMetric) -> Dict:
        """
        获取质量指标统计

        参数:
          metric: 质量指标

        返回:
          statistics: 统计信息
        """
        values = self.measurements.get(metric, [])

        if not values:
            return {}

        return {
            'count': len(values),
            'mean': statistics.mean(values),
            'median': statistics.median(values),
            'stdev': statistics.stdev(values) if len(values) > 1 else 0,
            'min': min(values),
            'max': max(values),
            'p50': statistics.median(values),
            'p95': sorted(values)[int(len(values) * 0.95)] if len(values) > 1 else values[0],
            'p99': sorted(values)[int(len(values) * 0.99)] if len(values) > 1 else values[0]
        }

```

## 📊 应用场景

### 场景1: API服务SLA

```python
# 定义SLA目标
sla_targets = [
    SLATarget(metric=QualityMetric.RESPONSE_TIME, threshold=200, weight=0.3, unit="ms"),
    SLATarget(metric=QualityMetric.AVAILABILITY, threshold=99.9, weight=0.4, unit="%"),
    SLATarget(metric=QualityMetric.THROUGHPUT, threshold=10000, weight=0.2, unit=" TPS"),
    SLATarget(metric=QualityMetric.ERROR_RATE, threshold=0.1, weight=0.1, unit="%"),
]

handler = ServiceQualityConstraintHandler(sla_targets)

# 记录测量值
handler.record_measurement(QualityMeasurement(
    metric=QualityMetric.RESPONSE_TIME,
    value=180,
    unit="ms"
))

# 当前指标
current_metrics = {
    QualityMetric.RESPONSE_TIME: 180,     # 达标
    QualityMetric.AVAILABILITY: 99.95,    # 达标
    QualityMetric.THROUGHPUT: 12000,      # 达标
    QualityMetric.ERROR_RATE: 0.05,       # 达标
}

# 验证SLA
meets_sla, violations = handler.validate(current_metrics)
sla_score = handler.calculate_sla_score(current_metrics)

print(f"SLA得分: {sla_score}")
print(f"达标状态: {'达标' if meets_sla else '未达标'}")

# 生成报告
report = handler.generate_quality_report(current_metrics)
print(report)
```

## 🎯 性能优化

### 滑动窗口统计

```python
from collections import deque

class SlidingWindowQualityHandler(ServiceQualityConstraintHandler):
    """滑动窗口质量处理器"""

    def __init__(self, sla_targets, window_size=1000):
        super().__init__(sla_targets)
        self.window_size = window_size
        self.sliding_windows = {
            metric: deque(maxlen=window_size)
            for metric in QualityMetric
        }

    def record_measurement(self, measurement: QualityMeasurement):
        """使用滑动窗口记录"""
        if measurement.metric not in self.sliding_windows:
            self.sliding_windows[measurement.metric] = deque(maxlen=self.window_size)

        self.sliding_windows[measurement.metric].append(measurement.value)

    def get_recent_average(self, metric: QualityMetric) -> Optional[float]:
        """获取最近N次测量的平均值"""
        window = self.sliding_windows.get(metric, [])

        if not window:
            return None

        return sum(window) / len(window)
```
