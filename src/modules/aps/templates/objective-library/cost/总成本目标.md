## 📖 定义

**总成本** = 资源成本 + 任务成本 + 惩罚成本

```
Total Cost = ∑(resource_cost_i × usage_time_i) + ∑task_cost_j + ∑penalty_k
```

## 🔧 代码实现

```python
from typing import Dict, List

class TotalCostObjective:
    """总成本目标函数"""

    def __init__(self, resource_costs: Dict[str, float], task_costs: Dict[str, float], weight: float = 1.0):
        self.name = "total_cost"
        self.optimization_type = "minimize"
        self.weight = weight
        self.resource_costs = resource_costs  # {resource_id: hourly_cost}
        self.task_costs = task_costs          # {task_id: fixed_cost}

    def evaluate(self, solution: Dict[str, List[Dict]]) -> float:
        """评估总成本"""
        total_cost = 0.0

        for resource_id, tasks in solution.items():
            # 资源使用成本
            resource_cost = self.resource_costs.get(resource_id, 0.0)
            resource_usage_time = 0.0

            for task in tasks:
                # 任务固定成本
                task_id = task.get('id', '')
                task_fixed_cost = self.task_costs.get(task_id, 0.0)
                total_cost += task_fixed_cost

                # 累计资源使用时间
                duration = task.get('duration', 0)
                resource_usage_time += duration

            # 资源使用成本 = 单位成本 × 使用时间
            total_cost += resource_cost * resource_usage_time

        return total_cost
```

## 📊 示例

```python
# 资源成本配置
resource_costs = {
    'M1': 100,  # 100元/小时
    'M2': 150,
    'M3': 120
}

# 任务成本配置
task_costs = {
    'T1': 50,   # 固定成本50元
    'T2': 30,
    'T3': 40
}

# 调度方案
solution = {
    'M1': [{'id': 'T1', 'duration': 2}, {'id': 'T2', 'duration': 3}],
    'M2': [{'id': 'T3', 'duration': 4}]
}

# 评估
objective = TotalCostObjective(resource_costs, task_costs)
cost = objective.evaluate(solution)
print(f"总成本: {cost}元")
# 输出: 总成本: 1220元 (M1: 100*5+50+30=580, M2: 150*4+40=640)
```
