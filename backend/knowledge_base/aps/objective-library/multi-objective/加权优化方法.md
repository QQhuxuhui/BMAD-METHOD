## 📖 定义

**加权求和法**: 将多个目标通过权重组合成单一目标

```
F(x) = ∑ w_i · f_i(x)

其中:
  w_i = 目标i的权重 (∑w_i = 1)
  f_i(x) = 目标i的归一化值
```

## 💡 权重设计方法

### 层次分析法 (AHP)

```python
import numpy as np

def ahp_weights(comparison_matrix):
    """
    AHP法计算权重

    参数:
      comparison_matrix: 成对比较矩阵 (n×n)
        元素 a_ij 表示目标i相对j的重要性
        1: 同等重要, 3: 稍微重要, 5: 明显重要, 7: 强烈重要, 9: 极端重要

    返回:
      权重向量
    """
    # 计算特征向量
    eigenvalues, eigenvectors = np.linalg.eig(comparison_matrix)

    # 最大特征值对应的特征向量
    max_index = np.argmax(eigenvalues)
    weights = np.abs(eigenvectors[:, max_index].real)

    # 归一化
    weights = weights / weights.sum()

    return weights
```

## 🔧 代码实现

```python
from typing import Dict, List

class WeightedSumOptimizer:
    """加权求和优化器"""

    def __init__(self, objectives: List, weights: List[float]):
        """
        参数:
          objectives: 目标函数列表
          weights: 权重列表（与objectives对应）
        """
        self.objectives = objectives
        self.weights = weights

        # 验证权重和为1
        assert abs(sum(weights) - 1.0) < 1e-6, "权重之和必须为1"

    def evaluate(self, solution: Dict) -> float:
        """
        计算加权目标值

        参数:
          solution: 包含所有目标评估值的解

        返回:
          加权总目标值
        """
        weighted_sum = 0.0

        for obj, weight in zip(self.objectives, self.weights):
            value = solution.get(obj.name, 0)

            # 处理最大化目标（转换为最小化）
            if obj.optimization_type == "maximize":
                value = -value

            weighted_sum += weight * value

        return weighted_sum

    def normalize(self, solutions: List[Dict]) -> List[Dict]:
        """归一化目标值到[0,1]区间"""
        normalized = []

        # 计算每个目标的最小值和最大值
        min_max = {}
        for obj in self.objectives:
            values = [sol.get(obj.name, 0) for sol in solutions]
            min_max[obj.name] = (min(values), max(values))

        # 归一化
        for sol in solutions:
            norm_sol = sol.copy()

            for obj in self.objectives:
                val = sol.get(obj.name, 0)
                min_val, max_val = min_max[obj.name]

                if max_val - min_val > 0:
                    norm_val = (val - min_val) / (max_val - min_val)
                else:
                    norm_val = 0.0

                norm_sol[f"{obj.name}_normalized"] = norm_val

            normalized.append(norm_sol)

        return normalized
```

## 📊 示例

```python
# 定义目标和权重
objectives = [
    Objective("cost", "minimize"),      # 成本
    Objective("time", "minimize"),      # 时间
    Objective("quality", "maximize")    # 质量
]

weights = [0.5, 0.3, 0.2]  # 成本最重要

# 创建优化器
optimizer = WeightedSumOptimizer(objectives, weights)

# 评估解
solution = {"cost": 100, "time": 10, "quality": 80}
score = optimizer.evaluate(solution)
print(f"加权评分: {score}")
```
