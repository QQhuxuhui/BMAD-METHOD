### SSE Hook实现策略

**选项1：使用@vueuse/core的useEventSource** [推荐]:

```typescript
import { useEventSource } from '@vueuse/core';

const { data, error, status, close } = useEventSource(
  'http://localhost:8000/api/v1/workflows/123/stream',
  [], // events to listen
  {
    withCredentials: true,
    headers: {
      Authorization: `Bearer ${token}`,
    },
  },
);
```

**优点**:

- 自动管理连接生命周期
- 内置retry逻辑
- TypeScript支持完善
- 与Vue 3 Composition API无缝集成

**选项2：手动实现EventSource**:

```typescript
const eventSource = new EventSource(url);
eventSource.onmessage = (event) => {
  const data = JSON.parse(event.data);
  // 处理事件
};
eventSource.onerror = () => {
  // 错误处理
};
```

**本Story采用**：选项1（@vueuse/core）+ 业务逻辑封装

### useWorkflowStream Composable设计

**文件位置**: `frontend/web/src/composables/useWorkflowStream.ts`

**完整实现框架**:

```typescript
import { ref, computed, onUnmounted } from 'vue';
import { useEventSource } from '@vueuse/core';
import type { WorkflowEvent } from '@/types/workflow';

export interface UseWorkflowStreamOptions {
  workflowId: string;
  autoConnect?: boolean;
}

export function useWorkflowStream(options: UseWorkflowStreamOptions) {
  const { workflowId, autoConnect = true } = options;

  // State
  const events = ref<WorkflowEvent[]>([]);
  const currentPhase = ref<string>('P0');
  const currentAgent = ref<string | null>(null);
  const isApprovalRequired = ref(false);
  const approvalContext = ref<any>(null);

  // 构建SSE URL
  const token = localStorage.getItem('token');
  const url = computed(() => `${import.meta.env.VITE_API_BASE_URL}/v1/workflows/${workflowId}/stream`);

  // 使用@vueuse的useEventSource
  const { data, status, error, close, open } = useEventSource(url, [], {
    immediate: autoConnect,
    withCredentials: true,
    headers: {
      Authorization: `Bearer ${token}`,
    },
  });

  // 监听data变化并解析事件
  watch(data, (newData) => {
    if (!newData) return;

    try {
      const event: WorkflowEvent = JSON.parse(newData);
      events.value.push(event);

      // 根据event type更新状态
      switch (event.event) {
        case 'phase_changed':
          currentPhase.value = event.data.phase;
          break;
        case 'agent_started':
          currentAgent.value = event.data.agent_name;
          break;
        case 'agent_completed':
          currentAgent.value = null;
          break;
        case 'approval_required':
          isApprovalRequired.value = true;
          approvalContext.value = event.data;
          break;
        case 'workflow_completed':
          close();
          break;
      }
    } catch (e) {
      console.error('Failed to parse SSE event:', e);
    }
  });

  // 计算属性
  const isConnected = computed(() => status.value === 'OPEN');
  const latestEvent = computed(() => events.value[events.value.length - 1]);

  // Methods
  const connect = () => open();
  const disconnect = () => close();
  const clearEvents = () => {
    events.value = [];
  };

  // 清理
  onUnmounted(() => {
    close();
  });

  return {
    // State
    events,
    currentPhase,
    currentAgent,
    isApprovalRequired,
    approvalContext,
    isConnected,
    latestEvent,
    error,
    status,
    // Methods
    connect,
    disconnect,
    clearEvents,
  };
}
```

### SSE Event类型定义

**文件位置**: `frontend/web/src/types/workflow.ts`

```typescript
export type WorkflowEventType =
  | 'workflow_started'
  | 'phase_changed'
  | 'agent_started'
  | 'agent_output'
  | 'agent_completed'
  | 'approval_required'
  | 'workflow_completed'
  | 'workflow_failed';

export interface WorkflowEvent {
  event: WorkflowEventType;
  data: {
    timestamp: string;
    workflow_id: string;
    phase?: string;
    agent_name?: string;
    agent_output?: any;
    approval_point?: 'P1' | 'P2' | 'P2.5';
    approval_context?: any;
    error_message?: string;
  };
}

export interface ApprovalDecision {
  decision: 'approved' | 'rejected' | 'modified';
  feedback?: string;
  modified_data?: Record<string, any>;
}
```

### HITL ApprovalModal组件设计

**文件位置**: `frontend/web/src/components/workflow/ApprovalModal.vue`

```vue
<template>
  <a-modal :visible="visible" :title="`${approvalPoint} 审批点 - 人工确认`" width="800px" :closable="false" :maskClosable="false">
    <template #footer>
      <a-space>
        <a-button @click="handleDecision('rejected')" danger>拒绝</a-button>
        <a-button @click="handleDecision('approved')" type="primary">批准</a-button>
        <a-button @click="showModifyModal = true">修改后批准</a-button>
      </a-space>
    </template>

    <a-descriptions :column="1" bordered>
      <a-descriptions-item label="审批点">{{ approvalPoint }}</a-descriptions-item>
      <a-descriptions-item label="工作流ID">{{ workflowId }}</a-descriptions-item>
      <a-descriptions-item label="当前阶段">{{ currentPhase }}</a-descriptions-item>
    </a-descriptions>

    <a-divider>智能体输出上下文</a-divider>

    <a-collapse v-if="approvalContext">
      <a-collapse-panel v-for="(value, key) in approvalContext" :key="key" :header="key">
        <pre>{{ JSON.stringify(value, null, 2) }}</pre>
      </a-collapse-panel>
    </a-collapse>

    <a-form-item label="审批意见（可选）">
      <a-textarea v-model:value="feedback" :rows="4" />
    </a-form-item>
  </a-modal>

  <!-- 修改数据Modal -->
  <a-modal v-model:visible="showModifyModal" title="修改智能体输出" width="900px" @ok="handleDecision('modified')">
    <a-alert message="请修改JSON数据后提交" type="info" show-icon style="margin-bottom: 16px" />
    <a-textarea v-model:value="modifiedDataJson" :rows="20" placeholder="{}" style="font-family: monospace" />
  </a-modal>
</template>

<script setup lang="ts">
import { ref, computed, watch } from 'vue';
import { message } from 'ant-design-vue';
import { workflowService } from '@/services/workflowService';
import type { ApprovalDecision } from '@/types/workflow';

const props = defineProps<{
  visible: boolean;
  workflowId: string;
  approvalPoint: 'P1' | 'P2' | 'P2.5';
  approvalContext: any;
  currentPhase: string;
}>();

const emit = defineEmits<{
  (e: 'update:visible', value: boolean): void;
  (e: 'submitted', decision: ApprovalDecision): void;
}>();

const feedback = ref('');
const showModifyModal = ref(false);
const modifiedDataJson = ref('');

// 初始化修改数据
watch(
  () => props.approvalContext,
  (ctx) => {
    if (ctx) {
      modifiedDataJson.value = JSON.stringify(ctx, null, 2);
    }
  },
  { immediate: true },
);

const handleDecision = async (decision: ApprovalDecision['decision']) => {
  try {
    const payload: ApprovalDecision = { decision };

    if (feedback.value) {
      payload.feedback = feedback.value;
    }

    if (decision === 'modified') {
      try {
        payload.modified_data = JSON.parse(modifiedDataJson.value);
      } catch (e) {
        message.error('JSON格式错误，请检查');
        return;
      }
    }

    await workflowService.resumeWorkflow(props.workflowId, payload);

    message.success('审批已提交');
    emit('submitted', payload);
    emit('update:visible', false);

    // 重置状态
    feedback.value = '';
    showModifyModal.value = false;
  } catch (error) {
    message.error('提交失败，请重试');
    console.error('Approval submission failed:', error);
  }
};
</script>
```

```vue
// 监听审批需求
watch(isApprovalRequired, (required) => {
  if (required) {
    approvalModalVisible.value = true;
  }
});

const handleViewDetail = (record) => {
  currentWorkflowId.value = record.id;
  detailVisible.value = true;
  connect(); // 建立SSE连接
};

// 关闭Drawer时断开SSE
watch(detailVisible, (visible) => {
  if (!visible) {
    disconnect();
  }
});

onMounted(async () => {
  loading.value = true;
  try {
    const response = await workflowService.getWorkflows();
    workflows.value = response.data;
  } finally {
    loading.value = false;
  }
});
</script>
```

### 限制和遗留问题

**本Story不处理**（留给Story 1.8）:

- 性能优化（虚拟滚动、节流）
- 多工作流并行监控
- 工作流对比功能
- 高级分析和统计图表

**已知限制**:

- EventSource不支持自定义headers（需后端支持query params传token，或使用cookie）
  - **解决方案**：后端支持`?token=xxx`查询参数
- @antv/g6文件较大（~500KB），可能影响首屏加载
  - **解决方案**：路由懒加载
- 浏览器通知需要用户授权，可能被拒绝
  - **解决方案**：降级为Ant Design notification
