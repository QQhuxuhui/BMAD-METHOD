## 📖 策略概述

**核心理念**: 整合多专家建议，提取各方案优势，通过智能融合生成统一的高质量解决方案

**融合技术**:

- 加权融合：基于专家权威性和方案质量
- 优势互补：提取各方案的优势部分
- 分层融合：在不同层次上融合建议
- 演化融合：通过迭代产生更优方案

## 🔧 代码实现

```python
from typing import List, Dict
from enum import Enum

class FusionStrategy(Enum):
    """融合策略类型"""
    WEIGHTED = "weighted_fusion"          # 加权融合
    COMPLEMENTARY = "complementary_fusion"  # 优势互补
    LAYERED = "layered_fusion"            # 分层融合
    EVOLUTIONARY = "evolutionary_fusion"   # 演化融合

class SolutionIntegration:
    """方案融合策略"""

    def __init__(self, strategy: FusionStrategy = FusionStrategy.WEIGHTED):
        self.strategy = strategy
        self.expert_weights = self._initialize_expert_weights()

    def _initialize_expert_weights(self) -> Dict[str, float]:
        """初始化专家权重"""
        return {
            'domain': 0.25,      # 领域专家
            'constraint': 0.20,  # 约束专家
            'algorithm': 0.30,   # 算法专家（权重最高）
            'objective': 0.15,   # 目标专家
            'extension': 0.10    # 扩展专家
        }

    def fuse(self, expert_suggestions: Dict) -> Dict:
        """
        融合多专家建议

        返回:
        {
            'integrated_solution': {...},
            'fusion_confidence': float,
            'contributing_experts': [...],
            'fusion_notes': [...]
        }
        """
        if self.strategy == FusionStrategy.WEIGHTED:
            return self._weighted_fusion(expert_suggestions)
        elif self.strategy == FusionStrategy.COMPLEMENTARY:
            return self._complementary_fusion(expert_suggestions)
        elif self.strategy == FusionStrategy.LAYERED:
            return self._layered_fusion(expert_suggestions)
        elif self.strategy == FusionStrategy.EVOLUTIONARY:
            return self._evolutionary_fusion(expert_suggestions)

    def _weighted_fusion(self, suggestions: Dict) -> Dict:
        """加权融合策略"""
        solution = {
            'algorithm': None,
            'constraints': [],
            'objectives': [],
            'domain_features': {},
            'confidence': 0.0
        }

        total_weight = 0.0
        contributing_experts = []

        # 算法选择：基于权重最高的推荐
        algo_votes = {}
        for expert, suggestion in suggestions.items():
            if 'recommended_algorithm' in suggestion:
                algo = suggestion['recommended_algorithm']
                weight = self.expert_weights.get(expert, 0.1)

                if algo not in algo_votes:
                    algo_votes[algo] = 0
                algo_votes[algo] += weight

        if algo_votes:
            solution['algorithm'] = max(algo_votes, key=algo_votes.get)
            contributing_experts.append('algorithm')

        # 约束整合：合并所有约束
        for expert, suggestion in suggestions.items():
            if 'constraints' in suggestion:
                solution['constraints'].extend(suggestion['constraints'])
                if expert == 'constraint':
                    contributing_experts.append('constraint')

        # 去重约束
        solution['constraints'] = list(set(solution['constraints']))

        # 目标整合：加权目标函数
        objective_weights = {}
        for expert, suggestion in suggestions.items():
            if 'objectives' in suggestion:
                weight = self.expert_weights.get(expert, 0.1)
                for obj in suggestion['objectives']:
                    if obj not in objective_weights:
                        objective_weights[obj] = 0
                    objective_weights[obj] += weight

        solution['objectives'] = list(objective_weights.keys())
        solution['objective_weights'] = objective_weights

        if objective_weights:
            contributing_experts.append('objective')

        # 领域特征整合
        if 'domain' in suggestions:
            solution['domain_features'] = suggestions['domain'].get('features', {})
            contributing_experts.append('domain')

        # 计算融合置信度
        solution['confidence'] = self._calculate_fusion_confidence(
            suggestions, contributing_experts
        )

        return {
            'integrated_solution': solution,
            'fusion_confidence': solution['confidence'],
            'contributing_experts': list(set(contributing_experts)),
            'fusion_notes': ['使用加权融合策略']
        }

    def _complementary_fusion(self, suggestions: Dict) -> Dict:
        """优势互补融合策略"""
        solution = {
            'components': {},
            'confidence': 0.0
        }

        # 提取各专家的优势部分
        strengths = {
            'domain': ['domain_identification', 'business_rules'],
            'constraint': ['constraint_modeling', 'feasibility_check'],
            'algorithm': ['algorithm_selection', 'performance_optimization'],
            'objective': ['multi_objective', 'tradeoff_analysis']
        }

        for expert, suggestion in suggestions.items():
            expert_strengths = strengths.get(expert, [])

            for strength in expert_strengths:
                if strength in suggestion:
                    solution['components'][strength] = {
                        'from_expert': expert,
                        'content': suggestion[strength],
                        'confidence': suggestion.get('confidence', 0.9)
                    }

        # 计算整体置信度
        if solution['components']:
            confidences = [c['confidence'] for c in solution['components'].values()]
            solution['confidence'] = sum(confidences) / len(confidences)

        return {
            'integrated_solution': solution,
            'fusion_confidence': solution['confidence'],
            'contributing_experts': list(suggestions.keys()),
            'fusion_notes': ['使用优势互补融合策略']
        }

    def _layered_fusion(self, suggestions: Dict) -> Dict:
        """分层融合策略"""
        solution = {
            'layers': {
                'domain_layer': {},
                'constraint_layer': {},
                'algorithm_layer': {},
                'objective_layer': {}
            },
            'confidence': 0.0
        }

        # 领域层融合
        if 'domain' in suggestions:
            solution['layers']['domain_layer'] = {
                'analysis': suggestions['domain'],
                'confidence': suggestions['domain'].get('confidence', 0.9)
            }

        # 约束层融合
        constraint_suggestions = []
        for expert, suggestion in suggestions.items():
            if 'constraints' in suggestion:
                constraint_suggestions.append({
                    'expert': expert,
                    'constraints': suggestion['constraints']
                })

        solution['layers']['constraint_layer'] = {
            'sources': constraint_suggestions,
            'merged': self._merge_constraints(constraint_suggestions)
        }

        # 算法层融合
        if 'algorithm' in suggestions:
            solution['layers']['algorithm_layer'] = {
                'design': suggestions['algorithm'],
                'confidence': suggestions['algorithm'].get('confidence', 0.9)
            }

        # 目标层融合
        if 'objective' in suggestions:
            solution['layers']['objective_layer'] = {
                'optimization': suggestions['objective'],
                'confidence': suggestions['objective'].get('confidence', 0.9)
            }

        # 计算分层置信度
        layer_confidences = []
        for layer in solution['layers'].values():
            if isinstance(layer, dict) and 'confidence' in layer:
                layer_confidences.append(layer['confidence'])

        solution['confidence'] = (
            sum(layer_confidences) / len(layer_confidences)
            if layer_confidences else 0.8
        )

        return {
            'integrated_solution': solution,
            'fusion_confidence': solution['confidence'],
            'contributing_experts': list(suggestions.keys()),
            'fusion_notes': ['使用分层融合策略']
        }

    def _evolutionary_fusion(self, suggestions: Dict) -> Dict:
        """演化融合策略（迭代优化）"""
        # 初始种群：各专家建议
        population = list(suggestions.values())

        # 迭代优化
        iterations = 3
        for i in range(iterations):
            # 评估当前种群
            scored_pop = self._score_population(population)

            # 选择优秀个体
            elite = self._select_elite(scored_pop, ratio=0.5)

            # 交叉生成新个体
            offspring = self._crossover(elite)

            # 更新种群
            population = elite + offspring

        # 选择最佳方案
        best_solution = max(
            self._score_population(population),
            key=lambda x: x['score']
        )

        return {
            'integrated_solution': best_solution['solution'],
            'fusion_confidence': best_solution['score'],
            'contributing_experts': list(suggestions.keys()),
            'fusion_notes': [
                f'使用演化融合策略，经过{iterations}轮迭代'
            ]
        }

    def _merge_constraints(self, constraint_suggestions: List[Dict]) -> List[str]:
        """合并约束列表"""
        merged = set()
        for suggestion in constraint_suggestions:
            merged.update(suggestion['constraints'])
        return list(merged)

    def _calculate_fusion_confidence(self, suggestions: Dict,
                                     contributing: List[str]) -> float:
        """计算融合置信度"""
        confidences = []

        for expert in contributing:
            if expert in suggestions:
                expert_conf = suggestions[expert].get('confidence', 0.9)
                expert_weight = self.expert_weights.get(expert, 0.1)
                confidences.append(expert_conf * expert_weight)

        if confidences:
            return sum(confidences) / sum(
                self.expert_weights.get(e, 0.1) for e in contributing
            )

        return 0.8  # 默认置信度

    def _score_population(self, population: List[Dict]) -> List[Dict]:
        """评分种群"""
        scored = []
        for solution in population:
            score = solution.get('confidence', 0.8)
            scored.append({'solution': solution, 'score': score})
        return scored

    def _select_elite(self, scored_pop: List[Dict], ratio: float) -> List[Dict]:
        """选择优秀个体"""
        count = max(1, int(len(scored_pop) * ratio))
        sorted_pop = sorted(scored_pop, key=lambda x: x['score'], reverse=True)
        return [item['solution'] for item in sorted_pop[:count]]

    def _crossover(self, elite: List[Dict]) -> List[Dict]:
        """交叉生成新个体"""
        offspring = []

        for i in range(len(elite) - 1):
            parent1 = elite[i]
            parent2 = elite[i + 1]

            # 简化交叉：合并父代特征
            child = {
                'algorithm': parent1.get('recommended_algorithm'),
                'constraints': list(set(
                    parent1.get('constraints', []) +
                    parent2.get('constraints', [])
                )),
                'confidence': (
                    parent1.get('confidence', 0.8) +
                    parent2.get('confidence', 0.8)
                ) / 2
            }
            offspring.append(child)

        return offspring
```

## 📋 使用示例

### 场景1: 加权融合多专家建议

```python
integrator = SolutionIntegration(strategy=FusionStrategy.WEIGHTED)

# 专家建议
suggestions = {
    'domain': {
        'domain': 'VRP',
        'features': {'cold_chain': True},
        'confidence': 0.92
    },
    'constraint': {
        'constraints': ['vehicle_capacity', 'time_window', 'temperature'],
        'confidence': 0.90
    },
    'algorithm': {
        'recommended_algorithm': 'hybrid_genetic',
        'confidence': 0.95
    },
    'objective': {
        'objectives': ['minimize_cost', 'maximize_service'],
        'confidence': 0.88
    }
}

# 融合方案
result = integrator.fuse(suggestions)

print(f"融合置信度: {result['fusion_confidence']:.2%}")
print(f"贡献专家: {result['contributing_experts']}")
print(f"融合方案:")
print(f"  算法: {result['integrated_solution']['algorithm']}")
print(f"  约束: {result['integrated_solution']['constraints']}")
print(f"  目标: {result['integrated_solution']['objectives']}")
```

### 场景2: 优势互补融合

```python
integrator = SolutionIntegration(strategy=FusionStrategy.COMPLEMENTARY)

result = integrator.fuse(suggestions)

print(f"融合组件:")
for component, details in result['integrated_solution']['components'].items():
    print(f"  {component}: 来自 {details['from_expert']}")
```

## 📊 融合策略对比

| 策略         | 优势     | 适用场景       | 复杂度 |
| ------------ | -------- | -------------- | ------ |
| **加权融合** | 简单高效 | 专家权威度明确 | 低     |
| **优势互补** | 充分利用 | 专家专长明确   | 中     |
| **分层融合** | 结构清晰 | 层次化问题     | 中     |
| **演化融合** | 质量最优 | 质量要求极高   | 高     |
