## 概述

V4.3 版本引入了两个核心功能来优化人机交互体验：

### 1. 状态管理 (State Management)

**目的**: 降低上下文消耗，支持断点续传

**核心价值**:

- ✅ 降低 **60-70%** 上下文消耗
- ✅ 支持会话中断和恢复
- ✅ 自动生成阶段摘要
- ✅ 持久化确认结果

### 2. 智能推荐 (Recommendation Engine)

**目的**: 减少用户决策负担，提高确认效率

**核心价值**:

- ✅ 自动推荐最合适的选项
- ✅ 提供可解释的推荐理由
- ✅ 高/中/低置信度标注
- ✅ 支持自定义推荐规则

## 核心功能

### 状态管理核心功能

```python
from tools.interaction.state_manager import StateManager, PhaseType

manager = StateManager()

# 1. 创建会话
session = manager.create_session("项目名称")

# 2. 开始阶段
phase = manager.start_phase(session, PhaseType.PHASE_1_5_FRAMEWORK_CONFIRMATION, "框架确认")

# 3. 记录交互
manager.add_interaction(session, PhaseType.PHASE_1_5_FRAMEWORK_CONFIRMATION, "决策变量", "1", {...})

manager.update_confirmed_item(session, PhaseType.PHASE_1_5_FRAMEWORK_CONFIRMATION, "variables", {...})

# 5. 完成阶段 (自动生成摘要)
manager.complete_phase(session, PhaseType.PHASE_1_5_FRAMEWORK_CONFIRMATION, {...})

# 6. 获取压缩上下文
compressed = manager.get_compressed_context(session)
```

### 5分钟上手示例

```python
from tools.interaction.state_manager import StateManager, PhaseType
from tools.interaction.recommendation_engine import RecommendationEngine

# 初始化
state_manager = StateManager()
rec_engine = RecommendationEngine()

# 创建会话
session = state_manager.create_session("生鲜配送调度")
print(f"会话ID: {session.session_id}")

# 开始Phase 1.5
phase = state_manager.start_phase(
    session,
    PhaseType.PHASE_1_5_FRAMEWORK_CONFIRMATION,
    "框架确认"
)

# 场景分类确认 (带推荐)
features = {
    "keywords": ["生鲜", "配送", "车辆"],
    "has_vehicle_capacity": True
}

options = [
    "✅ 确认 - VRP问题",
    "📝 修改场景",
    "💬 其他"
]

# 获取推荐
recommendation = rec_engine.get_recommendation(
    "scenario_classification", features, options
)

# 显示推荐
if recommendation:
    print(f"推荐选项 {recommendation.option_number}: {options[recommendation.option_index]}")
    print(f"理由: {recommendation.reason}")

    # 标注选项
    annotated = rec_engine.annotate_options(options, recommendation)
    for i, opt in enumerate(annotated, 1):
        print(f"{i}. {opt}")

# 用户选择
user_choice = input("请选择 (1-3): ")

# 保存确认结果
state_manager.update_confirmed_item(
    session,
    PhaseType.PHASE_1_5_FRAMEWORK_CONFIRMATION,
    "scenario_classification",
    {"type": "VRP", "confirmed": True}
)

# 记录交互
state_manager.add_interaction(
    session,
    PhaseType.PHASE_1_5_FRAMEWORK_CONFIRMATION,
    "场景分类",
    user_choice,
    {"action": "confirm"}
)

# 完成阶段
state_manager.complete_phase(
    session,
    PhaseType.PHASE_1_5_FRAMEWORK_CONFIRMATION,
    {"scenario_classification": {"type": "VRP"}}
)

print(f"\n✅ Phase 1.5 完成")
print(f"摘要: {phase.summary}")
```

```python
from tools.interaction.state_manager import StateManager, PhaseType
from tools.interaction.recommendation_engine import RecommendationEngine
import json

# 初始化
state_manager = StateManager()
rec_engine = RecommendationEngine()

session = state_manager.create_session("生鲜配送调度", mode="incremental")

phase = state_manager.start_phase(

    "📝 增加其他决策变量",
    "📝 删除某个决策变量"
]

recommendation = rec_engine.get_recommendation("decision_variables", features, options)

if recommendation:

}

)

state_manager.add_interaction(
    session,
    PhaseType.PHASE_1_5_FRAMEWORK_CONFIRMATION,

    "algorithm_category": "元启发式"
}

phase = state_manager.complete_phase(
    session,
    PhaseType.PHASE_1_5_FRAMEWORK_CONFIRMATION,

)

print("\n✅ Phase 1.5 完成")
```

```python
# 2. 及时记录交互
state_manager.add_interaction(session, phase_type, item_name, user_input, parsed_result)

# 3. 使用压缩上下文
compressed = state_manager.get_compressed_context(session)

# 4. 定期导出会话
state_manager.export_session(session, f"backup_{session.session_id}.json")
```

#### ❌ 不推荐做法

```python
# 1. 不要手动修改会话对象后忘记保存
session.phases[...] = ...  # ❌
# 应该使用: state_manager.update_confirmed_item(...)

# 2. 不要在阶段内多次调用complete_phase
# 每个阶段只应该调用一次

# 3. 不要加载完整的历史上下文
full_history = session.phases  # ❌ 消耗大量token
# 应该使用: compressed = state_manager.get_compressed_context(session)
```

### 2. 智能推荐最佳实践

#### ✅ 推荐做法

```python
# 1. 尽可能提供丰富的特征
features = {
    "scenario_type": "生鲜配送",
    "keywords": ["配送", "冷链"],
    "has_vehicle_capacity": True,
    "has_time_window": True,
    "time_criticality": "极高"
}

# 2. 总是处理recommendation为None的情况
recommendation = rec_engine.get_recommendation(...)
if recommendation:
    # 使用推荐
    ...
else:
    # 没有推荐，正常显示选项
    ...

# 3. 记录推荐是否被采纳
state_manager.add_interaction(
    session, phase_type, item_name, user_input,
    {
        "action": "confirm",
        "recommendation_followed": user_input == str(recommendation.option_number)
    }
)
```

#### ❌ 不推荐做法

```python
# 1. 不要假设一定有推荐
option = recommendation.option_number  # ❌ 可能为None
# 应该先检查: if recommendation: ...

# 2. 不要忽略推荐理由
print(f"推荐: {recommendation.option_number}")  # ❌
# 应该显示理由: print(recommendation.reason)

# 3. 不要为所有场景使用相同的特征
features = {"scenario_type": "VRP"}  # ❌ 特征太少
# 应该提供更多特征
```

```python
# ✅ 每个新阶段开始时，只加载必要的上下文
prev_summary = state_manager.get_previous_phase_summary(session, current_phase)

compressed = state_manager.get_compressed_context(
```

### Q4: 如何统计推荐采纳率？

**A**: 分析交互历史。

```python
phase = session.phases[phase_type.name]
interactions = phase.user_interactions

followed = sum(1 for i in interactions if i['parsed_result'].get('recommendation_followed'))
total = len(interactions)

print(f"推荐采纳率: {followed / total * 100:.1f}%")
```

### Q5: 如何添加自定义推荐规则？

```python
from tools.interaction.recommendation_engine import RecommendationRule, RecommendationConfidence

rule = RecommendationRule(
```

### Q6: 压缩上下文能降低多少token消耗？

**A**: 约降低 **60-70%**。

```python
# 完整上下文
full_context = json.dumps(session.phases)
full_size = len(full_context)

# 压缩上下文
compressed = state_manager.get_compressed_context(session)
compressed_size = len(json.dumps(compressed))

reduction = (1 - compressed_size / full_size) * 100
print(f"压缩率: {reduction:.1f}%")  # 通常 60-70%
```

### Q7: 如何在不同阶段之间传递信息？

```python
# Phase 1.5: 保存框架信息
```

### Q8: 推荐置信度的含义？

**A**:

- **高 (HIGH)**: 强烈推荐，置信度 > 80%
- **中 (MEDIUM)**: 建议选择，置信度 50-80%
- **低 (LOW)**: 可选，置信度 < 50%


### Q10: 如何在多进程环境中使用状态管理？

**A**: 使用文件锁避免并发写入。

```python
import fcntl

# 加载会话前加锁
lock_file = f"{state_manager.state_dir}/{session_id}.lock"

with open(lock_file, 'w') as lock:
    fcntl.flock(lock.fileno(), fcntl.LOCK_EX)

    # 安全地加载和修改会话
    session = state_manager.load_session(session_id)
    # ... 修改操作 ...
    state_manager.save_session(session)

    fcntl.flock(lock.fileno(), fcntl.LOCK_UN)
```
