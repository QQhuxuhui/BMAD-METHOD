## 📖 策略概述

**核心理念**: 通过多维度指标量化评估问题复杂度，指导专家组合和协作模式选择

**评估维度**:

- 问题规模：任务数量、约束数量、目标数量
- 问题特征：NP复杂度、动态性、不确定性
- 领域特性：领域专业性、跨域程度
- 时间约束：响应时间要求、优化深度

## 🔧 代码实现

```python
from enum import Enum
from typing import Dict, List

class ComplexityLevel(Enum):
    """复杂度等级"""
    SIMPLE = 1          # 简单
    MEDIUM = 2          # 中等
    COMPLEX = 3         # 复杂
    VERY_COMPLEX = 4    # 极复杂

class ComplexityAssessment:
    """复杂度评估策略"""

    def __init__(self):
        self.dimension_weights = {
            'scale': 0.30,        # 规模维度权重
            'characteristics': 0.35,  # 特征维度权重
            'domain': 0.20,       # 领域维度权重
            'constraints': 0.15   # 约束维度权重
        }

    def assess(self, problem: Dict) -> Dict:
        """
        评估问题复杂度

        返回:
        {
            'level': ComplexityLevel,
            'score': float,  # 0-100
            'dimensions': {...},
            'recommendations': {...}
        }
        """
        # 各维度评估
        scale_score = self._assess_scale(problem)
        char_score = self._assess_characteristics(problem)
        domain_score = self._assess_domain(problem)
        constraint_score = self._assess_constraints(problem)

        # 加权计算总分
        total_score = (
            scale_score * self.dimension_weights['scale'] +
            char_score * self.dimension_weights['characteristics'] +
            domain_score * self.dimension_weights['domain'] +
            constraint_score * self.dimension_weights['constraints']
        ) * 100

        # 确定复杂度等级
        level = self._score_to_level(total_score)

        # 生成建议
        recommendations = self._generate_recommendations(level, problem)

        return {
            'level': level,
            'score': total_score,
            'dimensions': {
                'scale': scale_score,
                'characteristics': char_score,
                'domain': domain_score,
                'constraints': constraint_score
            },
            'recommendations': recommendations
        }

    def _assess_scale(self, problem: Dict) -> float:
        """评估问题规模 (0-1)"""
        score = 0.0

        # 任务数量指标
        task_count = problem.get('task_count', 0)
        if task_count <= 10:
            score += 0.25
        elif task_count <= 50:
            score += 0.50
        elif task_count <= 200:
            score += 0.75
        else:
            score += 1.0

        # 约束数量指标
        constraint_count = len(problem.get('constraints', []))
        if constraint_count <= 3:
            score += 0.25
        elif constraint_count <= 8:
            score += 0.50
        elif constraint_count <= 15:
            score += 0.75
        else:
            score += 1.0

        # 目标数量指标
        objective_count = len(problem.get('objectives', []))
        if objective_count == 1:
            score += 0.25
        elif objective_count == 2:
            score += 0.50
        elif objective_count <= 4:
            score += 0.75
        else:
            score += 1.0

        # 归一化到0-1
        return score / 3.0

    def _assess_characteristics(self, problem: Dict) -> float:
        """评估问题特征 (0-1)"""
        score = 0.0

        # 算法复杂度
        complexity_class = problem.get('complexity_class', 'P')
        if complexity_class == 'P':
            score += 0.2
        elif complexity_class == 'NP':
            score += 0.5
        elif complexity_class in ['NP-hard', 'NP-complete']:
            score += 1.0

        # 动态性
        is_dynamic = problem.get('is_dynamic', False)
        if is_dynamic:
            score += 0.5

        # 不确定性
        uncertainty_level = problem.get('uncertainty', 'low')
        uncertainty_map = {'low': 0.2, 'medium': 0.5, 'high': 0.8, 'very_high': 1.0}
        score += uncertainty_map.get(uncertainty_level, 0.2)

        # 归一化
        return min(score / 2.5, 1.0)

    def _assess_domain(self, problem: Dict) -> float:
        """评估领域特性 (0-1)"""
        score = 0.0

        # 领域专业性
        domain_specificity = problem.get('domain_specificity', 'low')
        specificity_map = {'low': 0.2, 'medium': 0.5, 'high': 0.8, 'very_high': 1.0}
        score += specificity_map.get(domain_specificity, 0.2)

        # 跨域程度
        domains = problem.get('domains', [])
        if len(domains) == 1:
            score += 0.2
        elif len(domains) == 2:
            score += 0.5
        else:
            score += 1.0

        # 归一化
        return score / 2.0

    def _assess_constraints(self, problem: Dict) -> float:
        """评估约束条件 (0-1)"""
        score = 0.0

        # 时间约束
        time_limit = problem.get('time_limit')
        if time_limit:
            if 'minute' in str(time_limit).lower():
                score += 1.0  # 分钟级要求很紧
            elif 'hour' in str(time_limit).lower():
                score += 0.5  # 小时级中等
            else:
                score += 0.2  # 天级宽松

        # 资源约束
        resource_constraints = problem.get('resource_constraints', {})
        if resource_constraints:
            score += 0.5

        # 质量要求
        quality_req = problem.get('quality_requirement', 'medium')
        quality_map = {'low': 0.2, 'medium': 0.5, 'high': 0.8, 'critical': 1.0}
        score += quality_map.get(quality_req, 0.5)

        # 归一化
        return min(score / 2.5, 1.0)

    def _score_to_level(self, score: float) -> ComplexityLevel:
        """分数转换为复杂度等级"""
        if score < 30:
            return ComplexityLevel.SIMPLE
        elif score < 55:
            return ComplexityLevel.MEDIUM
        elif score < 75:
            return ComplexityLevel.COMPLEX
        else:
            return ComplexityLevel.VERY_COMPLEX

    def _generate_recommendations(self, level: ComplexityLevel,
                                  problem: Dict) -> Dict:
        """根据复杂度等级生成建议"""
        recommendations = {
            'expert_count': 0,
            'collaboration_mode': '',
            'time_estimate': '',
            'quality_level': '',
            'suggested_experts': []
        }

        if level == ComplexityLevel.SIMPLE:
            recommendations.update({
                'expert_count': 2,
                'collaboration_mode': 'parallel',
                'time_estimate': '15-30分钟',
                'quality_level': 'good',
                'suggested_experts': ['algorithm', 'constraint']
            })
        elif level == ComplexityLevel.MEDIUM:
            recommendations.update({
                'expert_count': 3,
                'collaboration_mode': 'hybrid',
                'time_estimate': '1-1.5小时',
                'quality_level': 'high',
                'suggested_experts': ['domain', 'algorithm', 'constraint']
            })
        elif level == ComplexityLevel.COMPLEX:
            recommendations.update({
                'expert_count': 4,
                'collaboration_mode': 'sequential',
                'time_estimate': '2-3小时',
                'quality_level': 'very_high',
                'suggested_experts': ['domain', 'constraint', 'algorithm', 'objective']
            })
        else:  # VERY_COMPLEX
            recommendations.update({
                'expert_count': 5,
                'collaboration_mode': 'sequential',
                'time_estimate': '4-6小时',
                'quality_level': 'critical',
                'suggested_experts': ['domain', 'constraint', 'objective',
                                     'algorithm', 'extension']
            })

        return recommendations
```

## 📋 使用示例

### 场景1: VRP问题复杂度评估

```python
assessment = ComplexityAssessment()

# 定义问题
vrp_problem = {
    'task_count': 150,
    'constraints': [
        'vehicle_capacity',
        'time_window',
        'driver_working_hours',
        'customer_priority',
        'vehicle_compatibility'
    ],
    'objectives': ['minimize_cost', 'minimize_time'],
    'complexity_class': 'NP-hard',
    'is_dynamic': True,
    'uncertainty': 'medium',
    'domain_specificity': 'high',
    'domains': ['logistics'],
    'time_limit': '2 hours',
    'quality_requirement': 'high'
}

# 评估复杂度
result = assessment.assess(vrp_problem)

print(f"复杂度等级: {result['level'].name}")
print(f"复杂度分数: {result['score']:.2f}/100")
print(f"维度评分:")
for dim, score in result['dimensions'].items():
    print(f"  {dim}: {score:.2f}")

print(f"\n建议:")
print(f"  专家数量: {result['recommendations']['expert_count']}")
print(f"  协作模式: {result['recommendations']['collaboration_mode']}")
print(f"  预计时间: {result['recommendations']['time_estimate']}")
print(f"  建议专家: {result['recommendations']['suggested_experts']}")
```

输出示例：

```
复杂度等级: COMPLEX
复杂度分数: 68.50/100
维度评分:
  scale: 0.67
  characteristics: 0.80
  domain: 0.60
  constraints: 0.70

建议:
  专家数量: 4
  协作模式: sequential
  预计时间: 2-3小时
  建议专家: ['domain', 'constraint', 'algorithm', 'objective']
```

## 📊 评估矩阵

| 维度       | 简单 | 中等   | 复杂    | 极复杂      |
| ---------- | ---- | ------ | ------- | ----------- |
| **规模**   |      |        |         |             |
| 任务数     | ≤10  | 11-50  | 51-200  | >200        |
| 约束数     | ≤3   | 4-8    | 9-15    | >15         |
| 目标数     | 1    | 2      | 3-4     | >4          |
| **特征**   |      |        |         |             |
| 算法复杂度 | P    | NP     | NP-hard | NP-complete |
| 动态性     | 否   | 部分   | 是      | 高度动态    |
| 不确定性   | 低   | 中     | 高      | 极高        |
| **领域**   |      |        |         |             |
| 专业性     | 低   | 中     | 高      | 极高        |
| 跨域       | 单域 | 双域   | 多域    | 跨行业      |
| **约束**   |      |        |         |             |
| 时间限制   | 天级 | 小时级 | 分钟级  | 秒级        |
| 质量要求   | 中   | 高     | 极高    | 关键        |
