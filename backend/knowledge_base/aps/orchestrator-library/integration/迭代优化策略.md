## 📖 策略概述

**核心理念**: 通过多轮迭代逐步优化方案质量，每轮针对性改进，直到满足质量标准或达到迭代上限

**优化流程**:

1. 初版方案生成
2. 质量评估诊断
3. 针对性改进
4. 验证确认
5. 循环或终止

## 🔧 代码实现

```python
from typing import Dict, List, Callable

class IterationStatus:
    """迭代状态"""
    def __init__(self):
        self.current_iteration = 0
        self.quality_history = []
        self.improvement_history = []

class IterativeOptimization:
    """迭代优化策略"""

    def __init__(self, max_iterations: int = 5, target_quality: float = 0.95):
        """
        参数:
        - max_iterations: 最大迭代次数
        - target_quality: 目标质量分数 (0-1)
        """
        self.max_iterations = max_iterations
        self.target_quality = target_quality
        self.status = IterationStatus()

    def optimize(self, initial_solution: Dict,
                 quality_assessor: Callable,
                 improvement_generator: Callable) -> Dict:
        """
        迭代优化主流程

        参数:
        - initial_solution: 初始方案
        - quality_assessor: 质量评估函数
        - improvement_generator: 改进方案生成函数

        返回:
        {
            'final_solution': {...},
            'final_quality': float,
            'iterations_performed': int,
            'optimization_history': [...]
        }
        """
        current_solution = initial_solution
        self.status = IterationStatus()

        for iteration in range(1, self.max_iterations + 1):
            self.status.current_iteration = iteration

            # Step 1: 评估当前方案质量
            quality_result = quality_assessor(current_solution)
            current_quality = quality_result['overall_quality']

            # 记录质量历史
            self.status.quality_history.append({
                'iteration': iteration,
                'quality': current_quality,
                'issues': quality_result.get('issues', [])
            })

            # 检查是否达到目标质量
            if current_quality >= self.target_quality:
                return self._build_result(
                    current_solution,
                    current_quality,
                    iteration,
                    termination='target_reached'
                )

            # 检查是否还有改进空间
            if iteration > 1:
                improvement = current_quality - self.status.quality_history[-2]['quality']
                self.status.improvement_history.append(improvement)

                # 如果连续两轮改进很小，提前终止
                if len(self.status.improvement_history) >= 2:
                    recent_improvements = self.status.improvement_history[-2:]
                    if all(imp < 0.01 for imp in recent_improvements):
                        return self._build_result(
                            current_solution,
                            current_quality,
                            iteration,
                            termination='minimal_improvement'
                        )

            # Step 2: 生成改进方案
            if iteration < self.max_iterations:
                improvement_plan = self._generate_improvement_plan(
                    current_solution,
                    quality_result
                )

                # Step 3: 应用改进
                improved_solution = improvement_generator(
                    current_solution,
                    improvement_plan
                )

                # 更新当前方案
                current_solution = improved_solution

        # 达到最大迭代次数
        final_quality = quality_assessor(current_solution)['overall_quality']
        return self._build_result(
            current_solution,
            final_quality,
            self.max_iterations,
            termination='max_iterations'
        )

    def _generate_improvement_plan(self, solution: Dict,
                                   quality_result: Dict) -> Dict:
        """生成改进计划"""
        plan = {
            'priorities': [],
            'actions': []
        }

        issues = quality_result.get('issues', [])

        # 按严重性排序问题
        sorted_issues = sorted(
            issues,
            key=lambda x: self._get_severity_score(x.get('severity', 'low')),
            reverse=True
        )

        for issue in sorted_issues:
            dimension = issue.get('dimension')
            gap = issue.get('gap', 0)

            # 生成针对性改进行动
            actions = self._get_improvement_actions(dimension, gap)

            plan['priorities'].append(dimension)
            plan['actions'].extend(actions)

        return plan

    def _get_severity_score(self, severity: str) -> int:
        """获取严重性分数"""
        severity_map = {
            'critical': 4,
            'high': 3,
            'medium': 2,
            'low': 1
        }
        return severity_map.get(severity, 1)

    def _get_improvement_actions(self, dimension: str, gap: float) -> List[Dict]:
        """获取改进行动"""
        actions = []

        if dimension == 'completeness':
            if gap > 0.10:
                actions.append({
                    'action': 'add_missing_components',
                    'priority': 'high',
                    'description': '补充缺失的核心组件'
                })
            if gap > 0.05:
                actions.append({
                    'action': 'enhance_documentation',
                    'priority': 'medium',
                    'description': '完善文档和说明'
                })

        elif dimension == 'consistency':
            actions.append({
                'action': 'resolve_conflicts',
                'priority': 'high',
                'description': '解决专家建议间的冲突'
            })
            actions.append({
                'action': 'align_interfaces',
                'priority': 'medium',
                'description': '对齐组件接口'
            })

        elif dimension == 'correctness':
            actions.append({
                'action': 'add_validation',
                'priority': 'high',
                'description': '增加验证和测试'
            })
            actions.append({
                'action': 'review_logic',
                'priority': 'medium',
                'description': '审查核心逻辑'
            })

        elif dimension == 'performance':
            actions.append({
                'action': 'optimize_algorithm',
                'priority': 'high',
                'description': '优化算法性能'
            })
            if gap > 0.10:
                actions.append({
                    'action': 'consider_approximation',
                    'priority': 'medium',
                    'description': '考虑近似算法'
                })

        return actions

    def _build_result(self, solution: Dict, quality: float,
                     iterations: int, termination: str) -> Dict:
        """构建结果"""
        return {
            'final_solution': solution,
            'final_quality': quality,
            'iterations_performed': iterations,
            'termination_reason': termination,
            'optimization_history': self.status.quality_history,
            'improvement_trajectory': self.status.improvement_history,
            'total_improvement': (
                quality - self.status.quality_history[0]['quality']
                if self.status.quality_history else 0
            )
        }

    def get_optimization_report(self) -> Dict:
        """获取优化报告"""
        if not self.status.quality_history:
            return {'status': 'no_optimization_performed'}

        initial_quality = self.status.quality_history[0]['quality']
        final_quality = self.status.quality_history[-1]['quality']
        total_improvement = final_quality - initial_quality

        return {
            'initial_quality': initial_quality,
            'final_quality': final_quality,
            'total_improvement': total_improvement,
            'improvement_rate': total_improvement / initial_quality if initial_quality > 0 else 0,
            'iterations': len(self.status.quality_history),
            'average_improvement_per_iteration': (
                total_improvement / len(self.status.quality_history)
                if self.status.quality_history else 0
            ),
            'quality_trajectory': [
                {'iteration': h['iteration'], 'quality': h['quality']}
                for h in self.status.quality_history
            ]
        }
```

## 📋 使用示例

### 场景1: 迭代优化VRP方案

```python
# 定义质量评估函数
def assess_quality(solution):
    # 简化示例：基于组件完整性评分
    required = ['algorithm', 'constraints', 'objectives', 'domain']
    present = sum(1 for r in required if r in solution)
    quality = present / len(required)

    issues = []
    if quality < 1.0:
        issues.append({
            'dimension': 'completeness',
            'gap': 1.0 - quality,
            'severity': 'high'
        })

    return {
        'overall_quality': quality,
        'issues': issues
    }

# 定义改进生成函数
def generate_improvement(solution, plan):
    improved = solution.copy()

    for action in plan['actions']:
        if action['action'] == 'add_missing_components':
            # 补充缺失组件
            if 'algorithm' not in improved:
                improved['algorithm'] = 'greedy'
            if 'constraints' not in improved:
                improved['constraints'] = []
            if 'objectives' not in improved:
                improved['objectives'] = ['minimize_cost']

    return improved

# 执行迭代优化
optimizer = IterativeOptimization(max_iterations=5, target_quality=0.95)

initial_solution = {
    'domain': 'VRP'
}

result = optimizer.optimize(
    initial_solution,
    assess_quality,
    generate_improvement
)

print(f"优化结果:")
print(f"  最终质量: {result['final_quality']:.2%}")
print(f"  执行轮次: {result['iterations_performed']}")
print(f"  终止原因: {result['termination_reason']}")
print(f"  总体改进: {result['total_improvement']:.2%}")

print(f"\n优化历史:")
for h in result['optimization_history']:
    print(f"  第{h['iteration']}轮: 质量 {h['quality']:.2%}")
```

### 场景2: 获取优化报告

# 执行优化后
report = optimizer.get_optimization_report()

print(f"  最终质量: {report['final_quality']:.2%}")

## 📊 迭代优化模式

| 模式         | 终止条件      | 适用场景       |
| ------------ | ------------- | -------------- |
| **目标导向** | 达到目标质量  | 质量标准明确   |
| **最大迭代** | 达到迭代上限  | 时间有限       |
| **最小改进** | 改进幅度<阈值 | 效率优先       |
| **混合模式** | 多条件组合    | 平衡质量和效率 |

## 💡 优化技巧

### 1. 自适应迭代次数

```python
# 根据初始质量调整迭代次数
if initial_quality < 0.6:
    max_iterations = 8  # 质量很低，需要更多轮
elif initial_quality < 0.8:
    max_iterations = 5  # 中等质量
else:
    max_iterations = 3  # 质量较高，微调即可
```

### 2. 动态目标调整

```python
# 根据改进速度调整目标
if average_improvement > 0.05:
    target_quality = min(0.98, target_quality + 0.02)  # 快速改进，提高目标
```

### 3. 早停机制

```python
# 质量足够好且改进停滞，提前终止
if quality > 0.90 and recent_improvement < 0.01:
    break
```
