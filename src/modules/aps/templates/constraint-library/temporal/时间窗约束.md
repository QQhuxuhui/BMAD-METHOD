## 📖 约束概述

### 定义

**时间窗约束**（Time Window Constraint）是调度问题中最常见的时间类约束，规定某项任务或服务必须在指定的时间范围内开始或完成。

**数学定义**：

```
对于任务 i，定义时间窗 [e_i, l_i]
其中：
  e_i = 最早开始时间 (Earliest Time)
  l_i = 最晚开始时间 (Latest Time)

约束条件：
  e_i ≤ t_i ≤ l_i

其中 t_i 是任务 i 的实际开始时间
```

### 约束类型

1. **硬时间窗** (Hard Time Window)
   - 绝对不能违反
   - 违反即方案不可行
   - 示例: 客户明确要求的送达时间

2. **软时间窗** (Soft Time Window)
   - 可以违反但有惩罚
   - 违反程度越大惩罚越高
   - 示例: 客户偏好的送达时间

3. **多重时间窗** (Multiple Time Windows)
   - 多个可选的时间窗
   - 满足任一时间窗即可
   - 示例: 客户提供的多个可选时间段

## 💡 核心理论

### 时间窗传播

**前向传播**：

```python
def forward_propagation(route):
    """
    前向传播：计算最早到达时间
    """
    current_time = start_time
    for i in range(len(route)):
        # 到达时间 = 出发时间 + 行驶时间
        arrival_time = current_time + travel_time(route[i-1], route[i])

        # 如果早于时间窗开始，需要等待
        start_time = max(arrival_time, route[i].earliest_time)

        # 服务完成时间
        current_time = start_time + route[i].service_time

    return current_time
```

**后向传播**：

```python
def backward_propagation(route):
    """
    后向传播：计算最晚出发时间
    """
    latest_times = {}
    current_time = end_time

    for i in range(len(route)-1, -1, -1):
        # 最晚出发时间 = 下一节点最晚到达时间 - 行驶时间 - 服务时间
        latest_departure = current_time - travel_time(route[i], route[i+1])
        latest_start = latest_departure - route[i].service_time

        # 不能晚于客户的时间窗上限
        latest_times[i] = min(latest_start, route[i].latest_time)

        current_time = latest_times[i]

    return latest_times
```

### 时间窗可行性

**可达性判断**：

```
节点 i 可到达节点 j 的条件：
  e_i + s_i + t_ij ≤ l_j

其中：
  e_i: 节点 i 的最早开始时间
  s_i: 节点 i 的服务时间
  t_ij: 从 i 到 j 的行驶时间
  l_j: 节点 j 的最晚开始时间
```

## 🔧 代码实现

### 约束处理器类

```python
from typing import Dict, List, Tuple
from dataclasses import dataclass
from datetime import datetime, timedelta

@dataclass
class TimeWindow:
    """时间窗数据结构"""
    earliest: datetime  # 最早时间
    latest: datetime    # 最晚时间
    is_hard: bool = True  # 是否硬约束

    def width(self) -> timedelta:
        """时间窗宽度"""
        return self.latest - self.earliest

    def contains(self, time: datetime) -> bool:
        """判断时间是否在窗口内"""
        return self.earliest <= time <= self.latest

    def violation(self, time: datetime) -> timedelta:
        """计算违反程度"""
        if time < self.earliest:
            return self.earliest - time
        elif time > self.latest:
            return time - self.latest
        return timedelta(0)

class TimeWindowConstraintHandler:
    """
    时间窗约束处理器

    功能:
      - 验证方案是否满足时间窗约束
      - 检测时间窗违反情况
      - 计算违反惩罚
      - 修复时间窗违反
    """

    def __init__(self, time_windows: Dict[str, TimeWindow],
                 travel_times: Dict[Tuple[str, str], timedelta],
                 service_times: Dict[str, timedelta]):
        """
        参数:
          time_windows: 每个节点的时间窗 {node_id: TimeWindow}
          travel_times: 节点间行驶时间 {(from, to): timedelta}
          service_times: 每个节点的服务时间 {node_id: timedelta}
        """
        self.time_windows = time_windows
        self.travel_times = travel_times
        self.service_times = service_times

    def validate(self, route: List[str], start_time: datetime) -> Tuple[bool, List[Dict]]:
        """
        验证路线是否满足时间窗约束

        参数:
          route: 路线节点列表
          start_time: 路线开始时间

        返回:
          (is_valid, violations)
          is_valid: 是否全部满足
          violations: 违反详情列表
        """
        violations = []
        current_time = start_time

        for i, node in enumerate(route):
            # 计算到达时间
            if i > 0:
                travel_time = self.travel_times.get((route[i-1], node), timedelta(0))
                current_time += travel_time

            # 获取时间窗
            time_window = self.time_windows.get(node)
            if not time_window:
                continue

            # 检查是否在时间窗内
            if current_time < time_window.earliest:
                # 提前到达，需要等待
                wait_time = time_window.earliest - current_time
                current_time = time_window.earliest

                if not time_window.is_hard:
                    # 软约束，记录等待时间
                    violations.append({
                        'type': 'early_arrival',
                        'node': node,
                        'actual_time': current_time - wait_time,
                        'earliest_time': time_window.earliest,
                        'wait_time': wait_time,
                        'penalty': wait_time.total_seconds() / 3600  # 小时
                    })

            elif current_time > time_window.latest:
                # 迟到
                delay = current_time - time_window.latest

                violations.append({
                    'type': 'late_arrival',
                    'node': node,
                    'actual_time': current_time,
                    'latest_time': time_window.latest,
                    'delay': delay,
                    'penalty': delay.total_seconds() / 3600 * 10,  # 迟到惩罚更高
                    'is_hard': time_window.is_hard
                })

                # 硬时间窗违反直接返回
                if time_window.is_hard:
                    return False, violations

            # 服务时间
            current_time += self.service_times.get(node, timedelta(0))

        # 只有硬时间窗全满足才算有效
        is_valid = all(not v.get('is_hard', False) for v in violations)
        return is_valid, violations

    def get_penalty(self, violations: List[Dict]) -> float:
        """
        计算时间窗违反的总惩罚

        惩罚策略:
          - 提前到达: 等待时间（小时数）
          - 迟到: 延迟时间（小时数）× 10
          - 硬约束违反: 返回无穷大
        """
        total_penalty = 0.0

        for v in violations:
            if v.get('is_hard', False):
                return float('inf')  # 硬约束违反，无穷大惩罚

            total_penalty += v.get('penalty', 0)

        return total_penalty

    def repair(self, route: List[str], violations: List[Dict],
               start_time: datetime) -> Tuple[List[str], datetime]:
        """
        修复时间窗违反

        策略:
          1. 调整出发时间（如果可能）
          2. 重新排序节点（优先级高的先访问）
          3. 移除无法满足的节点（软约束）

        返回:
          (repaired_route, adjusted_start_time)
        """
        # 策略1: 调整出发时间
        # 找到第一个违反节点，尝试提前出发
        for v in violations:
            if v['type'] == 'late_arrival' and v.get('is_hard'):
                delay = v['delay']
                adjusted_start = start_time - delay

                # 验证调整后是否可行
                is_valid, new_violations = self.validate(route, adjusted_start)
                if is_valid:
                    return route, adjusted_start

        # 策略2: 重新排序（按时间窗紧急程度）
        if len(route) > 2:  # 至少有起点和终点
            middle_nodes = route[1:-1]  # 不包括起点和终点

            # 按时间窗紧迫性排序（时间窗早的优先）
            sorted_nodes = sorted(
                middle_nodes,
                key=lambda n: self.time_windows[n].earliest
            )

            reordered_route = [route[0]] + sorted_nodes + [route[-1]]

            # 验证重新排序后的路线
            is_valid, new_violations = self.validate(reordered_route, start_time)
            if is_valid or len(new_violations) < len(violations):
                return reordered_route, start_time

        # 策略3: 移除软约束违反节点
        hard_violation_nodes = {
            v['node'] for v in violations
            if v.get('is_hard', False)
        }

        if hard_violation_nodes:
            repaired_route = [n for n in route if n not in hard_violation_nodes]
            return repaired_route, start_time

        # 无法修复，返回原路线
        return route, start_time

    def check_feasibility(self, node1: str, node2: str) -> bool:
        """
        检查从 node1 到 node2 的时间窗可行性

        判断条件:
          e1 + s1 + t12 ≤ l2
        """
        tw1 = self.time_windows.get(node1)
        tw2 = self.time_windows.get(node2)

        if not tw1 or not tw2:
            return True

        travel_time = self.travel_times.get((node1, node2), timedelta(0))
        service_time = self.service_times.get(node1, timedelta(0))

        earliest_arrival = tw1.earliest + service_time + travel_time

        return earliest_arrival <= tw2.latest
```

## 📊 应用场景

### 场景1: 配送调度

**问题描述**：

```yaml
客户A: 要求 9:00-12:00 送达 (硬时间窗)
客户B: 偏好 10:00-11:00 送达 (软时间窗)
客户C: 要求 14:00-16:00 送达 (硬时间窗)

约束:
  - 硬时间窗必须满足
  - 软时间窗尽量满足，违反有惩罚
```

**解决方案**：

```python
# 定义时间窗
time_windows = {
    'A': TimeWindow(
        earliest=datetime(2025, 1, 1, 9, 0),
        latest=datetime(2025, 1, 1, 12, 0),
        is_hard=True
    ),
    'B': TimeWindow(
        earliest=datetime(2025, 1, 1, 10, 0),
        latest=datetime(2025, 1, 1, 11, 0),
        is_hard=False
    ),
    'C': TimeWindow(
        earliest=datetime(2025, 1, 1, 14, 0),
        latest=datetime(2025, 1, 1, 16, 0),
        is_hard=True
    )
}

# 创建约束处理器
handler = TimeWindowConstraintHandler(
    time_windows=time_windows,
    travel_times={('A', 'B'): timedelta(minutes=30), ...},
    service_times={'A': timedelta(minutes=15), ...}
)

# 验证路线
route = ['depot', 'A', 'B', 'C', 'depot']
start_time = datetime(2025, 1, 1, 8, 30)

is_valid, violations = handler.validate(route, start_time)

if not is_valid:
    # 修复违反
    repaired_route, adjusted_start = handler.repair(route, violations, start_time)
```

### 场景2: 生产调度

**问题描述**：

```yaml
任务T1: 必须在 8:00-10:00 开始
任务T2: 必须在 T1 完成后1小时内开始
任务T3: 必须在 14:00 前完成

约束:
  - T1 时间窗硬约束
  - T2 依赖 T1
  - T3 截止期硬约束
```

### 2. 时间窗预处理

```python
def preprocess_time_windows(self):
    """
    预处理时间窗，计算可达性矩阵
    加速后续的可行性检查
    """
    n = len(self.time_windows)
    self.feasibility_matrix = [[False] * n for _ in range(n)]

    for i in range(n):
        for j in range(n):
            self.feasibility_matrix[i][j] = self.check_feasibility(i, j)
```

### 3. 并行验证

```python
from concurrent.futures import ThreadPoolExecutor

def parallel_validate(self, routes: List[List[str]]) -> List[Tuple[bool, List[Dict]]]:
    """
    并行验证多个路线
    适用于种群搜索算法
    """
    with ThreadPoolExecutor(max_workers=8) as executor:
        results = list(executor.map(
            lambda r: self.validate(r, self.start_time),
            routes
        ))
    return results
```

## 🧪 测试用例

```python
import unittest
from datetime import datetime, timedelta

class TestTimeWindowConstraint(unittest.TestCase):
    def setUp(self):
        self.time_windows = {
            'A': TimeWindow(
                earliest=datetime(2025, 1, 1, 9, 0),
                latest=datetime(2025, 1, 1, 12, 0),
                is_hard=True
            ),
            'B': TimeWindow(
                earliest=datetime(2025, 1, 1, 10, 0),
                latest=datetime(2025, 1, 1, 11, 0),
                is_hard=False
            )
        }

        self.handler = TimeWindowConstraintHandler(
            time_windows=self.time_windows,
            travel_times={('A', 'B'): timedelta(minutes=30)},
            service_times={'A': timedelta(minutes=15), 'B': timedelta(minutes=10)}
        )

    def test_valid_route(self):
        """测试满足时间窗的路线"""
        route = ['A', 'B']
        start_time = datetime(2025, 1, 1, 9, 0)

        is_valid, violations = self.handler.validate(route, start_time)

        self.assertTrue(is_valid)
        self.assertEqual(len(violations), 0)

    def test_late_arrival(self):
        """测试迟到情况"""
        route = ['A', 'B']
        start_time = datetime(2025, 1, 1, 11, 30)  # 太晚出发

        is_valid, violations = self.handler.validate(route, start_time)

        self.assertFalse(is_valid)
        self.assertGreater(len(violations), 0)
        self.assertEqual(violations[0]['type'], 'late_arrival')

    def test_early_arrival_soft(self):
        """测试提前到达软约束"""
        route = ['B']
        start_time = datetime(2025, 1, 1, 9, 0)  # 提前到达

        is_valid, violations = self.handler.validate(route, start_time)

        # 软约束，仍然有效，但有等待时间
        self.assertTrue(is_valid)
        self.assertEqual(len(violations), 1)
        self.assertEqual(violations[0]['type'], 'early_arrival')

    def test_repair(self):
        """测试修复功能"""
        route = ['A', 'B']
        start_time = datetime(2025, 1, 1, 11, 30)

        _, violations = self.handler.validate(route, start_time)
        repaired_route, adjusted_start = self.handler.repair(route, violations, start_time)

        # 验证修复后的路线
        is_valid, _ = self.handler.validate(repaired_route, adjusted_start)
        self.assertTrue(is_valid)
```
