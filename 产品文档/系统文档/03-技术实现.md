**偏离检测算法**:
```python
def check_deviation(current_analysis, baseline_contract):
    # 1. 提取当前分析的关键要素
    current_elements = extract_elements(current_analysis)

    # 2. 提取基线的关键要素
    baseline_elements = baseline_contract['key_elements']

    # 3. 计算相似度
    similarity = cosine_similarity(
        vectorize(current_elements),
        vectorize(baseline_elements)
    )

    # 4. 判断偏离
    deviation_score = 1 - similarity

    if deviation_score > 0.40:  # 相似度 < 0.60
        return {
            'deviation_score': deviation_score,
            'deviation_alert': True,
            'recommendation': 'trigger_replan'
        }

    return {
        'deviation_score': deviation_score,
        'deviation_alert': False
    }
```

```python
def finalize_model(confirmed_model):
    # 1. 规范化格式
    model = normalize_format(confirmed_model)

    # 2. 生成模型哈希（用于一致性检查）
    model_hash = hashlib.sha256(
        json.dumps(model, sort_keys=True).encode()
    ).hexdigest()

    # 3. 添加元数据
    model_baseline = {
        'model': model,
        'hash': model_hash,
        'finalized_at': datetime.now().isoformat(),
        'mode': current_mode  # "mode_a" | "mode_b"
    }

    # 4. 保存到文件
    save_file(
        f"aps-outputs/models/ten_element_model_{timestamp}.yaml",
        model
    )

    return model, model_baseline
```

**并行执行机制** (模式A):
```python
async def parallel_exec(agents, inputs):
    # 1. 并发调用所有专家
    tasks = [
        agent.analyze(inputs['ten_element_model'])
        for agent in agents
    ]

    # 2. 等待所有结果
    results = await asyncio.gather(*tasks)

    # 3. 返回聚合结果
    return {
        'domain_analysis': results[0],
        'constraint_analysis': results[1],
        'objective_analysis': results[2],
        'algorithm_recommendations': results[3]
    }
```

**串行执行机制** (模式B):
```python
def sequential_exec(agent, inputs, confirmation_config):
    # 1. 调用专家分析
    analysis = agent.analyze(inputs)

    # 2. 触发用户确认 (P0)
    confirmed_analysis = human_confirm(
        analysis,
        template=confirmation_config['template']
    )

    # 3. 返回确认后的结果
    return confirmed_analysis
```

```python
def verify_code_solution_consistency(code, solution, document_path):
    inconsistencies = []

    # 1. 验证约束实现
    solution_constraints = solution['constraint_strategy']['constraints']
    for constraint in solution_constraints:
        if not check_constraint_implemented(code, constraint):
            inconsistencies.append({
                'type': 'missing_constraint',
                'constraint': constraint['name'],
                'expected_in': 'constraints.py'
            })

    # 2. 验证目标函数实现
    solution_objectives = solution['objective_strategy']['objectives']
    for objective in solution_objectives:
        if not check_objective_implemented(code, objective):
            inconsistencies.append({
                'type': 'missing_objective',
                'objective': objective['name'],
                'expected_in': 'objectives.py'
            })

    # 3. 验证算法实现
    solution_algorithm = solution['algorithm_configuration']['algorithm']
    if not check_algorithm_implemented(code, solution_algorithm):
        inconsistencies.append({
            'type': 'missing_algorithm',
            'algorithm': solution_algorithm['name']
        })

    # 4. 验证引用标注
    for file_name, file_code in code.items():
        if not check_citations_present(file_code):
            inconsistencies.append({
                'type': 'missing_citation',
                'file': file_name
            })

    # 5. 返回结果
    if len(inconsistencies) == 0:
        return {
            'overall_status': 'PASS',
            'message': '代码与方案完全一致'
        }
    else:
        return {
            'overall_status': 'FAIL',
            'inconsistencies_found': inconsistencies
        }
```

```python
    file_name = f"{phase_id}_state_{timestamp}.yaml"
    file_path = os.path.join(state_folder, file_name)
    latest_path = os.path.join(state_folder, f"{phase_id}_state_latest.yaml")

    # 4. 保存到文件
    with open(file_path, 'w') as f:
        yaml.dump(state, f, allow_unicode=True)

    # 5. 创建/更新 latest 链接
    if os.path.exists(latest_path):
        os.remove(latest_path)
    shutil.copy2(file_path, latest_path)

    # 6. 立即验证
    verification_result = verify_state_file(file_path, phase_id)

    if not verification_result['all_checks_passed']:
        raise StateVerificationError(
            f"State verification failed: {verification_result['errors']}"
        )

    return {
        'saved_state_file': file_path,
        'verification_result': verification_result,
        'file_metadata': {
            'size': os.path.getsize(file_path),
            'timestamp': timestamp
        }
```

```python
def load_phase_state(phase_id, config, use_latest=True):
    state_folder = config['state_folder']

    # 1. 确定文件路径
    if use_latest:
        file_path = os.path.join(state_folder, f"{phase_id}_state_latest.yaml")
    else:
        # 查找最新的时间戳文件
        pattern = os.path.join(state_folder, f"{phase_id}_state_*.yaml")
        files = glob.glob(pattern)
        if not files:
            raise StateFileNotFoundError(f"No state file found for {phase_id}")
        file_path = max(files, key=os.path.getctime)

    # 2. 检查文件存在
    if not os.path.exists(file_path):
        raise StateFileNotFoundError(
            f"State file not found: {file_path}\n"
            f"Please re-execute from {phase_id}"
        )

    # 3. 加载前验证
    if config['load_policy']['verify_before_load']:
        verification = verify_state_file(file_path, phase_id)
        if not verification['valid']:
            raise StateVerificationError(
                f"State file corrupted: {file_path}"
            )

    # 4. 读取文件
    with open(file_path, 'r') as f:
        state = yaml.safe_load(f)

    # 5. 返回状态和元数据
    return {
        'state_data': state['state_data'],
        'metadata': state['metadata'],
        'load_success': True,
        'file_metadata': {
            'path': file_path,
            'size': os.path.getsize(file_path),
            'modified': os.path.getmtime(file_path)
        }
    }
```

### 状态验证实现

```python
def verify_state_file(file_path, expected_phase_id):
    checks = []
    errors = []

    try:
        # 1. 文件存在性
        if not os.path.exists(file_path):
            errors.append("File not found")
            return {'valid': False, 'errors': errors}
        checks.append('file_exists')

        # 2. 读取并解析
        with open(file_path, 'r') as f:
            state = yaml.safe_load(f)
        checks.append('yaml_parsable')

        # 3. 基本字段
        required_fields = ['phase_id', 'timestamp', 'version', 'state_data', 'metadata']
        for field in required_fields:
            if field not in state:
                errors.append(f"Missing required field: {field}")
            else:
                checks.append(f'has_{field}')

        # 4. Phase ID匹配
        if state.get('phase_id') != expected_phase_id:
            errors.append(
                f"Phase ID mismatch: expected {expected_phase_id}, "
                f"got {state.get('phase_id')}"
            )
        else:
            checks.append('phase_id_match')

        # 5. 版本兼容性
        if state.get('version') != '4.3':
            errors.append(f"Version mismatch: {state.get('version')}")
        else:
            checks.append('version_compatible')

        # 6. Phase特定字段
        required_state_fields = get_required_fields_for_phase(expected_phase_id)
        for field in required_state_fields:
            if field not in state['state_data']:
                errors.append(f"Missing state field: {field}")
            else:
                checks.append(f'has_state_{field}')

        # 7. 总体结果
        all_checks_passed = len(errors) == 0

        return {
            'valid': all_checks_passed,
            'all_checks_passed': all_checks_passed,
            'checks': checks,
            'errors': errors
        }

    except Exception as e:
        return {
            'valid': False,
            'all_checks_passed': False,
            'errors': [str(e)]
        }
```

### 执行器架构

```python
class WorkflowExecutor:
    def __init__(self, workflow_config, state_manager):
        self.config = workflow_config
        self.state_manager = state_manager
        self.context = {}  # 执行上下文

    async def execute(self):
        """执行完整工作流"""
        for phase in self.config['phases']:
            await self.execute_phase(phase)

    async def execute_phase(self, phase):
        """执行单个Phase"""
        print(f"Starting {phase['phase_name']}...")

        for step in phase['steps']:
            await self.execute_step(step, phase['phase_id'])

        print(f"Completed {phase['phase_name']}")

    async def execute_step(self, step, phase_id):
        """执行单个Step"""
        # 1. 前置条件检查
        if 'pre_condition_check' in step:
            self.check_preconditions(step['pre_condition_check'])

        # 2. 准备输入
        inputs = self.prepare_inputs(step.get('inputs', []))

        # 3. 根据action类型执行
        if step['action'] == 'exec':
            outputs = await self.execute_task(step['target'], inputs)

        elif step['action'] == 'human_confirmation':
            outputs = await self.human_confirm(step, inputs)

        elif step['action'] == 'conditional_flow':
            outputs = await self.conditional_flow(step, inputs)

        elif step['action'] == 'parallel_exec':
            outputs = await self.parallel_exec(step, inputs)

        # 4. 保存输出到上下文
        for key, value in outputs.items():
            self.context[key] = value

        # 5. 后置验证
        if 'post_action_verify' in step:
            self.verify_post_conditions(step['post_action_verify'])

    async def execute_task(self, task_path, inputs):
        """执行任务文件"""
        task = load_task(task_path)
        return await task.execute(inputs)

    async def human_confirm(self, step, inputs):
        """人机交互确认"""
        template = load_template(step['template'])
        formatted = template.format(**inputs)

        print(formatted)
        response = input("Your confirmation (Y/N): ")

        return {
            'confirmed': response.upper() == 'Y',
            'user_input': response
        }

    async def conditional_flow(self, step, inputs):
        """条件分支执行"""
        mode = self.context.get('selected_mode', 'mode_a')

        if mode == 'mode_a':
            branch_steps = step['conditional_flow']['mode_a']['steps']
        else:
            branch_steps = step['conditional_flow']['mode_b']['steps']

        outputs = {}
        for sub_step in branch_steps:
            result = await self.execute_step(sub_step, None)
            outputs.update(result)

        return outputs

    async def parallel_exec(self, step, inputs):
        """并行执行多个智能体"""
        agents = [load_agent(a) for a in step['agents']]

        tasks = [agent.analyze(inputs) for agent in agents]
        results = await asyncio.gather(*tasks)

        return dict(zip(step['outputs'], results))
```
