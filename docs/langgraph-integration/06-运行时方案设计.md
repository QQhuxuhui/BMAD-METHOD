# BMAD-APS LangGraph 运行时方案

> 基于需求重新设计的轻量级方案
> 日期: 2025-10-31
> 版本: 2.0

---

## 核心理念

**不编译，而是运行时动态加载**

```
BMAD Agent 定义 (Markdown)
    ↓ 运行时读取
LangGraph 图 (内存中)
    ↓
FastAPI 服务
    ↓
Web 浏览器
```

---

## 架构对比

### 文档原方案 vs 运行时方案

| 维度          | 编译器方案          | 运行时方案  |
| ------------- | ------------------- | ----------- |
| **开发周期**  | 8-12周              | 2-3周       |
| **项目数**    | 2个仓库             | 1个仓库     |
| **代码生成**  | ✅ 需要编译器       | ❌ 不需要   |
| **同步机制**  | ✅ 复杂脚本         | ❌ 不需要   |
| **调试周期**  | 修改→编译→同步→测试 | 修改→重启   |
| **热重载**    | ❌ 不支持           | ✅ 支持     |
| **Doc一致性** | ⚠️ 可能不一致       | ✅ 完全一致 |
| **维护成本**  | 高                  | 低          |

## 核心实现

### 1. AgentLoader (agent_loader.py)

```python
"""
智能体加载器
运行时读取 agents/*.md，构建 LangGraph 节点
"""
from pathlib import Path
from typing import Dict, Callable
import re
from langchain_anthropic import ChatAnthropic
from langchain_core.messages import SystemMessage

class AgentLoader:
    """从 Markdown 文档动态加载智能体"""

    def __init__(self, agents_dir: Path):
        self.agents_dir = agents_dir
        self._cache = {}  # 可选：缓存已解析的智能体

    def load_agent(self, agent_id: str) -> Callable:
        """
        加载单个智能体，返回节点函数

        Args:
            agent_id: 智能体ID（如 "orchestrator"）

        Returns:
            LangGraph 节点函数
        """
        # 检查缓存
        if agent_id in self._cache:
            return self._cache[agent_id]

        # 读取 Markdown 文件
        agent_file = self.agents_dir / f"{agent_id}.md"
        if not agent_file.exists():
            raise FileNotFoundError(f"Agent file not found: {agent_file}")

        with open(agent_file, 'r', encoding='utf-8') as f:
            content = f.read()

        # 解析智能体定义
        agent_def = self._parse_agent(content)

        # 构建节点函数
        node_fn = self._build_node_function(agent_id, agent_def)

        # 缓存
        self._cache[agent_id] = node_fn

        return node_fn

    def _parse_agent(self, content: str) -> Dict:
        """解析智能体 Markdown 文档"""
        return {
            "name": self._extract_name(content),
            "persona": self._extract_persona(content),
            "libraries": self._extract_libraries(content),
            "system_prompt": self._build_system_prompt(content)
        }

    def _extract_name(self, content: str) -> str:
        """提取智能体名称"""
        match = re.search(r'<agent[^>]*name="([^"]+)"', content)
        if match:
            return match.group(1)
        return "Unknown Agent"

    def _extract_persona(self, content: str) -> str:
        """提取 Persona"""
        match = re.search(
            r'##\s*(?:Persona|角色).*?\n(.*?)(?=\n##|\Z)',
            content,
            re.DOTALL | re.IGNORECASE
        )
        if match:
            return match.group(1).strip()
        return ""

    def _extract_libraries(self, content: str) -> list:
        """提取知识库引用"""
        libraries = []
        match = re.search(
            r'expert_libraries:\s*\n((?:\s+-\s+.+\n)+)',
            content
        )
        if match:
            lib_text = match.group(1)
            libraries = re.findall(r'-\s+(.+)', lib_text)
        return [lib.strip() for lib in libraries]

    def _build_system_prompt(self, content: str) -> str:
        """构建系统提示"""
        name = self._extract_name(content)
        persona = self._extract_persona(content)

        # 提取职责
        responsibilities = self._extract_section(content, "职责|Responsibilities")

        # 提取约束
        constraints = self._extract_section(content, "约束|Constraints")

        prompt = f"""你是 {name}。

{persona}

## 核心职责
{responsibilities}

## 工作约束
{constraints}
"""
        return prompt.strip()

    def _extract_section(self, content: str, section_name: str) -> str:
        """提取指定章节"""
        pattern = rf'##\s*(?:{section_name}).*?\n(.*?)(?=\n##|\Z)'
        match = re.search(pattern, content, re.DOTALL | re.IGNORECASE)
        if match:
            return match.group(1).strip()
        return ""

    def _build_node_function(self, agent_id: str, agent_def: Dict) -> Callable:
        """构建 LangGraph 节点函数"""

        # 初始化 LLM
        llm = ChatAnthropic(
            model="claude-3-5-sonnet-20241022",
            temperature=0.7
        )

        # 系统提示
        system_prompt = agent_def["system_prompt"]

        def node_function(state: Dict) -> Dict:
            """动态生成的节点函数"""

            # 构建消息
            messages = [SystemMessage(content=system_prompt)]
            messages.extend(state.get("messages", []))

            # 调用 LLM
            response = llm.invoke(messages)

            # 返回状态更新
            return {
                "messages": [response],
                f"{agent_id}_analysis": response.content
            }

        return node_function

    def load_all_agents(self) -> Dict[str, Callable]:
        """加载目录下所有智能体"""
        agents = {}
        for agent_file in self.agents_dir.glob("*.md"):
            agent_id = agent_file.stem
            agents[agent_id] = self.load_agent(agent_id)
        return agents
```

### 2. WorkflowBuilder (workflow_builder.py)

```python
"""
工作流构建器
运行时读取 workflow.yaml，构建 LangGraph StateGraph
"""
from pathlib import Path
from typing import Dict, Callable
import yaml
from langgraph.graph import StateGraph, START, END
from langgraph.checkpoint.postgres import PostgresSaver

from .agent_loader import AgentLoader

class WorkflowBuilder:
    """从 YAML 定义动态构建 LangGraph 工作流"""

    def __init__(
        self,
        agent_loader: AgentLoader,
        db_uri: str
    ):
        self.agent_loader = agent_loader
        self.checkpointer = PostgresSaver.from_conn_string(db_uri)

    def build_from_yaml(self, workflow_file: Path) -> StateGraph:
        """
        从 YAML 文件构建工作流图

        Args:
            workflow_file: workflow.yaml 路径

        Returns:
            编译后的 StateGraph
        """
        # 读取 YAML
        with open(workflow_file, 'r', encoding='utf-8') as f:
            workflow_def = yaml.safe_load(f)

        # 创建 StateGraph
        from .state_manager import APSState
        builder = StateGraph(APSState)

        # 加载所有引用的智能体
        agent_nodes = self._load_agents_from_workflow(workflow_def)

        # 添加节点
        for node_id, node_fn in agent_nodes.items():
            builder.add_node(node_id, node_fn)

        # 添加边
        self._add_edges(builder, workflow_def)

        # 编译图
        return builder.compile(checkpointer=self.checkpointer)

    def _load_agents_from_workflow(self, workflow_def: Dict) -> Dict[str, Callable]:
        """从工作流定义中提取并加载智能体"""
        agent_nodes = {}

        # 从 workflow.yaml 的 steps 中提取 agent IDs
        for step in workflow_def.get("steps", []):
            target = step.get("target", "")

            # 解析 target: "bmad/aps/agents/orchestrator.md"
            if "/agents/" in target:
                agent_id = Path(target).stem
                if agent_id not in agent_nodes:
                    agent_nodes[agent_id] = self.agent_loader.load_agent(agent_id)

        return agent_nodes

    def _add_edges(self, builder: StateGraph, workflow_def: Dict):
        """添加边（直接边和条件边）"""
        steps = workflow_def.get("steps", [])

        for i, step in enumerate(steps):
            step_id = step["step_id"]

            # 入口节点
            if i == 0:
                builder.add_edge(START, step_id)

            # 后续节点
            if i < len(steps) - 1:
                next_step = steps[i + 1]["step_id"]

                # 检查是否有条件路由
                if "routing" in step:
                    # 添加条件边
                    routing_fn = self._build_routing_function(step["routing"])
                    builder.add_conditional_edges(
                        step_id,
                        routing_fn,
                        self._extract_routing_targets(step["routing"])
                    )
                else:
                    # 直接边
                    builder.add_edge(step_id, next_step)
            else:
                # 最后一个节点 -> END
                builder.add_edge(step_id, END)

    def _build_routing_function(self, routing_config: Dict) -> Callable:
        """构建路由函数"""
        condition = routing_config.get("condition")

        def routing_function(state: Dict) -> str:
            """动态生成的路由函数"""
            # 简单的条件判断
            # TODO: 支持更复杂的表达式
            if eval(condition, {"state": state}):
                return routing_config["if_true"]
            else:
                return routing_config["if_false"]

        return routing_function

    def _extract_routing_targets(self, routing_config: Dict) -> Dict[str, str]:
        """提取路由目标映射"""
        return {
            routing_config["if_true"]: routing_config["if_true"],
            routing_config["if_false"]: routing_config["if_false"]
        }
```

### 3. FastAPI 集成 (api/main.py)

```python
"""
FastAPI 应用入口
"""
from fastapi import FastAPI, HTTPException, Depends
from fastapi.middleware.cors import CORSMiddleware
from pathlib import Path
import os

from .routes import scheduling, sessions, health
from ..runtime.agent_loader import AgentLoader
from ..runtime.workflow_builder import WorkflowBuilder

# 创建 FastAPI app
app = FastAPI(
    title="BMAD-APS API",
    version="1.0.0",
    description="APS LangGraph API Service"
)

# CORS
app.add_middleware(
    CORSMiddleware,
    allow_origins=["*"],  # 生产环境需要配置
    allow_credentials=True,
    allow_methods=["*"],
    allow_headers=["*"],
)

# 全局依赖
BMAD_DIR = Path(__file__).parent.parent / "bmad" / "aps"
AGENT_LOADER = AgentLoader(BMAD_DIR / "agents")
WORKFLOW_BUILDER = WorkflowBuilder(
    agent_loader=AGENT_LOADER,
    db_uri=os.getenv("DATABASE_URL")
)

# 路由
app.include_router(health.router, prefix="/api/v1", tags=["health"])
app.include_router(scheduling.router, prefix="/api/v1", tags=["scheduling"])
app.include_router(sessions.router, prefix="/api/v1", tags=["sessions"])

@app.on_event("startup")
async def startup():
    """启动时初始化"""
    print("🚀 BMAD-APS API starting...")

    # 预加载工作流（可选）
    workflow_file = BMAD_DIR / "workflows" / "scheduling-orchestration" / "workflow.yaml"
    if workflow_file.exists():
        graph = WORKFLOW_BUILDER.build_from_yaml(workflow_file)
        app.state.graph = graph
        print(f"✅ Workflow loaded: {workflow_file}")

    print("✅ BMAD-APS API ready!")

@app.get("/")
async def root():
    return {
        "service": "BMAD-APS API",
        "version": "1.0.0",
        "status": "running"
    }
```

### 4. API 端点示例 (api/routes/scheduling.py)

````python
"""
调度 API 端点
"""
from fastapi import APIRouter, HTTPException, Request
from pydantic import BaseModel
from typing import Optional
import uuid

router = APIRouter()

class SchedulingRequest(BaseModel):
    """调度请求"""
    user_input: str
    user_name: Optional[str] = "User"
    language: Optional[str] = "Chinese"

class SchedulingResponse(BaseModel):
    """调度响应"""
    session_id: str
    status: str
    result: Optional[dict] = None

@router.post("/scheduling", response_model=SchedulingResponse)
async def create_scheduling_task(
    request: SchedulingRequest,
    app: Request
):
    """
    创建调度任务

    示例请求:
    ```json
    {
        "user_input": "帮我设计一个车辆路径优化方案",
        "user_name": "张三",
        "language": "Chinese"
    }
    ```
    """
    # 生成会话ID
    session_id = str(uuid.uuid4())

    # 获取 LangGraph 图
    graph = app.app.state.graph

    # 初始状态
    initial_state = {
        "messages": [{"role": "user", "content": request.user_input}],
        "user_name": request.user_name,
        "communication_language": request.language
    }

    # 运行工作流
    try:
        config = {"configurable": {"thread_id": session_id}}
        result = graph.invoke(initial_state, config)

        return SchedulingResponse(
            session_id=session_id,
            status="completed",
            result=result
        )
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))

@router.get("/scheduling/{session_id}/stream")
````
