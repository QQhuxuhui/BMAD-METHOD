```python
from fastapi import FastAPI
from fastapi.middleware.cors import CORSMiddleware
from langgraph_service.api.v1 import health, info, workflows
from langgraph_service.config.settings import get_settings
from langgraph_service.middleware.logging import setup_logging
from langgraph_service.middleware.error_handler import add_exception_handlers

settings = get_settings()
setup_logging()

app = FastAPI(
    title="BMAD LangGraph Service",
    description="智能体工作流服务API",
    version="1.0.0",
    docs_url="/api/docs",
    redoc_url="/api/redoc"
)

# CORS配置

    CORSMiddleware,
    allow_origins=settings.cors_origins,
    allow_credentials=True,
    allow_methods=["*"],

# 注册路由
app.include_router(health.router)
app.include_router(info.router, prefix="/api/v1")
app.include_router(workflows.router, prefix="/api/v1")

# 异常处理
add_exception_handlers(app)

@app.on_event("startup")
async def startup():
    # 初始化数据库连接等
    pass

@app.on_event("shutdown")
async def shutdown():
    # 清理资源
    pass
```

### LangGraph集成示例

**example.py (工作流)**:

```python
from typing import TypedDict
from langgraph.graph import StateGraph, END
from langgraph.checkpoint.postgres import PostgresSaver

class WorkflowState(TypedDict):
    message: str
    result: str

def process_node(state: WorkflowState) -> WorkflowState:
    return {"result": f"Processed: {state['message']}"}

def create_example_workflow(checkpointer):
    workflow = StateGraph(WorkflowState)
    workflow.add_node("process", process_node)
    workflow.set_entry_point("process")
    workflow.add_edge("process", END)
    return workflow.compile(checkpointer=checkpointer)
```

### 健康检查端点

**health.py**:

```python
from fastapi import APIRouter, Depends
from sqlalchemy.ext.asyncio import AsyncSession
from redis import Redis

router = APIRouter(tags=["health"])

@router.get("/health")
async def health_check():
    return {"status": "ok"}

@router.get("/health/ready")
async def readiness_check(
    db: AsyncSession = Depends(get_db),
    redis: Redis = Depends(get_redis)
):
    # 检查数据库
    try:
        await db.execute("SELECT 1")
    except Exception as e:
        return {"status": "unhealthy", "database": "down"}

    # 检查Redis
    try:
        redis.ping()
    except Exception as e:
        return {"status": "unhealthy", "redis": "down"}

    return {"status": "ready", "database": "up", "redis": "up"}
```

### 编码标准

**命名约定** [Source: architecture/15-coding-standards.md]

- API路由：kebab-case（例如：`/api/v1/workflows/invoke`）
- Python类：PascalCase（例如：`WorkflowState`）
- Python函数：snake_case（例如：`create_example_workflow`）
- 环境变量：UPPER_SNAKE_CASE（例如：`DATABASE_URL`）

**关键规则**:

1. 所有异步操作使用async/await
2. 使用Pydantic进行数据验证
3. 使用structlog记录日志
4. 使用类型注解
