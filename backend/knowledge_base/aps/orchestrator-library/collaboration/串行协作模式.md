## 📖 模式概述

**核心理念**: 按照知识依赖链顺序执行，前一专家的输出作为后一专家的输入，确保高质量深度分析

**适用场景**:

- 质量要求高，需要深度分析
- 专家间依赖度高，需要知识传递
- 复杂问题需要层层递进分析

**优势**:

- **高质量**: 每一步都有前序知识支撑
- **深度分析**: 层层递进，逐步深化
- **知识传递**: 专家间形成知识链条

## 🔧 代码实现

```python
class SequentialCollaborationMode:
    """串行协作模式实现"""

    def __init__(self, expert_chain: List[Expert]):
        """
        expert_chain: 按执行顺序排列的专家列表
        典型顺序: 领域专家 → 约束专家 → 算法专家 → 目标专家
        """
        self.expert_chain = expert_chain
        self.execution_history = []

    def orchestrate(self, task: Dict) -> Dict:
        """
        串行协作编排流程

        工作流程:
        1. 领域识别阶段
        2. 约束建模阶段
        3. 算法设计阶段
        4. 目标优化阶段
        """
        current_context = {
            'original_task': task,
            'knowledge_chain': []
        }

        # 按顺序执行专家分析
        for expert in self.expert_chain:
            step_result = self._execute_expert_step(expert, current_context)

            # 记录执行历史
            self.execution_history.append({
                'expert': expert.name,
                'input': current_context.copy(),
                'output': step_result
            })

            # 更新上下文，传递给下一专家
            current_context['knowledge_chain'].append({
                'expert': expert.name,
                'contribution': step_result
            })

            # 检查是否需要提前终止
            if step_result.get('critical_issue'):
                return self._early_termination(step_result)

        # 所有专家完成后生成最终方案
        final_solution = self._generate_final_solution(current_context)

        return final_solution

    def _execute_expert_step(self, expert: Expert, context: Dict) -> Dict:
        """执行单个专家分析步骤"""
        # 准备专家输入
        expert_input = {
            'task': context['original_task'],
            'prior_knowledge': context['knowledge_chain'],
            'expert_focus': expert.specialty
        }

        # 执行专家分析
        expert_output = expert.analyze(expert_input)

        # 质量检查
        quality_check = self._validate_step_quality(expert_output, expert)

        return {
            'analysis': expert_output,
            'quality': quality_check,
            'timestamp': self._get_timestamp()
        }

    def _validate_step_quality(self, output: Dict, expert: Expert) -> Dict:
        """验证单步输出质量"""
        quality_metrics = {
            'completeness': 0.0,
            'consistency': 0.0,
            'actionability': 0.0
        }

        # 完整性检查
        required_fields = expert.get_required_output_fields()
        present_fields = [f for f in required_fields if f in output]
        quality_metrics['completeness'] = len(present_fields) / len(required_fields)

        # 一致性检查（与之前专家的输出）
        if self.execution_history:
            last_output = self.execution_history[-1]['output']['analysis']
            quality_metrics['consistency'] = self._check_consistency_with_prior(
                output, last_output
            )
        else:
            quality_metrics['consistency'] = 1.0

        # 可操作性检查
        quality_metrics['actionability'] = self._check_actionability(output)

        overall_quality = sum(quality_metrics.values()) / len(quality_metrics)

        return {
            'metrics': quality_metrics,
            'overall': overall_quality,
            'passed': overall_quality >= 0.8
        }

    def _check_consistency_with_prior(self, current: Dict, prior: Dict) -> float:
        """检查与前序专家输出的一致性"""
        # 示例：检查领域识别与约束建模的一致性

        # 如果前序专家识别为VRP问题，约束专家应包含车辆容量约束
        if 'domain' in prior and prior['domain'] == 'VRP':
            if 'constraints' in current:
                vrp_constraints = ['vehicle_capacity', 'time_window', 'distance']
                has_vrp_constraints = any(c in str(current['constraints'])
                                         for c in vrp_constraints)
                return 1.0 if has_vrp_constraints else 0.5

        # 默认一致性
        return 0.9

    def _check_actionability(self, output: Dict) -> float:
        """检查输出的可操作性"""
        actionable_indicators = [
            'recommended_algorithm' in output,
            'implementation_steps' in output,
            'parameters' in output,
            'expected_performance' in output
        ]

        return sum(actionable_indicators) / len(actionable_indicators)

    def _early_termination(self, step_result: Dict) -> Dict:
        """提前终止处理"""
        return {
            'status': 'terminated',
            'reason': step_result['critical_issue'],
            'completed_steps': len(self.execution_history),
            'partial_results': self.execution_history
        }

    def _generate_final_solution(self, context: Dict) -> Dict:
        """生成最终解决方案"""
        solution = {
            'domain_analysis': None,
            'constraint_model': None,
            'algorithm_design': None,
            'objective_function': None,
            'integration_quality': 0.0
        }

        # 从知识链中提取各专家贡献
        for knowledge in context['knowledge_chain']:
            expert_name = knowledge['expert']
            contribution = knowledge['contribution']['analysis']

            if 'domain' in expert_name.lower():
                solution['domain_analysis'] = contribution
            elif 'constraint' in expert_name.lower():
                solution['constraint_model'] = contribution
            elif 'algorithm' in expert_name.lower():
                solution['algorithm_design'] = contribution
            elif 'objective' in expert_name.lower():
                solution['objective_function'] = contribution

        # 计算整体集成质量
        quality_scores = [k['contribution']['quality']['overall']
                         for k in context['knowledge_chain']]
        solution['integration_quality'] = sum(quality_scores) / len(quality_scores)

        # 生成执行计划
        solution['execution_plan'] = self._create_execution_plan(solution)

        return solution

    def _create_execution_plan(self, solution: Dict) -> List[Dict]:
        """创建可执行计划"""
        plan = []

        # Step 1: 领域适配
        if solution['domain_analysis']:
            plan.append({
                'step': 1,
                'action': 'domain_adaptation',
                'details': solution['domain_analysis']
            })

        # Step 2: 约束建模
        if solution['constraint_model']:
            plan.append({
                'step': 2,
                'action': 'constraint_modeling',
                'details': solution['constraint_model']
            })

        # Step 3: 算法实现
        if solution['algorithm_design']:
            plan.append({
                'step': 3,
                'action': 'algorithm_implementation',
                'details': solution['algorithm_design']
            })

        # Step 4: 目标优化
        if solution['objective_function']:
            plan.append({
                'step': 4,
                'action': 'objective_optimization',
                'details': solution['objective_function']
            })

        return plan

    def _get_timestamp(self) -> str:
        """获取时间戳"""
        from datetime import datetime
        return datetime.now().isoformat()

    def get_performance_metrics(self) -> Dict:
        """串行模式性能指标"""
        return {
            'expected_time': 'N * T',  # N个专家，每个T时间
            'quality_level': 'high',
            'knowledge_depth': 'deep',
            'suitable_complexity': ['complex', 'very_complex']
        }
```

## 📋 使用示例

### 场景1: 复杂VRP问题深度分析

```python
# 创建专家链（按依赖顺序）
expert_chain = [
    DomainExpert(),      # 1. 领域识别
    ConstraintExpert(),  # 2. 约束建模
    AlgorithmExpert(),   # 3. 算法设计
    ObjectiveExpert()    # 4. 目标优化
]

sequential_mode = SequentialCollaborationMode(expert_chain)

# 定义复杂任务
task = {
    'description': '200个客户的多车型多目标VRPTW',
    'quality_requirement': 'high',
    'time_limit': '2小时'
}

# 执行串行协作
solution = sequential_mode.orchestrate(task)

print(f"串行协作完成，集成质量: {solution['integration_quality']:.2f}")
print(f"执行计划: {len(solution['execution_plan'])} 个步骤")
for step in solution['execution_plan']:
    print(f"  步骤{step['step']}: {step['action']}")
```

### 场景2: 质量优先的生产调度

```python
# 5专家深度分析
expert_chain = [
    DomainExpert(),      # 识别生产模式
    ConstraintExpert(),  # 建模工艺约束
    ObjectiveExpert(),   # 设计优化目标
    AlgorithmExpert(),   # 选择调度算法
    ExtensionExpert()    # 算法库扩展建议
]

sequential_mode = SequentialCollaborationMode(expert_chain)

task = {
    'description': '复杂工艺流程的生产调度',
    'quality': 'critical',
    'accuracy_requirement': '>95%'
}

solution = sequential_mode.orchestrate(task)

# 查看知识传递链
for i, history in enumerate(sequential_mode.execution_history):
    print(f"步骤{i+1} - {history['expert']}:")
    print(f"  质量: {history['output']['quality']['overall']:.2f}")
```

## 📊 性能特征

**时间效率**:

- 总时间: O(N \* T)，其中N为专家数，T为单专家平均时间
- 无并行加速
- 适合质量优先场景

**质量保证**:

- 深度分析: 高（层层递进，逐步深化）
- 一致性: 高（每步验证与前序知识的一致性）
- 知识传递: 完整（形成知识链条）

**适用边界**:

- ✅ 适合: 复杂到极复杂问题
- ✅ 适合: 质量要求高的场景
- ✅ 适合: 专家依赖度高的任务
- ❌ 不适合: 时间紧迫的场景
- ❌ 不适合: 简单独立的问题
