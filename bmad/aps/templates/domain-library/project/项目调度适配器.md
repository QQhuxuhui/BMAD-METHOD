    def allocate_resources(self, tasks: List[Task], resources: Dict) -> Dict:
        """资源分配"""
        allocation = {}
        for task in tasks:
            allocation[task.id] = []
            for resource_type in task.resources:
                if resource_type in resources and resources[resource_type] > 0:
                    allocation[task.id].append(resource_type)
                    resources[resource_type] -= 1
        return allocation
