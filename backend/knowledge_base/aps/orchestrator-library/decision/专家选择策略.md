## 📖 策略概述

**核心理念**: 根据问题特征和复杂度，智能选择最优专家组合，确保能力覆盖和协作效率

**选择原则**:

- 能力覆盖：确保所有问题维度有专家覆盖
- 最小充分：避免专家冗余，保持精简高效
- 依赖匹配：考虑专家间的依赖关系
- 协作兼容：选择协作能力强的组合

## 🔧 代码实现

```python
from typing import List, Dict, Set
from dataclasses import dataclass

@dataclass
class ExpertCapability:
    """专家能力模型"""
    name: str
    specialty: str
    capabilities: Set[str]
    dependency: List[str]  # 依赖的其他专家
    collaboration_score: float  # 协作能力评分 (0-1)
    workload_factor: float  # 工作负载系数

class ExpertSelectionStrategy:
    """专家选择策略"""

    def __init__(self):
        self.experts = self._initialize_experts()

    def _initialize_experts(self) -> Dict[str, ExpertCapability]:
        """初始化专家能力模型"""
        return {
            'domain': ExpertCapability(
                name='领域应用专家',
                specialty='domain_analysis',
                capabilities={
                    'domain_identification',
                    'business_rules',
                    'best_practices',
                    'industry_patterns'
                },
                dependency=[],
                collaboration_score=0.9,
                workload_factor=1.0
            ),
            'constraint': ExpertCapability(
                name='约束模式专家',
                specialty='constraint_modeling',
                capabilities={
                    'constraint_identification',
                    'constraint_validation',
                    'feasibility_check',
                    'constraint_propagation'
                },
                dependency=['domain'],
                collaboration_score=0.85,
                workload_factor=1.2
            ),
            'algorithm': ExpertCapability(
                name='调度算法专家',
                specialty='algorithm_design',
                capabilities={
                    'algorithm_selection',
                    'performance_analysis',
                    'complexity_analysis',
                    'heuristic_design'
                },
                dependency=['domain', 'constraint'],
                collaboration_score=0.9,
                workload_factor=1.5
            ),
            'objective': ExpertCapability(
                name='目标优化专家',
                specialty='objective_optimization',
                capabilities={
                    'objective_design',
                    'multi_objective_optimization',
                    'tradeoff_analysis',
                    'evaluation_framework'
                },
                dependency=['constraint'],
                collaboration_score=0.8,
                workload_factor=1.0
            ),
            'extension': ExpertCapability(
                name='算法扩展专家',
                specialty='algorithm_extension',
                capabilities={
                    'algorithm_library_design',
                    'knowledge_expansion',
                    'pattern_extraction',
                    'documentation_generation'
                },
                dependency=['algorithm'],
                collaboration_score=0.75,
                workload_factor=0.8
            )
        }

    def select(self, problem: Dict, complexity_level: str) -> Dict:
        """
        选择最优专家组合

        参数:
        - problem: 问题描述
        - complexity_level: 复杂度等级 (simple/medium/complex/very_complex)

        返回:
        {
            'selected_experts': [...],
            'roles': {...},
            'execution_order': [...],
            'estimated_workload': float
        }
        """
        # 识别需要的能力
        required_capabilities = self._identify_required_capabilities(problem)

        # 根据复杂度选择基础专家组
        base_experts = self._select_base_experts(complexity_level)

        # 根据能力需求调整专家组
        selected_experts = self._adjust_by_capabilities(
            base_experts, required_capabilities
        )

        # 验证依赖关系
        selected_experts = self._validate_dependencies(selected_experts)

        # 分配角色
        roles = self._assign_roles(selected_experts, problem)

        # 确定执行顺序
        execution_order = self._determine_execution_order(selected_experts)

        # 估算工作负载
        workload = self._estimate_workload(selected_experts)

        return {
            'selected_experts': selected_experts,
            'roles': roles,
            'execution_order': execution_order,
            'estimated_workload': workload,
            'collaboration_score': self._calculate_collaboration_score(selected_experts)
        }

    def _identify_required_capabilities(self, problem: Dict) -> Set[str]:
        """识别问题需要的能力"""
        capabilities = set()

        # 基于问题类型识别
        if 'domain' in problem or 'industry' in problem:
            capabilities.add('domain_identification')
            capabilities.add('industry_patterns')

        # 基于约束识别
        if 'constraints' in problem and len(problem['constraints']) > 0:
            capabilities.add('constraint_identification')
            capabilities.add('constraint_validation')

        # 基于目标识别
        if 'objectives' in problem:
            if len(problem['objectives']) > 1:
                capabilities.add('multi_objective_optimization')
                capabilities.add('tradeoff_analysis')
            else:
                capabilities.add('objective_design')

        # 基于算法需求
        if problem.get('complexity_class') in ['NP-hard', 'NP-complete']:
            capabilities.add('algorithm_selection')
            capabilities.add('heuristic_design')

        # 基于扩展需求
        if problem.get('require_extension'):
            capabilities.add('algorithm_library_design')
            capabilities.add('knowledge_expansion')

        return capabilities

    def _select_base_experts(self, complexity_level: str) -> List[str]:
        """根据复杂度选择基础专家组"""
        if complexity_level == 'simple':
            return ['algorithm', 'constraint']
        elif complexity_level == 'medium':
            return ['domain', 'algorithm', 'constraint']
        elif complexity_level == 'complex':
            return ['domain', 'constraint', 'algorithm', 'objective']
        else:  # very_complex
            return ['domain', 'constraint', 'objective', 'algorithm', 'extension']

    def _adjust_by_capabilities(self, base_experts: List[str],
                                required_caps: Set[str]) -> List[str]:
        """根据能力需求调整专家组"""
        selected = set(base_experts)

        # 检查是否覆盖所有必需能力
        covered_caps = set()
        for expert_id in selected:
            covered_caps.update(self.experts[expert_id].capabilities)

        # 添加缺失能力的专家
        missing_caps = required_caps - covered_caps
        if missing_caps:
            for expert_id, expert in self.experts.items():
                if expert_id not in selected:
                    if expert.capabilities & missing_caps:
                        selected.add(expert_id)
                        covered_caps.update(expert.capabilities)

        return list(selected)

    def _validate_dependencies(self, experts: List[str]) -> List[str]:
        """验证并补充依赖的专家"""
        selected = set(experts)
        changed = True

        while changed:
            changed = False
            for expert_id in list(selected):
                for dep in self.experts[expert_id].dependency:
                    if dep not in selected:
                        selected.add(dep)
                        changed = True

        return list(selected)

    def _assign_roles(self, experts: List[str], problem: Dict) -> Dict:
        """为专家分配角色"""
        roles = {}

        for expert_id in experts:
            if expert_id == 'domain':
                roles[expert_id] = 'lead_analyst'  # 领域主导分析
            elif expert_id == 'algorithm':
                if problem.get('complexity_class') in ['NP-hard', 'NP-complete']:
                    roles[expert_id] = 'core_designer'  # 核心设计者
                else:
                    roles[expert_id] = 'implementer'  # 实现者
            elif expert_id == 'objective':
                if len(problem.get('objectives', [])) > 2:
                    roles[expert_id] = 'optimization_lead'  # 优化主导
                else:
                    roles[expert_id] = 'evaluator'  # 评估者
            else:
                roles[expert_id] = 'specialist'  # 专业顾问

        return roles

    def _determine_execution_order(self, experts: List[str]) -> List[str]:
        """确定执行顺序（拓扑排序）"""
        # 构建依赖图
        graph = {expert_id: set(self.experts[expert_id].dependency)
                for expert_id in experts}

        # 拓扑排序
        order = []
        visited = set()

        def visit(node):
            if node in visited:
                return
            visited.add(node)
            for dep in graph.get(node, []):
                if dep in experts:
                    visit(dep)
            order.append(node)

        for expert_id in experts:
            visit(expert_id)

        return order

    def _estimate_workload(self, experts: List[str]) -> float:
        """估算总工作负载"""
        total_workload = sum(
            self.experts[expert_id].workload_factor
            for expert_id in experts
        )
        return total_workload

    def _calculate_collaboration_score(self, experts: List[str]) -> float:
        """计算团队协作评分"""
        if not experts:
            return 0.0

        scores = [self.experts[expert_id].collaboration_score
                 for expert_id in experts]
        return sum(scores) / len(scores)
```

## 📋 使用示例

### 场景1: 复杂VRP问题专家选择

```python
selector = ExpertSelectionStrategy()

# 定义问题
vrp_problem = {
    'domain': 'logistics',
    'industry': 'cold_chain',
    'constraints': [
        'vehicle_capacity',
        'time_window',
        'temperature_control',
        'driver_working_hours',
        'customer_priority'
    ],
    'objectives': ['minimize_cost', 'minimize_time', 'maximize_service'],
    'complexity_class': 'NP-hard',
    'require_extension': False
}

# 选择专家
result = selector.select(vrp_problem, complexity_level='complex')

print(f"选中专家: {result['selected_experts']}")
print(f"执行顺序: {result['execution_order']}")
print(f"专家角色:")
for expert, role in result['roles'].items():
    print(f"  {expert}: {role}")
print(f"预计工作负载: {result['estimated_workload']:.2f}")
print(f"协作评分: {result['collaboration_score']:.2f}")
```

输出示例：

```
选中专家: ['domain', 'constraint', 'algorithm', 'objective']
执行顺序: ['domain', 'constraint', 'objective', 'algorithm']
专家角色:
  domain: lead_analyst
  constraint: specialist
  algorithm: core_designer
  objective: optimization_lead
预计工作负载: 4.70
协作评分: 0.86
```

### 场景2: 简单排班问题专家选择

```python
simple_problem = {
    'constraints': ['shift_coverage', 'rest_time'],
    'objectives': ['minimize_cost'],
    'complexity_class': 'P'
}

result = selector.select(simple_problem, complexity_level='simple')
print(f"选中专家: {result['selected_experts']}")
print(f"执行顺序: {result['execution_order']}")

```

## 📊 专家选择矩阵

| 复杂度     | 基础专家组          | 可选扩展 | 协作模式 |
| ---------- | ------------------- | -------- | -------- |
| **简单**   | 算法+约束           | -        | 并行     |
| **中等**   | 领域+算法+约束      | 目标     | 混合     |
| **复杂**   | 领域+约束+算法+目标 | 扩展     | 串行     |
| **极复杂** | 全专家团队          | -        | 串行     |
