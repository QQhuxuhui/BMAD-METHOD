## 📖 约束概述

### 定义

**合规性约束**（Compliance Constraint）规定业务操作必须符合法律法规、行业标准、内部政策的要求，确保合法合规运营。

**数学定义**：

```
对于操作 o 和规则集 R = {r_1, r_2, ..., r_n}

合规性约束:
  ∀r_i ∈ R: complies(o, r_i) = true

违规惩罚:
  penalty(o) = ∑ severity(r_i) × violation(o, r_i)
```

### 约束类型

#### 1. **法律法规约束** (Legal Compliance)

```
必须遵守的法律法规

例如:
  - 劳动法：加班时间 ≤ 36小时/月
  - 环保法：排放量 ≤ 标准值
  - 数据保护法：GDPR, CCPA
  - 安全生产法：安全操作规程
```

#### 2. **行业标准约束** (Industry Standards)

```
行业规范和最佳实践

例如:
  - ISO 9001：质量管理体系
  - ISO 27001：信息安全管理
  - GMP：药品生产质量管理规范
  - HACCP：食品安全管理体系
```

#### 3. **内部政策约束** (Internal Policy)

```
企业内部规章制度

例如:
  - 采购政策：3家供应商比价
  - 审批流程：超过10万需总经理审批
  - 质量政策：合格率 ≥ 99%
```

#### 4. **审计追溯约束** (Audit Trail)

```
可审计性要求

要求:
  - 完整记录操作日志
  - 关键决策留痕
  - 数据不可篡改
  - 保留期限要求
```

## 💡 核心理论

### 规则引擎 (Rule Engine)

```python
class Rule:
    """规则定义"""

    def __init__(self, rule_id, name, condition, action, severity='medium'):
        self.rule_id = rule_id
        self.name = name
        self.condition = condition  # 条件函数
        self.action = action  # 违规时的处理
        self.severity = severity  # 严重程度: low, medium, high, critical

    def evaluate(self, context):
        """评估规则"""
        try:
            return self.condition(context)
        except Exception as e:
            return False

class RuleEngine:
    """规则引擎"""

    def __init__(self):
        self.rules = []

    def add_rule(self, rule):
        """添加规则"""
        self.rules.append(rule)

    def evaluate_all(self, context):
        """评估所有规则"""
        violations = []

        for rule in self.rules:
            if not rule.evaluate(context):
                violations.append({
                    'rule_id': rule.rule_id,
                    'rule_name': rule.name,
                    'severity': rule.severity,
                    'action': rule.action
                })

        return violations
```

### 审计日志模式

```python
import hashlib
import json
from datetime import datetime

class AuditLog:
    """审计日志（不可篡改）"""

    def __init__(self):
        self.logs = []
        self.previous_hash = "0"

    def add_log(self, user, action, resource, details):
        """添加日志记录"""
        log_entry = {
            'timestamp': datetime.now().isoformat(),
            'user': user,
            'action': action,
            'resource': resource,
            'details': details,
            'previous_hash': self.previous_hash
        }

        # 计算当前记录的哈希
        log_data = json.dumps(log_entry, sort_keys=True)
        current_hash = hashlib.sha256(log_data.encode()).hexdigest()

        log_entry['hash'] = current_hash

        self.logs.append(log_entry)
        self.previous_hash = current_hash

        return log_entry

    def verify_integrity(self):
        """验证日志完整性"""
        previous_hash = "0"

        for log in self.logs:
            # 重新计算哈希
            temp_log = log.copy()
            stored_hash = temp_log.pop('hash')

            if temp_log['previous_hash'] != previous_hash:
                return False, f"Log at {temp_log['timestamp']} has broken chain"

            log_data = json.dumps(temp_log, sort_keys=True)
            calculated_hash = hashlib.sha256(log_data.encode()).hexdigest()

            if calculated_hash != stored_hash:
                return False, f"Log at {temp_log['timestamp']} has been tampered"

            previous_hash = stored_hash

        return True, "Audit log integrity verified"
```

## 🔧 代码实现

```python
from typing import Dict, List, Tuple, Callable, Any
from dataclasses import dataclass, field
from datetime import datetime, timedelta
from enum import Enum

class Severity(Enum):
    """违规严重程度"""
    LOW = "low"
    MEDIUM = "medium"
    HIGH = "high"
    CRITICAL = "critical"

@dataclass
class ComplianceRule:
    """合规规则"""
    id: str
    name: str
    description: str
    category: str  # legal, standard, policy
    condition: Callable[[Any], bool]
    severity: Severity = Severity.MEDIUM
    penalty: float = 0

@dataclass
class ComplianceViolation:
    """合规违规"""
    rule_id: str
    rule_name: str
    severity: Severity
    description: str
    timestamp: datetime = field(default_factory=datetime.now)
    penalty: float = 0
    remediation: str = ""

class ComplianceConstraintHandler:
    """合规性约束处理器"""

    def __init__(self):
        self.rules: List[ComplianceRule] = []
        self.audit_log = []

    def add_rule(self, rule: ComplianceRule):
        """添加合规规则"""
        self.rules.append(rule)

    def validate(self, operation: Dict[str, Any]) -> Tuple[bool, List[ComplianceViolation]]:
        """
        验证操作是否合规

        参数:
          operation: 待验证的操作

        返回:
          (is_compliant, violations)
        """
        violations = []

        for rule in self.rules:
            try:
                is_compliant = rule.condition(operation)

                if not is_compliant:
                    violation = ComplianceViolation(
                        rule_id=rule.id,
                        rule_name=rule.name,
                        severity=rule.severity,
                        description=f"违反规则: {rule.description}",
                        penalty=rule.penalty
                    )
                    violations.append(violation)

                    # 记录审计日志
                    self._log_violation(operation, violation)

            except Exception as e:
                # 规则评估失败也视为违规
                violation = ComplianceViolation(
                    rule_id=rule.id,
                    rule_name=rule.name,
                    severity=Severity.CRITICAL,
                    description=f"规则评估失败: {str(e)}",
                    penalty=100
                )
                violations.append(violation)

        is_compliant = len(violations) == 0
        return is_compliant, violations

    def _log_violation(self, operation: Dict[str, Any], violation: ComplianceViolation):
        """记录违规审计日志"""
        log_entry = {
            'timestamp': datetime.now().isoformat(),
            'operation': operation.get('id', 'unknown'),
            'violation': {
                'rule_id': violation.rule_id,
                'rule_name': violation.rule_name,
                'severity': violation.severity.value,
                'description': violation.description
            }
        }
        self.audit_log.append(log_entry)

    def calculate_compliance_score(self, violations: List[ComplianceViolation]) -> float:
        """
        计算合规分数 (0-100)

        100: 完全合规
        0: 严重违规
        """
        if not violations:
            return 100.0

        # 根据严重程度计算扣分
        severity_weights = {
            Severity.LOW: 5,
            Severity.MEDIUM: 15,
            Severity.HIGH: 30,
            Severity.CRITICAL: 50
        }

        total_deduction = sum(
            severity_weights.get(v.severity, 10)
            for v in violations
        )

        score = max(0, 100 - total_deduction)
        return score

    def generate_compliance_report(self, violations: List[ComplianceViolation]) -> str:
        """生成合规报告"""
        if not violations:
            return "✅ 完全合规，无违规记录"

        report = f"⚠️ 发现 {len(violations)} 项违规\n\n"

        # 按严重程度分类
        by_severity = {}
        for v in violations:
            severity = v.severity.value
            if severity not in by_severity:
                by_severity[severity] = []
            by_severity[severity].append(v)

        # 生成报告
        for severity in ['critical', 'high', 'medium', 'low']:
            if severity in by_severity:
                items = by_severity[severity]
                report += f"{'🔴' if severity == 'critical' else '🟡'} {severity.upper()} ({len(items)}项):\n"

                for item in items:
                    report += f"  - {item.rule_name}: {item.description}\n"

                report += "\n"

        # 合规分数
        score = self.calculate_compliance_score(violations)
        report += f"📊 合规分数: {score}/100\n"

        return report

```

## 📊 应用场景

### 场景1: 劳动法合规检查

```python
handler = ComplianceConstraintHandler()

# 添加劳动法规则
overtime_rule = ComplianceRule(
    id='labor_law_overtime',
    name='加班时间限制',
    description='每月加班不得超过36小时',
    category='legal',
    condition=lambda op: op.get('overtime_hours', 0) <= 36,
    severity=Severity.CRITICAL,
    penalty=1000
)

rest_rule = ComplianceRule(
    id='labor_law_rest',
    name='每周休息日',
    description='每周至少休息1天',
    category='legal',
    condition=lambda op: op.get('rest_days', 0) >= 1,
    severity=Severity.HIGH,
    penalty=500
)

handler.add_rule(overtime_rule)
handler.add_rule(rest_rule)

# 验证员工排班
schedule = {
    'id': 'employee_123',
    'overtime_hours': 40,  # 违规：超过36小时
    'rest_days': 1         # 合规
}

is_compliant, violations = handler.validate(schedule)

if not is_compliant:
    report = handler.generate_compliance_report(violations)
    print(report)
```

## 🎯 性能优化

### 规则缓存和索引

```python
class OptimizedComplianceHandler(ComplianceConstraintHandler):
    """优化版合规处理器"""

    def __init__(self):
        super().__init__()
        self.rule_index = {}  # 按类别索引
        self.rule_cache = {}   # 评估结果缓存

    def add_rule(self, rule: ComplianceRule):
        super().add_rule(rule)

        # 构建索引
        if rule.category not in self.rule_index:
            self.rule_index[rule.category] = []
        self.rule_index[rule.category].append(rule)

    def validate_by_category(self, operation: Dict[str, Any],
                            category: str) -> Tuple[bool, List[ComplianceViolation]]:
        """按类别验证（更高效）"""
        violations = []

        rules = self.rule_index.get(category, [])

        for rule in rules:
            if not rule.condition(operation):
                violation = ComplianceViolation(
                    rule_id=rule.id,
                    rule_name=rule.name,
                    severity=rule.severity,
                    description=f"违反规则: {rule.description}",
                    penalty=rule.penalty
                )
                violations.append(violation)

        return len(violations) == 0, violations
```
