## 📖 目标概述

### 定义

**总延迟时间**（Total Tardiness）是所有任务超过截止期的时间总和，衡量准时交付的性能。

**数学定义**：

```
对于调度方案 S，任务集 J = {j_1, j_2, ..., j_n}

Total Tardiness = ∑ T_j

其中:
  T_j = max{0, C_j - d_j}
  C_j = 任务 j 的完成时间
  d_j = 任务 j 的截止期

目标:
  minimize Total Tardiness
```

### 延迟衡量指标

#### 1. **单任务延迟** (Job Tardiness)

```
T_j = max{0, C_j - d_j}

特点:
  - 提前完成不计分 (C_j ≤ d_j → T_j = 0)
  - 延迟时间线性累积
```

#### 2. **延迟任务数** (Number of Tardy Jobs)

```
∑ U_j，其中 U_j = 1 if T_j > 0 else 0

用途: 最小化逾期订单数量
```

#### 3. **最大延迟** (Maximum Tardiness)

```
T_max = max{T_j | j ∈ J}

用途: 控制最差延迟情况
```

## 💡 核心理论

### EDD规则 (Earliest Due Date)

```python
def edd_schedule(jobs):
    """
    EDD规则: 按截止期升序排列

    性质:
    - 最小化最大延迟 (T_max)
    - 不一定最小化总延迟
    """
    # 按截止期排序
    sorted_jobs = sorted(jobs, key=lambda j: j.get('deadline', float('inf')))

    schedule = []
    current_time = 0

    for job in sorted_jobs:
        start_time = current_time
        duration = job.get('duration', 0)
        finish_time = start_time + duration
        deadline = job.get('deadline', float('inf'))

        tardiness = max(0, finish_time - deadline)

        schedule.append({
            'job_id': job.get('id'),
            'start_time': start_time,
            'finish_time': finish_time,
            'deadline': deadline,
            'tardiness': tardiness
        })

        current_time = finish_time

    return schedule
```

### MDD规则 (Modified Due Date)

```python
def mdd_schedule(jobs):
    """
    MDD规则: 动态优先级调度

    优先级 = min{d_j, C_j + p_j}

    其中:
      d_j = 截止期
      C_j = 当前时间
      p_j = 处理时间
    """
    remaining_jobs = jobs.copy()
    schedule = []
    current_time = 0

    while remaining_jobs:
        # 计算每个任务的MDD优先级
        priorities = []
        for job in remaining_jobs:
            duration = job.get('duration', 0)
            deadline = job.get('deadline', float('inf'))
            priority = min(deadline, current_time + duration)
            priorities.append((priority, job))

        # 选择优先级最高（值最小）的任务
        _, next_job = min(priorities, key=lambda x: x[0])

        # 调度该任务
        start_time = current_time
        duration = next_job.get('duration', 0)
        finish_time = start_time + duration
        deadline = next_job.get('deadline', float('inf'))
        tardiness = max(0, finish_time - deadline)

        schedule.append({
            'job_id': next_job.get('id'),
            'start_time': start_time,
            'finish_time': finish_time,
            'deadline': deadline,
            'tardiness': tardiness
        })

        current_time = finish_time
        remaining_jobs.remove(next_job)

    return schedule
```

## 🔧 代码实现

```python
from typing import Dict, List
from dataclasses import dataclass

@dataclass
class Job:
    """任务定义"""
    id: str
    duration: float
    deadline: float
    weight: float = 1.0  # 权重（用于加权延迟）

class TotalTardinessObjective:
    """总延迟时间目标函数"""

    def __init__(self, weight: float = 1.0):
        self.name = "total_tardiness"
        self.optimization_type = "minimize"
        self.weight = weight
        self.evaluation_history = []

    def evaluate(self, solution: List[Dict]) -> float:
        """
        评估调度方案的总延迟

        参数:
          solution: 任务调度序列

        返回:
          总延迟时间
        """
        total_tardiness = 0.0

        for task in solution:
            finish_time = task.get('finish_time', 0)
            deadline = task.get('deadline', float('inf'))

            # 计算延迟
            tardiness = max(0, finish_time - deadline)
            total_tardiness += tardiness

        self.evaluation_history.append({
            'value': total_tardiness,
            'num_tardy_jobs': sum(1 for t in solution if t.get('finish_time', 0) > t.get('deadline', float('inf')))
        })

        return total_tardiness

    def calculate_metrics(self, solution: List[Dict]) -> Dict:
        """
        计算延迟相关指标

        返回:
          {
            'total_tardiness': 总延迟
            'num_tardy_jobs': 延迟任务数
            'max_tardiness': 最大延迟
            'avg_tardiness': 平均延迟
            'on_time_rate': 准时率
          }
        """
        tardiness_values = []
        num_tardy = 0
        total_jobs = len(solution)

        for task in solution:
            finish_time = task.get('finish_time', 0)
            deadline = task.get('deadline', float('inf'))
            tardiness = max(0, finish_time - deadline)

            tardiness_values.append(tardiness)
            if tardiness > 0:
                num_tardy += 1

        total_tardiness = sum(tardiness_values)
        max_tardiness = max(tardiness_values) if tardiness_values else 0
        avg_tardiness = total_tardiness / total_jobs if total_jobs > 0 else 0
        on_time_rate = (total_jobs - num_tardy) / total_jobs if total_jobs > 0 else 1.0

        return {
            'total_tardiness': total_tardiness,
            'num_tardy_jobs': num_tardy,
            'max_tardiness': max_tardiness,
            'avg_tardiness': avg_tardiness,
            'on_time_rate': on_time_rate * 100  # 百分比
        }

    def edd_schedule(self, jobs: List[Job]) -> List[Dict]:
        """EDD规则调度"""
        sorted_jobs = sorted(jobs, key=lambda j: j.deadline)

        schedule = []
        current_time = 0

        for job in sorted_jobs:
            start_time = current_time
            finish_time = start_time + job.duration
            tardiness = max(0, finish_time - job.deadline)

            schedule.append({
                'job_id': job.id,
                'start_time': start_time,
                'finish_time': finish_time,
                'deadline': job.deadline,
                'tardiness': tardiness
            })

            current_time = finish_time

        return schedule
```

## 📊 应用场景

### 场景1: 订单交付优化

```python
# 订单列表
orders = [
    Job(id='O1', duration=3, deadline=10),
    Job(id='O2', duration=5, deadline=8),
    Job(id='O3', duration=2, deadline=12),
    Job(id='O4', duration=4, deadline=15),
    Job(id='O5', duration=3, deadline=9)
]

# 使用EDD规则
objective = TotalTardinessObjective()
schedule = objective.edd_schedule(orders)

# 评估结果
metrics = objective.calculate_metrics(schedule)

print("调度结果:")
for task in schedule:
    print(f"订单 {task['job_id']}: "
          f"开始={task['start_time']}, "
          f"完成={task['finish_time']}, "
          f"截止={task['deadline']}, "
          f"延迟={task['tardiness']}")

print(f"\n性能指标:")
print(f"总延迟: {metrics['total_tardiness']}")
print(f"延迟订单数: {metrics['num_tardy_jobs']}")
print(f"准时率: {metrics['on_time_rate']:.1f}%")
```
